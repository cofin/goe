```python
# SPDX-FileCopyrightText: 2026 The GOE Authors

"""OpenTelemetry span and metric helpers for GOE offload chunks, transport engines, and OS commands."""

from __future__ import annotations

import os
from collections.abc import Iterator, Mapping, Sequence
from contextlib import contextmanager
from typing import Any
from urllib.parse import urlparse

from opentelemetry.trace import Span, SpanKind, Status, StatusCode

from goe.observability.context import get_current_trace_context, inject_env_context
from goe.observability.runtime import get_observability_runtime, get_tracer

__all__ = [
    "finalize_chunk_span",
    "finalize_transport_span",
    "inject_spark_trace_properties",
    "traced_offload_chunk",
    "traced_os_cmd",
    "traced_transport",
]


@contextmanager
def traced_offload_chunk(
    chunk_number: int,
    source_owner: str,
    source_table: str,
    target_db: str,
    target_table: str,
    partition_count: int = 0,
    dry_run: bool = False,
) -> Iterator[Span]:
    """Wrap an offload chunk execution in a goe.chunk span."""
    tracer = get_tracer()
    with tracer.start_as_current_span("goe.chunk", kind=SpanKind.INTERNAL) as span:
        if span.is_recording():
            span.set_attribute("goe.chunk.number", chunk_number)
            span.set_attribute("goe.chunk.partition_count", partition_count)
            span.set_attribute("goe.source.owner", source_owner)
            span.set_attribute("goe.source.table", source_table)
            span.set_attribute("goe.target.db", target_db)
            span.set_attribute("goe.target.table", target_table)
            span.set_attribute("goe.chunk.dry_run", dry_run)
        try:
            yield span
            if span.is_recording():
                span.set_status(Status(StatusCode.OK))
        except Exception as exc:
            if span.is_recording():
                span.record_exception(exc)
                span.set_status(Status(StatusCode.ERROR, str(exc)))
            raise


def finalize_chunk_span(
    span: Span,
    backend: str,
    rows_staged: int | None,
    frontend_bytes: int | None,
    transport_bytes: int | None,
    backend_bytes: int | None,
) -> None:
    """Record chunk row/byte counts on the chunk span and update goe.offload.bytes_loaded metric."""
    if span.is_recording():
        if rows_staged is not None:
            span.set_attribute("goe.chunk.rows_staged", rows_staged)
        if frontend_bytes is not None:
            span.set_attribute("goe.chunk.frontend_bytes", frontend_bytes)
        if transport_bytes is not None:
            span.set_attribute("goe.chunk.transport_bytes", transport_bytes)
        if backend_bytes is not None:
            span.set_attribute("goe.chunk.backend_bytes", backend_bytes)
    get_observability_runtime().metrics.record_bytes_loaded(backend=backend, bytes_loaded=backend_bytes)


@contextmanager
def traced_transport(
    transport_method: str,
    source_owner: str,
    source_table: str,
    target_db: str,
    target_table: str,
    staging_format: str,
    staging_location: str | None = None,
    scn: int | str | None = None,
    parallelism: int = 1,
    dry_run: bool = False,
) -> Iterator[Span]:
    """Wrap an OffloadTransport.transport() invocation in a goe.transport.<method> span."""
    method_clean = (transport_method or "unknown").lower()
    tracer = get_tracer()
    with tracer.start_as_current_span(f"goe.transport.{method_clean}", kind=SpanKind.CLIENT) as span:
        if span.is_recording():
            span.set_attribute("goe.transport.method", transport_method)
            span.set_attribute("goe.transport.source_owner", source_owner)
            span.set_attribute("goe.transport.source_table", source_table)
            span.set_attribute("goe.transport.target_db", target_db)
            span.set_attribute("goe.transport.target_table", target_table)
            span.set_attribute("goe.transport.staging_format", staging_format or "unknown")
            span.set_attribute("goe.transport.parallelism", parallelism)
            span.set_attribute("goe.transport.dry_run", dry_run)
            if staging_location:
                span.set_attribute("goe.transport.staging_location", staging_location)
            if scn is not None:
                span.set_attribute("goe.transport.scn", str(scn))
        try:
            yield span
            if span.is_recording():
                span.set_status(Status(StatusCode.OK))
        except Exception as exc:
            if span.is_recording():
                span.record_exception(exc)
                span.set_status(Status(StatusCode.ERROR, str(exc)))
            raise


def finalize_transport_span(
    span: Span,
    transport_method: str,
    backend: str,
    source_owner: str,
    rows_imported: int | None,
    bytes_staged: int | None,
    staging_format: str,
    staging_location: str | None = None,
) -> None:
    """Attach rows_imported and bytes_staged to the transport span and record transport metrics."""
    if span.is_recording():
        if rows_imported is not None:
            span.set_attribute("goe.transport.rows_imported", rows_imported)
        if bytes_staged is not None:
            span.set_attribute("goe.transport.bytes_staged", bytes_staged)
    scheme = urlparse(staging_location).scheme if staging_location else "local"
    get_observability_runtime().metrics.record_transport(
        transport_method=transport_method,
        backend=backend,
        source_owner=source_owner,
        rows=rows_imported,
        bytes_staged=bytes_staged,
        storage_format=staging_format,
        scheme=scheme or "local",
    )


@contextmanager
def traced_os_cmd(
    cmd: Sequence[str],
    dry_run: bool = False,
    base_env: Mapping[str, str] | None = None,
) -> Iterator[ tuple[Span, dict[str, str]] ]:
    """Wrap an external OS command execution in a goe.os_cmd span and return a TRACEPARENT-enriched env dict."""
    executable = os.path.basename(cmd[0]) if cmd else "unknown"
    tracer = get_tracer()
    env_with_trace = inject_env_context(base_env if base_env is not None else os.environ)
    with tracer.start_as_current_span(f"goe.os_cmd.{executable}", kind=SpanKind.CLIENT) as span:
        if span.is_recording():
            span.set_attribute("process.executable.name", executable)
            span.set_attribute("goe.os_cmd.dry_run", dry_run)
        try:
            yield span, env_with_trace
            if span.is_recording():
                span.set_status(Status(StatusCode.OK))
        except Exception as exc:
            if span.is_recording():
                span.record_exception(exc)
                span.set_status(Status(StatusCode.ERROR, str(exc)))
            raise


def inject_spark_trace_properties(spark_properties: dict[str, Any]) -> dict[str, Any]:
    """Inject TRACEPARENT and GOE execution_id into Spark driver and executor configuration properties."""
    enriched = dict(spark_properties)
    snap = get_current_trace_context()
    if snap.traceparent:
        enriched.setdefault("spark.driverEnv.TRACEPARENT", snap.traceparent)
        enriched.setdefault("spark.executorEnv.TRACEPARENT", snap.traceparent)
    if snap.execution_id:
        enriched.setdefault("spark.goe.executionId", snap.execution_id)
    return enriched
```
