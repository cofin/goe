```python
# SPDX-FileCopyrightText: 2026 The GOE Authors
# SPDX-License-Identifier: Apache-2.0

"""Database observability configuration, statement observer, and Oracle session trace tagging."""

from __future__ import annotations

import logging
from typing import Any

from sqlspec.observability import (
    LoggingConfig,
    ObservabilityConfig,
    RedactionConfig,
    StatementEvent,
    TelemetryConfig,
)

from goe.observability.context import format_oracle_client_info, format_sqlcommenter
from goe.observability.runtime import get_observability_runtime, get_tracer_provider

__all__ = [
    "apply_oracle_session_trace_info",
    "build_db_observability_config",
    "build_oracle_plsql_client_info_command",
    "build_transport_sql_comment",
    "db_statement_observer",
]

logger = logging.getLogger(__name__)
logger.addHandler(logging.NullHandler())


def db_statement_observer(event: StatementEvent) -> None:
    """Emit debug diagnostic logs for executed database statements with trace correlation."""
    logger.debug(
        "Database statement executed: system=%s op=%s duration_s=%.6f trace_id=%s corr_id=%s hash=%s",
        event.db_system,
        event.operation,
        event.duration_s,
        event.trace_id,
        event.correlation_id,
        event.sql_hash,
    )


def build_db_observability_config(print_sql: bool = False) -> ObservabilityConfig:
    """Build a database ObservabilityConfig wired to GOE's active OpenTelemetry TracerProvider."""
    runtime = get_observability_runtime()
    return ObservabilityConfig(
        print_sql=print_sql,
        telemetry=TelemetryConfig(
            enable_spans=runtime.settings.enabled,
            provider_factory=get_tracer_provider,
            service_name=runtime.settings.service_name,
        ),
        redaction=RedactionConfig(
            mask_parameters=True,
            mask_literals=False,
        ),
        logging=LoggingConfig(
            include_query_text=print_sql,
            include_sql_hash=True,
            include_trace_context=True,
            include_correlation_id=True,
        ),
        statement_observers=(db_statement_observer,),
    )


def apply_oracle_session_trace_info(db_conn: Any) -> str | None:
    """Stamp W3C traceparent or GOE ExecutionId onto python-oracledb connection clientinfo."""
    client_info = format_oracle_client_info()
    if client_info and db_conn is not None:
        try:
            db_conn.clientinfo = client_info
        except Exception:
            return None
    return client_info


def build_oracle_plsql_client_info_command(
    include_semi_colons: bool = False,
    escape_semi_colons: bool = False,
) -> str | None:
    """Build a DBMS_APPLICATION_INFO.SET_CLIENT_INFO PL/SQL statement for transport sessions."""
    client_info = format_oracle_client_info()
    if not client_info:
        return None
    safe_info = client_info.replace("'", "''")
    semi = ";" if include_semi_colons else ""
    if escape_semi_colons and semi:
        semi = r"\;"
    return f"DBMS_APPLICATION_INFO.SET_CLIENT_INFO('{safe_info}'){semi}"


def build_transport_sql_comment(existing_comment: str | None = None) -> str | None:
    """Combine an existing transport SQL comment with SQLCommenter trace attributes when enabled."""
    runtime = get_observability_runtime()
    if not runtime.settings.sqlcommenter:
        return existing_comment
    commenter = format_sqlcommenter()
    if not commenter:
        return existing_comment
    if existing_comment:
        return f"{existing_comment} {commenter}"
    return commenter
```
