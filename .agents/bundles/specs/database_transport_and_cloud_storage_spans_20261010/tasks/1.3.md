```python
# SPDX-FileCopyrightText: 2026 The GOE Authors
# SPDX-License-Identifier: Apache-2.0

"""OpenTelemetry span, metric, and GCP label helpers for Google Cloud Storage and BigQuery."""

from __future__ import annotations

import re
from collections.abc import Iterator, Mapping
from contextlib import contextmanager
from typing import Any

from opentelemetry.trace import Span, SpanKind, Status, StatusCode

from goe.observability.context import get_current_trace_context
from goe.observability.runtime import get_observability_runtime, get_tracer

__all__ = [
    "GCP_LABEL_MAX_LEN",
    "build_bigquery_job_labels",
    "record_bigquery_job_metadata",
    "sanitize_gcp_label_value",
    "traced_bigquery_job",
    "traced_gcs_operation",
]

GCP_LABEL_MAX_LEN = 63
_GCP_LABEL_INVALID_RE = re.compile(r"[^a-z0-9_-]+")


def sanitize_gcp_label_value(raw_value: str) -> str:
    """Sanitize a string to comply with GCP label value rules (lowercase [a-z0-9_-], max 63 chars)."""
    cleaned = _GCP_LABEL_INVALID_RE.sub("_", raw_value.strip().lower())
    return cleaned[:GCP_LABEL_MAX_LEN]


def build_bigquery_job_labels(extra_labels: Mapping[str, str] | None = None) -> dict[str, str]:
    """Build GCP-compliant BigQuery job labels from the active GOE execution and trace context."""
    snap = get_current_trace_context()
    labels: dict[str, str] = {"goe_system": "goe"}
    if snap.execution_id:
        labels["goe_execution_id"] = sanitize_gcp_label_value(snap.execution_id)
    if snap.command_type:
        labels["goe_command"] = sanitize_gcp_label_value(snap.command_type)
    if snap.step_id:
        labels["goe_step"] = sanitize_gcp_label_value(snap.step_id)
    if snap.trace_id:
        labels["goe_trace_id"] = sanitize_gcp_label_value(snap.trace_id)
    if extra_labels:
        for key, val in extra_labels.items():
            if key and val is not None:
                labels[sanitize_gcp_label_value(key)] = sanitize_gcp_label_value(str(val))
    return labels


@contextmanager
def traced_gcs_operation(
    operation: str,
    uri: str | None = None,
    bucket: str | None = None,
    path: str | None = None,
    dry_run: bool = False,
) -> Iterator[Span]:
    """Wrap a Google Cloud Storage operation in a goe.storage.gcs.<operation> CLIENT span."""
    tracer = get_tracer()
    with tracer.start_as_current_span(f"goe.storage.gcs.{operation}", kind=SpanKind.CLIENT) as span:
        if span.is_recording():
            span.set_attribute("goe.storage.scheme", "gs")
            span.set_attribute("goe.storage.operation", operation)
            span.set_attribute("goe.storage.dry_run", dry_run)
            if uri:
                span.set_attribute("goe.storage.uri", uri)
            if bucket:
                span.set_attribute("goe.storage.bucket", bucket)
            if path:
                span.set_attribute("goe.storage.path", path)
        try:
            yield span
            if span.is_recording():
                span.set_status(Status(StatusCode.OK))
        except Exception as exc:
            if span.is_recording():
                span.record_exception(exc)
                span.set_status(Status(StatusCode.ERROR, str(exc)))
            raise


@contextmanager
def traced_bigquery_job(
    operation: str,
    project: str | None = None,
    dataset: str | None = None,
    table: str | None = None,
    dry_run: bool = False,
) -> Iterator[Span]:
    """Wrap a Google BigQuery job execution in a goe.backend.bigquery.<operation> CLIENT span."""
    tracer = get_tracer()
    with tracer.start_as_current_span(f"goe.backend.bigquery.{operation}", kind=SpanKind.CLIENT) as span:
        if span.is_recording():
            span.set_attribute("db.system", "bigquery")
            span.set_attribute("goe.bigquery.operation", operation)
            span.set_attribute("goe.bigquery.dry_run", dry_run)
            if project:
                span.set_attribute("goe.bigquery.project", project)
            if dataset:
                span.set_attribute("goe.bigquery.dataset", dataset)
            if table:
                span.set_attribute("goe.bigquery.table", table)
        try:
            yield span
            if span.is_recording():
                span.set_status(Status(StatusCode.OK))
        except Exception as exc:
            if span.is_recording():
                span.record_exception(exc)
                span.set_status(Status(StatusCode.ERROR, str(exc)))
            raise


def record_bigquery_job_metadata(span: Span, job: Any, operation: str) -> None:
    """Extract BigQuery job statistics onto the active span and record bytes_billed and slot_ms metrics."""
    if job is None:
        return
    job_id = getattr(job, "job_id", None)
    location = getattr(job, "location", None)
    project = getattr(job, "project", None)
    statement_type = getattr(job, "statement_type", None)
    bytes_processed = getattr(job, "total_bytes_processed", None)
    bytes_billed = getattr(job, "total_bytes_billed", None)
    slot_millis = getattr(job, "slot_millis", None)
    dml_rows = getattr(job, "num_dml_affected_rows", None)
    output_rows = getattr(job, "output_rows", None)

    if span.is_recording():
        if job_id is not None:
            span.set_attribute("goe.bigquery.job_id", str(job_id))
        if location is not None:
            span.set_attribute("goe.bigquery.location", str(location))
        if project is not None:
            span.set_attribute("goe.bigquery.project", str(project))
        if statement_type is not None:
            span.set_attribute("goe.bigquery.statement_type", str(statement_type))
        if isinstance(bytes_processed, int):
            span.set_attribute("goe.bigquery.total_bytes_processed", bytes_processed)
        if isinstance(bytes_billed, int):
            span.set_attribute("goe.bigquery.total_bytes_billed", bytes_billed)
        if isinstance(slot_millis, int):
            span.set_attribute("goe.bigquery.slot_millis", slot_millis)
        if isinstance(dml_rows, int):
            span.set_attribute("goe.bigquery.num_dml_affected_rows", dml_rows)
        if isinstance(output_rows, int):
            span.set_attribute("goe.bigquery.output_rows", output_rows)

    get_observability_runtime().metrics.record_bigquery_job(
        operation=operation,
        project=str(project) if project else "unknown",
        bytes_billed=bytes_billed if isinstance(bytes_billed, int) else None,
        slot_ms=slot_millis if isinstance(slot_millis, int) else None,
    )
```

### 3.2 `BigQueryBackendApi._default_job_config()` Integration (`src/goe/offload/bigquery/bigquery_backend_api.py:282-287`)

```python
# SPDX-FileCopyrightText: 2026 The GOE Authors
# SPDX-License-Identifier: Apache-2.0

"""Integration snippet for BigQueryBackendApi._default_job_config with GCP job labels."""

from google.cloud import bigquery

from goe.observability.cloud import build_bigquery_job_labels


def _default_job_config(self, query_parameters=None):
    job_config = bigquery.QueryJobConfig(labels=build_bigquery_job_labels())
    if self._bigquery_kms_key():
        job_config.destination_encryption_configuration = self._encryption_configuration()
    if query_parameters:
        job_config.query_parameters = query_parameters
    return job_config
```
