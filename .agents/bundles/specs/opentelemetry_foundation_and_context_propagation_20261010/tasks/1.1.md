"""Typed OpenTelemetry and structured logging configuration for GOE."""

from __future__ import annotations

import os
from typing import Any, Literal, cast

import msgspec

__all__ = [
    "ExporterType",
    "LogFormatType",
    "OTLPProtocolType",
    "ObservabilitySettings",
]

ExporterType = Literal["none", "console", "jsonl", "otlp", "gcp", "memory"]
OTLPProtocolType = Literal["grpc", "http/protobuf"]
LogFormatType = Literal["text", "json"]

VALID_EXPORTERS: frozenset[str] = frozenset({"none", "console", "jsonl", "otlp", "gcp", "memory"})
VALID_PROTOCOLS: frozenset[str] = frozenset({"grpc", "http/protobuf"})
VALID_LOG_FORMATS: frozenset[str] = frozenset({"text", "json"})


class ObservabilitySettings(msgspec.Struct, frozen=True, kw_only=True):
    """Immutable configuration settings for GOE OpenTelemetry and log correlation."""

    enabled: bool = False
    service_name: str = "goe"
    service_version: str = "2.0.0"
    environment: str = "production"
    exporter: ExporterType = "none"
    endpoint: str | None = None
    protocol: OTLPProtocolType = "grpc"
    sample_rate: float = 1.0
    log_correlation: bool = True
    sqlcommenter: bool = False
    log_format: LogFormatType = "text"
    log_json_sidecar: bool = False
    jsonl_export_path: str | None = None
    gcp_project_id: str | None = None

    @classmethod
    def from_env_and_config(cls, config: Any | None = None) -> ObservabilitySettings:
        """Build ObservabilitySettings from OrchestrationConfig attributes and environment variables."""
        enabled = _resolve_bool(config, "otel_enabled", "GOE_OTEL_ENABLED", default=False)
        service_name = _resolve_str(config, "otel_service_name", "GOE_OTEL_SERVICE_NAME", default="goe") or "goe"
        environment = _resolve_str(config, "otel_environment", "GOE_OTEL_ENVIRONMENT", default="production") or "production"
        raw_exporter = (_resolve_str(config, "otel_exporter", "GOE_OTEL_EXPORTER", default="none") or "none").lower()
        exporter = cast(ExporterType, raw_exporter if raw_exporter in VALID_EXPORTERS else "none")
        endpoint = _resolve_str(config, "otel_endpoint", "GOE_OTEL_ENDPOINT", default=None)
        raw_protocol = (_resolve_str(config, "otel_protocol", "GOE_OTEL_PROTOCOL", default="grpc") or "grpc").lower()
        protocol = cast(OTLPProtocolType, raw_protocol if raw_protocol in VALID_PROTOCOLS else "grpc")
        sample_rate = _resolve_float(config, "otel_sample_rate", "GOE_OTEL_SAMPLE_RATE", default=1.0)
        log_correlation = _resolve_bool(config, "otel_log_correlation", "GOE_OTEL_LOG_CORRELATION", default=True)
        sqlcommenter = _resolve_bool(config, "otel_sqlcommenter", "GOE_OTEL_SQLCOMMENTER", default=False)
        raw_log_format = (_resolve_str(config, "log_format", "GOE_LOG_FORMAT", default="text") or "text").lower()
        log_format = cast(LogFormatType, raw_log_format if raw_log_format in VALID_LOG_FORMATS else "text")
        log_json_sidecar = _resolve_bool(config, "log_json_sidecar", "GOE_LOG_JSON_SIDECAR", default=False)
        jsonl_export_path = _resolve_str(config, "otel_jsonl_path", "GOE_OTEL_JSONL_PATH", default=None)
        gcp_project_id = _resolve_str(
            config,
            "bigquery_dataset_project",
            "GOOGLE_CLOUD_PROJECT",
            default=None,
        )
        return cls(
            enabled=enabled,
            service_name=service_name,
            environment=environment,
            exporter=exporter,
            endpoint=endpoint,
            protocol=protocol,
            sample_rate=max(0.0, min(1.0, sample_rate)),
            log_correlation=log_correlation,
            sqlcommenter=sqlcommenter,
            log_format=log_format,
            log_json_sidecar=log_json_sidecar,
            jsonl_export_path=jsonl_export_path,
            gcp_project_id=gcp_project_id,
        )


def _resolve_str(config: Any | None, attr_name: str, env_name: str, default: str | None) -> str | None:
    """Resolve a string setting from config attribute or environment variable."""
    if config is not None and hasattr(config, attr_name):
        val = getattr(config, attr_name)
        if val is not None:
            return str(val)
    env_val = os.environ.get(env_name)
    if env_val is not None and env_val != "":
        return env_val
    return default


def _resolve_bool(config: Any | None, attr_name: str, env_name: str, default: bool) -> bool:
    """Resolve a boolean setting from config attribute or environment variable."""
    if config is not None and hasattr(config, attr_name):
        val = getattr(config, attr_name)
        if isinstance(val, bool):
            return val
        if isinstance(val, str):
            return val.strip().lower() in {"1", "true", "yes", "on", "y"}
    env_val = os.environ.get(env_name)
    if env_val is not None:
        return env_val.strip().lower() in {"1", "true", "yes", "on", "y"}
    return default


def _resolve_float(config: Any | None, attr_name: str, env_name: str, default: float) -> float:
    """Resolve a float setting from config attribute or environment variable."""
    raw = _resolve_str(config, attr_name, env_name, default=None)
    if raw is None:
        return default
    try:
        return float(raw)
    except ValueError:
        return default

"""Typed OpenTelemetry metrics catalog for GOE operations."""

from __future__ import annotations

from opentelemetry import metrics
from opentelemetry.metrics import Counter, Histogram, Meter

__all__ = [
    "GOEMetrics",
    "METER_NAME",
    "get_goe_metrics",
]

METER_NAME = "goe.observability"


class GOEMetrics:
    """Encapsulates the canonical OpenTelemetry metric instruments for GOE."""

    _meter: Meter
    _commands_total: Counter
    _command_duration: Histogram
    _steps_duration: Histogram
    _rows_transported: Counter
    _bytes_staged: Counter
    _bytes_loaded: Counter
    _bigquery_bytes_billed: Counter
    _bigquery_slot_ms: Counter

    def __init__(self, meter: Meter | None = None) -> None:
        """Initialize metric instruments from the provided or global Meter."""
        self._meter = meter or metrics.get_meter(METER_NAME)
        self._commands_total = self._meter.create_counter(
            name="goe.commands.total",
            unit="1",
            description="Total number of GOE orchestration commands executed",
        )
        self._command_duration = self._meter.create_histogram(
            name="goe.command.duration",
            unit="s",
            description="End-to-end duration of GOE orchestration commands in seconds",
        )
        self._steps_duration = self._meter.create_histogram(
            name="goe.steps.duration",
            unit="s",
            description="Duration of individual GOE orchestration steps in seconds",
        )
        self._rows_transported = self._meter.create_counter(
            name="goe.offload.rows_transported",
            unit="1",
            description="Total rows extracted and staged from the source RDBMS",
        )
        self._bytes_staged = self._meter.create_counter(
            name="goe.offload.bytes_staged",
            unit="By",
            description="Total bytes written to staging storage during transport",
        )
        self._bytes_loaded = self._meter.create_counter(
            name="goe.offload.bytes_loaded",
            unit="By",
            description="Total bytes loaded into backend target tables",
        )
        self._bigquery_bytes_billed = self._meter.create_counter(
            name="goe.bigquery.bytes_billed",
            unit="By",
            description="Total bytes billed by Google BigQuery jobs",
        )
        self._bigquery_slot_ms = self._meter.create_counter(
            name="goe.bigquery.slot_ms",
            unit="ms",
            description="Total slot milliseconds consumed by Google BigQuery jobs",
        )

    def record_command(
        self,
        command_type: str,
        status: str,
        backend: str,
        dry_run: bool,
        duration_s: float,
    ) -> None:
        """Record command completion count and duration histogram."""
        attrs: dict[str, str | bool] = {
            "command_type": command_type,
            "status": status,
            "backend": backend or "unknown",
            "dry_run": dry_run,
        }
        self._commands_total.add(1, attributes=attrs)
        self._command_duration.record(max(0.0, duration_s), attributes=attrs)

    def record_step(
        self,
        step_id: str,
        command_type: str,
        status: str,
        mandatory: bool,
        duration_s: float,
    ) -> None:
        """Record step execution duration histogram."""
        attrs: dict[str, str | bool] = {
            "step_id": step_id,
            "command_type": command_type or "unknown",
            "status": status,
            "mandatory": mandatory,
        }
        self._steps_duration.record(max(0.0, duration_s), attributes=attrs)

    def record_transport(
        self,
        transport_method: str,
        backend: str,
        source_owner: str,
        rows: int | None,
        bytes_staged: int | None,
        storage_format: str,
        scheme: str,
    ) -> None:
        """Record rows transported and staging bytes written."""
        if rows is not None and rows > 0:
            self._rows_transported.add(
                rows,
                attributes={
                    "transport_method": transport_method,
                    "backend": backend or "unknown",
                    "source_owner": source_owner or "unknown",
                },
            )
        if bytes_staged is not None and bytes_staged > 0:
            self._bytes_staged.add(
                bytes_staged,
                attributes={
                    "transport_method": transport_method,
                    "storage_format": storage_format or "unknown",
                    "scheme": scheme or "unknown",
                },
            )

    def record_bytes_loaded(self, backend: str, bytes_loaded: int | None) -> None:
        """Record bytes loaded into the final backend table."""
        if bytes_loaded is not None and bytes_loaded > 0:
            self._bytes_loaded.add(bytes_loaded, attributes={"backend": backend or "unknown"})

    def record_bigquery_job(
        self,
        operation: str,
        project: str,
        bytes_billed: int | None,
        slot_ms: int | None,
    ) -> None:
        """Record BigQuery job bytes billed and slot milliseconds consumed."""
        attrs = {"operation": operation, "project": project or "unknown"}
        if bytes_billed is not None and bytes_billed > 0:
            self._bigquery_bytes_billed.add(bytes_billed, attributes=attrs)
        if slot_ms is not None and slot_ms > 0:
            self._bigquery_slot_ms.add(slot_ms, attributes=attrs)


_GLOBAL_METRICS: GOEMetrics | None = None


def get_goe_metrics(meter: Meter | None = None) -> GOEMetrics:
    """Return the active GOEMetrics instance."""
    global _GLOBAL_METRICS
    if meter is not None:
        return GOEMetrics(meter=meter)
    if _GLOBAL_METRICS is None:
        _GLOBAL_METRICS = GOEMetrics()
    return _GLOBAL_METRICS

"""OpenTelemetry provider lifecycle, JSONL exporter, and in-memory test harness for GOE."""

from __future__ import annotations

from pathlib import Path
from socket import gethostname
from typing import TYPE_CHECKING, Any, Sequence

import msgspec
from opentelemetry import metrics, propagate, trace
from opentelemetry.baggage.propagation import W3CBaggagePropagator
from opentelemetry.metrics import Meter, MeterProvider, NoOpMeterProvider
from opentelemetry.propagators.composite import CompositePropagator
from opentelemetry.trace import NoOpTracerProvider, Tracer, TracerProvider
from opentelemetry.trace.propagation.tracecontext import TraceContextTextMapPropagator

from goe.observability.config import ObservabilitySettings
from goe.observability.metrics import GOEMetrics

if TYPE_CHECKING:
    from opentelemetry.sdk.metrics.export import InMemoryMetricReader
    from opentelemetry.sdk.trace import ReadableSpan
    from opentelemetry.sdk.trace.export import SpanExporter, SpanExportResult
    from opentelemetry.sdk.trace.export.in_memory_span_exporter import InMemorySpanExporter

__all__ = [
    "JSONLSpanExporter",
    "ObservabilityRuntime",
    "TRACER_NAME",
    "configure_observability",
    "get_meter",
    "get_meter_provider",
    "get_observability_runtime",
    "get_tracer",
    "get_tracer_provider",
    "shutdown_observability",
]

TRACER_NAME = "goe.observability"


class JSONLSpanExporter:
    """SpanExporter writing completed ReadableSpan records to a local JSONL file via msgspec."""

    _file_path: Path

    def __init__(self, file_path: str | Path) -> None:
        """Initialize the JSONL span exporter with target file path."""
        self._file_path = Path(file_path)
        self._file_path.parent.mkdir(parents=True, exist_ok=True)

    def export(self, spans: Sequence[ReadableSpan]) -> SpanExportResult:
        """Serialize and append spans as JSON lines."""
        from opentelemetry.sdk.trace.export import SpanExportResult

        try:
            with self._file_path.open("ab") as fh:
                for span in spans:
                    ctx = span.get_span_context()
                    parent_ctx = span.parent
                    payload = {
                        "name": span.name,
                        "trace_id": format(ctx.trace_id, "032x") if ctx else None,
                        "span_id": format(ctx.span_id, "016x") if ctx else None,
                        "parent_span_id": format(parent_ctx.span_id, "016x") if parent_ctx else None,
                        "kind": span.kind.name,
                        "start_time_ns": span.start_time,
                        "end_time_ns": span.end_time,
                        "status_code": span.status.status_code.name,
                        "status_description": span.status.description,
                        "attributes": dict(span.attributes or {}),
                        "events": [
                            {
                                "name": ev.name,
                                "timestamp_ns": ev.timestamp,
                                "attributes": dict(ev.attributes or {}),
                            }
                            for ev in span.events
                        ],
                    }
                    fh.write(msgspec.json.encode(payload) + b"\n")
            return SpanExportResult.SUCCESS
        except OSError:
            return SpanExportResult.FAILURE

    def shutdown(self) -> None:
        """Shutdown the exporter."""
        return None

    def force_flush(self, timeout_millis: int = 30000) -> bool:
        """Flush buffered spans."""
        return True


class ObservabilityRuntime:
    """Manages TracerProvider, MeterProvider, and exporter lifecycle for a GOE process."""

    settings: ObservabilitySettings
    tracer_provider: TracerProvider
    meter_provider: MeterProvider
    metrics: GOEMetrics
    _memory_span_exporter: InMemorySpanExporter | None
    _memory_metric_reader: InMemoryMetricReader | None

    def __init__(
        self,
        settings: ObservabilitySettings,
        tracer_provider: TracerProvider,
        meter_provider: MeterProvider,
        memory_span_exporter: InMemorySpanExporter | None = None,
        memory_metric_reader: InMemoryMetricReader | None = None,
    ) -> None:
        """Initialize the runtime with configured providers."""
        self.settings = settings
        self.tracer_provider = tracer_provider
        self.meter_provider = meter_provider
        self._memory_span_exporter = memory_span_exporter
        self._memory_metric_reader = memory_metric_reader
        self.metrics = GOEMetrics(meter=self.meter_provider.get_meter(TRACER_NAME))

    @classmethod
    def create(
        cls,
        settings: ObservabilitySettings | None = None,
        extra_span_processors: Sequence[Any] | None = None,
    ) -> ObservabilityRuntime:
        """Construct an ObservabilityRuntime from settings, falling back to NoOp when disabled."""
        resolved = settings or ObservabilitySettings.from_env_and_config()
        propagate.set_global_textmap(
            CompositePropagator([TraceContextTextMapPropagator(), W3CBaggagePropagator()])
        )
        if not resolved.enabled or resolved.exporter == "none":
            noop_tp = NoOpTracerProvider()
            noop_mp = NoOpMeterProvider()
            return cls(settings=resolved, tracer_provider=noop_tp, meter_provider=noop_mp)

        from opentelemetry.sdk.metrics import MeterProvider as SdkMeterProvider
        from opentelemetry.sdk.metrics.export import InMemoryMetricReader
        from opentelemetry.sdk.resources import Resource
        from opentelemetry.sdk.trace import TracerProvider as SdkTracerProvider
        from opentelemetry.sdk.trace.export import (
            BatchSpanProcessor,
            ConsoleSpanExporter,
            SimpleSpanProcessor,
        )
        from opentelemetry.sdk.trace.export.in_memory_span_exporter import InMemorySpanExporter
        from opentelemetry.sdk.trace.sampling import TraceIdRatioBased

        resource = Resource.create(
            {
                "service.name": resolved.service_name,
                "service.version": resolved.service_version,
                "service.instance.id": gethostname(),
                "deployment.environment": resolved.environment,
            }
        )
        sampler = TraceIdRatioBased(resolved.sample_rate)
        sdk_tp = SdkTracerProvider(resource=resource, sampler=sampler)

        for processor in extra_span_processors or ():
            sdk_tp.add_span_processor(processor)

        mem_span_exporter: InMemorySpanExporter | None = None
        mem_metric_reader: InMemoryMetricReader | None = None
        metric_readers: list[Any] = []

        if resolved.exporter == "memory":
            mem_span_exporter = InMemorySpanExporter()
            mem_metric_reader = InMemoryMetricReader()
            sdk_tp.add_span_processor(SimpleSpanProcessor(mem_span_exporter))
            metric_readers.append(mem_metric_reader)
        elif resolved.exporter == "console":
            sdk_tp.add_span_processor(SimpleSpanProcessor(ConsoleSpanExporter()))
        elif resolved.exporter == "jsonl":
            export_path = resolved.jsonl_export_path or "/tmp/goe_traces.jsonl"
            sdk_tp.add_span_processor(SimpleSpanProcessor(JSONLSpanExporter(export_path)))
        elif resolved.exporter == "otlp":
            span_exporter = _build_otlp_span_exporter(resolved)
            sdk_tp.add_span_processor(BatchSpanProcessor(span_exporter))
        elif resolved.exporter == "gcp":
            span_exporter = _build_gcp_span_exporter(resolved)
            sdk_tp.add_span_processor(BatchSpanProcessor(span_exporter))

        sdk_mp = SdkMeterProvider(resource=resource, metric_readers=metric_readers)
        return cls(
            settings=resolved,
            tracer_provider=sdk_tp,
            meter_provider=sdk_mp,
            memory_span_exporter=mem_span_exporter,
            memory_metric_reader=mem_metric_reader,
        )

    def get_tracer(self, name: str = TRACER_NAME) -> Tracer:
        """Return a Tracer bound to this runtime's TracerProvider."""
        return self.tracer_provider.get_tracer(name)

    def get_meter(self, name: str = TRACER_NAME) -> Meter:
        """Return a Meter bound to this runtime's MeterProvider."""
        return self.meter_provider.get_meter(name)

    def get_finished_spans(self) -> tuple[ReadableSpan, ...]:
        """Return finished spans when configured with the memory exporter."""
        if self._memory_span_exporter is None:
            return ()
        return self._memory_span_exporter.get_finished_spans()

    def clear_finished_spans(self) -> None:
        """Clear finished spans in the memory exporter."""
        if self._memory_span_exporter is not None:
            self._memory_span_exporter.clear()

    def get_metrics_data(self) -> Any:
        """Return collected metrics data when configured with the memory metric reader."""
        if self._memory_metric_reader is None:
            return None
        return self._memory_metric_reader.get_metrics_data()

    def shutdown(self) -> None:
        """Flush and shut down SDK TracerProvider and MeterProvider if active."""
        shutdown_tp = getattr(self.tracer_provider, "shutdown", None)
        if callable(shutdown_tp):
            shutdown_tp()
        shutdown_mp = getattr(self.meter_provider, "shutdown", None)
        if callable(shutdown_mp):
            shutdown_mp()


def _build_otlp_span_exporter(settings: ObservabilitySettings) -> SpanExporter:
    """Instantiate OTLP gRPC or HTTP SpanExporter."""
    if settings.protocol == "http/protobuf":
        from opentelemetry.exporter.otlp.proto.http.trace_exporter import OTLPSpanExporter as HttpOTLPSpanExporter

        return HttpOTLPSpanExporter(endpoint=settings.endpoint)
    from opentelemetry.exporter.otlp.proto.grpc.trace_exporter import OTLPSpanExporter as GrpcOTLPSpanExporter

    return GrpcOTLPSpanExporter(endpoint=settings.endpoint)


def _build_gcp_span_exporter(settings: ObservabilitySettings) -> SpanExporter:
    """Instantiate Google Cloud Trace exporter using ADC."""
    from opentelemetry.exporter.cloud_trace import CloudTraceSpanExporter

    return CloudTraceSpanExporter(project_id=settings.gcp_project_id)


_ACTIVE_RUNTIME: ObservabilityRuntime | None = None


def configure_observability(
    settings: ObservabilitySettings | None = None,
    config: Any | None = None,
    extra_span_processors: Sequence[Any] | None = None,
) -> ObservabilityRuntime:
    """Configure and register the active GOE ObservabilityRuntime."""
    global _ACTIVE_RUNTIME
    if _ACTIVE_RUNTIME is not None:
        _ACTIVE_RUNTIME.shutdown()
    resolved = settings or ObservabilitySettings.from_env_and_config(config)
    _ACTIVE_RUNTIME = ObservabilityRuntime.create(
        settings=resolved,
        extra_span_processors=extra_span_processors,
    )
    return _ACTIVE_RUNTIME


def get_observability_runtime() -> ObservabilityRuntime:
    """Return the active ObservabilityRuntime, initializing a default NoOp runtime if unset."""
    global _ACTIVE_RUNTIME
    if _ACTIVE_RUNTIME is None:
        _ACTIVE_RUNTIME = ObservabilityRuntime.create(ObservabilitySettings())
    return _ACTIVE_RUNTIME


def get_tracer_provider() -> TracerProvider:
    """Return the active TracerProvider."""
    return get_observability_runtime().tracer_provider


def get_meter_provider() -> MeterProvider:
    """Return the active MeterProvider."""
    return get_observability_runtime().meter_provider


def get_tracer(name: str = TRACER_NAME) -> Tracer:
    """Return a Tracer from the active ObservabilityRuntime."""
    return get_observability_runtime().get_tracer(name)


def get_meter(name: str = TRACER_NAME) -> Meter:
    """Return a Meter from the active ObservabilityRuntime."""
    return get_observability_runtime().get_meter(name)


def shutdown_observability() -> None:
    """Shut down and reset the active ObservabilityRuntime."""
    global _ACTIVE_RUNTIME
    if _ACTIVE_RUNTIME is not None:
        _ACTIVE_RUNTIME.shutdown()
