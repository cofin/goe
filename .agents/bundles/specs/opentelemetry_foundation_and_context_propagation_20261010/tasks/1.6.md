# Task 1.6: CLI flags, `CommonOptions` fields, offload entry-point bootstrap and env template docs

## Objective

1. Add `--otel-enabled/--no-otel-enabled`, `--otel-exporter`, `--otel-endpoint`, and `--log-format` to the shared Click options; `extract_common_options()` returns those keys only when explicitly supplied so default output is unchanged.
2. Add matching `None`-default fields to `CommonOptions` so typed option structs carry CLI overrides into `ObservabilitySettings.from_env_and_config()`.
3. Bootstrap telemetry at the top of `offload_by_cli()` with `ensure_observability(options)` so the runtime (and the log-format settings consumed by chapter 4.2 task 1.3) exists before the command log file opens.
4. Document `GOE_OTEL_*`, `GOE_LOG_FORMAT`, and `GOE_LOG_JSON_SIDECAR` in `templates/conf/offload.env.template.common`.

```python
"""Behavior tests for OpenTelemetry CLI and options wiring."""

from typing import Any

import rich_click as click
from click.testing import CliRunner

from goe.cli.common import common_options, extract_common_options
from goe.config.options import CommonOptions
from goe.observability.config import ObservabilitySettings


def test_flags_absent_leave_output_unchanged() -> None:
    resolved = _run([])
    assert "otel_enabled" not in resolved
    assert "log_format" not in resolved


def test_flags_present_are_returned() -> None:
    resolved = _run(["--otel-enabled", "--otel-exporter", "memory", "--log-format", "json"])
    assert resolved["otel_enabled"] is True
    assert resolved["otel_exporter"] == "memory"
    assert resolved["log_format"] == "json"


def test_common_options_fields_drive_settings(monkeypatch) -> None:
    monkeypatch.delenv("GOE_OTEL_ENABLED", raising=False)
    options = CommonOptions.from_mapping({"otel_enabled": True, "otel_exporter": "memory", "log_format": "json"})
    settings = ObservabilitySettings.from_env_and_config(options)
    assert settings.enabled is True
    assert settings.exporter == "memory"
    assert settings.log_format == "json"
    assert ObservabilitySettings.from_env_and_config(CommonOptions.from_mapping({})).enabled is False


def _run(args: list[str]) -> dict[str, Any]:
    """Invoke a throwaway command decorated with common_options and return the extracted options."""
    captured: dict[str, Any] = {}

    @click.command()
    @common_options
    @click.pass_context
    def probe(ctx: click.Context, **kwargs: Any) -> None:
        captured.update(extract_common_options(ctx, kwargs))

    result = CliRunner().invoke(probe, args)
    assert result.exit_code == 0, result.output
    return captured
```
