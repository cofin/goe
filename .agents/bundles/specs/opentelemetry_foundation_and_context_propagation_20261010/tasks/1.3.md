# Task 1.3: `OrchestrationConfig`, `CommonOptions`, CLI Flags & `offload.env.template` Wiring

## Objective

Expose OpenTelemetry and structured logging settings cleanly across GOE's configuration and CLI layers so operators can configure telemetry via `offload.env` environment variables or CLI flags:
1. Add default functions in `src/goe/config/orchestration_defaults.py`:
   - `otel_enabled_default() -> bool` (`GOE_OTEL_ENABLED`, default `False`)
   - `otel_service_name_default() -> str` (`GOE_OTEL_SERVICE_NAME`, default `"goe"`)
   - `otel_exporter_default() -> str` (`GOE_OTEL_EXPORTER`, default `"none"`)
   - `otel_endpoint_default() -> str | None` (`GOE_OTEL_ENDPOINT`, default `None`)
   - `otel_protocol_default() -> str` (`GOE_OTEL_PROTOCOL`, default `"grpc"`)
   - `otel_sample_rate_default() -> float` (`GOE_OTEL_SAMPLE_RATE`, default `1.0`)
   - `otel_sqlcommenter_default() -> bool` (`GOE_OTEL_SQLCOMMENTER`, default `False`)
   - `log_format_default() -> str` (`GOE_LOG_FORMAT`, default `"text"`)
   - `log_json_sidecar_default() -> bool` (`GOE_LOG_JSON_SIDECAR`, default `False`)
2. Register these keys in `EXPECTED_CONFIG_ARGS` and `OrchestrationConfig` (`src/goe/config/orchestration_config.py:47-250`), `CommonOptions` (`src/goe/config/options.py` from PRD 2 Chapter 2.1), and `src/goe/cli/common.py` (`--otel-enabled`, `--otel-exporter`, `--otel-endpoint`, `--log-format`).
3. Document the `GOE_OTEL_*` and `GOE_LOG_FORMAT` variables in `templates/conf/offload.env.template`.

```python
# SPDX-FileCopyrightText: 2026 The GOE Authors

"""Default helpers for OpenTelemetry and structured logging options in orchestration_defaults.py."""

from __future__ import annotations

import os


def otel_enabled_default() -> bool:
    """Return default value for GOE_OTEL_ENABLED."""
    val = os.environ.get("GOE_OTEL_ENABLED", "false")
    return val.strip().lower() in {"1", "true", "yes", "on", "y"}


def otel_service_name_default() -> str:
    """Return default value for GOE_OTEL_SERVICE_NAME."""
    return os.environ.get("GOE_OTEL_SERVICE_NAME", "goe")


def otel_exporter_default() -> str:
    """Return default value for GOE_OTEL_EXPORTER."""
    return os.environ.get("GOE_OTEL_EXPORTER", "none")


def otel_endpoint_default() -> str | None:
    """Return default value for GOE_OTEL_ENDPOINT."""
    return os.environ.get("GOE_OTEL_ENDPOINT") or None


def otel_protocol_default() -> str:
    """Return default value for GOE_OTEL_PROTOCOL."""
    return os.environ.get("GOE_OTEL_PROTOCOL", "grpc")


def otel_sample_rate_default() -> float:
    """Return default value for GOE_OTEL_SAMPLE_RATE."""
    try:
        return float(os.environ.get("GOE_OTEL_SAMPLE_RATE", "1.0"))
    except ValueError:
        return 1.0


def otel_sqlcommenter_default() -> bool:
    """Return default value for GOE_OTEL_SQLCOMMENTER."""
    val = os.environ.get("GOE_OTEL_SQLCOMMENTER", "false")
    return val.strip().lower() in {"1", "true", "yes", "on", "y"}


def log_format_default() -> str:
    """Return default value for GOE_LOG_FORMAT."""
    return os.environ.get("GOE_LOG_FORMAT", "text")


def log_json_sidecar_default() -> bool:
    """Return default value for GOE_LOG_JSON_SIDECAR."""
    val = os.environ.get("GOE_LOG_JSON_SIDECAR", "false")
    return val.strip().lower() in {"1", "true", "yes", "on", "y"}
```

### 3.2 `src/goe/cli/common.py` Option Decorators

```python
# SPDX-FileCopyrightText: 2026 The GOE Authors
# SPDX-License-Identifier: Apache-2.0

"""CLI common option additions for OpenTelemetry and structured logging."""

from __future__ import annotations

import click

from goe.config import orchestration_defaults


def add_observability_cli_options(fn):
    """Attach optional OpenTelemetry and log format CLI options to a Click command."""
    options = [
        click.option(
            "--otel-enabled/--no-otel-enabled",
            "otel_enabled",
            default=None,
            help="Enable OpenTelemetry distributed tracing and metrics collection.",
        ),
        click.option(
            "--otel-exporter",
            "otel_exporter",
            type=click.Choice(["none", "console", "jsonl", "otlp", "gcp", "memory"], case_sensitive=False),
            default=None,
            help="OpenTelemetry exporter backend (none, console, jsonl, otlp, gcp).",
        ),
        click.option(
            "--otel-endpoint",
            "otel_endpoint",
            type=str,
            default=None,
            help="OTLP collector endpoint URL.",
        ),
        click.option(
            "--log-format",
            "log_format",
            type=click.Choice(["text", "json"], case_sensitive=False),
            default=None,
            help="Log file output format (text or json).",
        ),
    ]
    for option in reversed(options):
        fn = option(fn)
    return fn
```
