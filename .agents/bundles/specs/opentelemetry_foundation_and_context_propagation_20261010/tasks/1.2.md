# Task 1.2: `ExecutionId` <-> W3C Trace Context Bridge, `ExecutionContextSpanProcessor` & Carrier Propagation

## Objective

Connect GOE's domain execution identifier (`ExecutionId` in `src/goe/orchestration/execution_id.py:44-89`) with OpenTelemetry's W3C `traceparent`, `baggage`, and `sqlspec.utils.correlation.CorrelationContext`:
1. Implement `src/goe/observability/context.py` providing:
   - `ExecutionTraceContext` (`msgspec.Struct` holding `execution_id`, `command_type`, `step_id`, `source_owner`, `source_table`, `trace_id`, `span_id`, `traceparent`).
   - `bind_execution_context(...)` context manager that sets Python `ContextVar` values, OpenTelemetry `baggage` (`goe.execution.id`, `goe.command.type`, `goe.step.id`), and enters `sqlspec.utils.correlation.CorrelationContext(execution_id)`.
   - `ExecutionContextSpanProcessor` (`SpanProcessor` implementation automatically registered by `ObservabilityRuntime.create()`) that stamps `goe.execution.id`, `goe.command.type`, `goe.step.id`, `goe.source.owner`, and `goe.source.table` onto every started span (including `sqlspec.query` child spans).
   - Carrier injection/extraction helpers for dictionaries (`inject_carrier`, `extract_carrier`), subprocess environments (`inject_env_context`, `extract_env_context`), Oracle `V$SESSION.CLIENT_INFO` (`format_oracle_client_info`, strictly `<= 64` bytes), and `SQLCommenter` (`format_sqlcommenter`).
2. Add convenience helper `ExecutionId.to_trace_id_int() -> int` on `ExecutionId` (`src/goe/orchestration/execution_id.py`) converting the 16-byte UUID (`int.from_bytes(self.id.bytes, "big")`) for deterministic correlation tests.

```python
# SPDX-FileCopyrightText: 2026 The GOE Authors
# SPDX-License-Identifier: Apache-2.0

"""ExecutionId, W3C TraceContext, Baggage, and SQLSpec CorrelationContext bridge."""

from __future__ import annotations

from collections.abc import Iterator, Mapping, MutableMapping
from contextlib import contextmanager
from contextvars import ContextVar
from typing import TYPE_CHECKING, Any

import msgspec
from opentelemetry import baggage, context, propagate, trace
from opentelemetry.context import Context
from sqlspec.utils.correlation import CorrelationContext

if TYPE_CHECKING:
    from opentelemetry.sdk.trace import ReadableSpan, Span

    from goe.orchestration.execution_id import ExecutionId

__all__ = [
    "ATTR_COMMAND_TYPE",
    "ATTR_EXECUTION_ID",
    "ATTR_SOURCE_OWNER",
    "ATTR_SOURCE_TABLE",
    "ATTR_STEP_ID",
    "ExecutionContextSpanProcessor",
    "ExecutionTraceContext",
    "bind_execution_context",
    "bind_step_context",
    "extract_carrier",
    "extract_env_context",
    "format_oracle_client_info",
    "format_sqlcommenter",
    "format_w3c_traceparent",
    "get_current_trace_context",
    "inject_carrier",
    "inject_env_context",
]

ATTR_EXECUTION_ID = "goe.execution.id"
ATTR_COMMAND_TYPE = "goe.command.type"
ATTR_STEP_ID = "goe.step.id"
ATTR_SOURCE_OWNER = "goe.source.owner"
ATTR_SOURCE_TABLE = "goe.source.table"

ORACLE_CLIENT_INFO_MAX_BYTES = 64

_EXECUTION_ID_VAR: ContextVar[str | None] = ContextVar("goe_execution_id", default=None)
_COMMAND_TYPE_VAR: ContextVar[str | None] = ContextVar("goe_command_type", default=None)
_STEP_ID_VAR: ContextVar[str | None] = ContextVar("goe_step_id", default=None)
_SOURCE_OWNER_VAR: ContextVar[str | None] = ContextVar("goe_source_owner", default=None)
_SOURCE_TABLE_VAR: ContextVar[str | None] = ContextVar("goe_source_table", default=None)


class ExecutionTraceContext(msgspec.Struct, frozen=True, kw_only=True):
    """Snapshot of the active GOE execution and W3C trace context."""

    execution_id: str | None = None
    command_type: str | None = None
    step_id: str | None = None
    source_owner: str | None = None
    source_table: str | None = None
    trace_id: str | None = None
    span_id: str | None = None
    traceparent: str | None = None


class ExecutionContextSpanProcessor:
    """SpanProcessor that stamps active GOE execution attributes onto every started span."""

    def on_start(self, span: Span, parent_context: Context | None = None) -> None:
        """Attach execution_id, command_type, step_id, and source table attributes on span start."""
        if not span.is_recording():
            return
        exec_id = _EXECUTION_ID_VAR.get() or _baggage_str(ATTR_EXECUTION_ID, parent_context)
        if exec_id:
            span.set_attribute(ATTR_EXECUTION_ID, exec_id)
        cmd_type = _COMMAND_TYPE_VAR.get() or _baggage_str(ATTR_COMMAND_TYPE, parent_context)
        if cmd_type:
            span.set_attribute(ATTR_COMMAND_TYPE, cmd_type)
        step_id = _STEP_ID_VAR.get() or _baggage_str(ATTR_STEP_ID, parent_context)
        if step_id:
            span.set_attribute(ATTR_STEP_ID, step_id)
        owner = _SOURCE_OWNER_VAR.get()
        if owner:
            span.set_attribute(ATTR_SOURCE_OWNER, owner)
        table = _SOURCE_TABLE_VAR.get()
        if table:
            span.set_attribute(ATTR_SOURCE_TABLE, table)

    def on_end(self, span: ReadableSpan) -> None:
        """No-op on span completion."""
        return None

    def shutdown(self) -> None:
        """No-op on processor shutdown."""
        return None

    def force_flush(self, timeout_millis: int = 30000) -> bool:
        """Return True immediately since attributes are mutated synchronously on start."""
        return True


def _baggage_str(key: str, ctx: Context | None = None) -> str | None:
    """Read a string value from OpenTelemetry baggage."""
    val = baggage.get_baggage(key, context=ctx)
    return str(val) if val is not None else None


@contextmanager
def bind_execution_context(
    execution_id: ExecutionId | str | None,
    command_type: str | None = None,
    step_id: str | None = None,
    source_owner: str | None = None,
    source_table: str | None = None,
) -> Iterator[ExecutionTraceContext]:
    """Bind GOE execution metadata across ContextVars, OTel Baggage, and SQLSpec CorrelationContext."""
    exec_str = str(execution_id) if execution_id is not None else _EXECUTION_ID_VAR.get()
    tok_exec = _EXECUTION_ID_VAR.set(exec_str)
    tok_cmd = _COMMAND_TYPE_VAR.set(command_type or _COMMAND_TYPE_VAR.get())
    tok_step = _STEP_ID_VAR.set(step_id or _STEP_ID_VAR.get())
    tok_owner = _SOURCE_OWNER_VAR.set(source_owner or _SOURCE_OWNER_VAR.get())
    tok_table = _SOURCE_TABLE_VAR.set(source_table or _SOURCE_TABLE_VAR.get())

    otel_ctx = context.get_current()
    if exec_str:
        otel_ctx = baggage.set_baggage(ATTR_EXECUTION_ID, exec_str, context=otel_ctx)
    if command_type:
        otel_ctx = baggage.set_baggage(ATTR_COMMAND_TYPE, command_type, context=otel_ctx)
    if step_id:
        otel_ctx = baggage.set_baggage(ATTR_STEP_ID, step_id, context=otel_ctx)
    otel_token = context.attach(otel_ctx)

    try:
        if exec_str:
            with CorrelationContext.context(exec_str):
                yield get_current_trace_context()
        else:
            yield get_current_trace_context()
    finally:
        context.detach(otel_token)
        _SOURCE_TABLE_VAR.reset(tok_table)
        _SOURCE_OWNER_VAR.reset(tok_owner)
        _STEP_ID_VAR.reset(tok_step)
        _COMMAND_TYPE_VAR.reset(tok_cmd)
        _EXECUTION_ID_VAR.reset(tok_exec)


@contextmanager
def bind_step_context(step_id: str) -> Iterator[ExecutionTraceContext]:
    """Bind the active orchestration step_id for the duration of a step."""
    tok_step = _STEP_ID_VAR.set(step_id)
    otel_ctx = baggage.set_baggage(ATTR_STEP_ID, step_id, context=context.get_current())
    otel_token = context.attach(otel_ctx)
    try:
        yield get_current_trace_context()
    finally:
        context.detach(otel_token)
        _STEP_ID_VAR.reset(tok_step)


def get_current_trace_context() -> ExecutionTraceContext:
    """Return a snapshot of the active GOE execution and W3C span context."""
    span = trace.get_current_span()
    span_ctx = span.get_span_context() if span is not None else None
    trace_id: str | None = None
    span_id: str | None = None
    traceparent: str | None = None
    if span_ctx is not None and span_ctx.is_valid:
        trace_id = format(span_ctx.trace_id, "032x")
        span_id = format(span_ctx.span_id, "016x")
        flags = format(int(span_ctx.trace_flags), "02x")
        traceparent = f"00-{trace_id}-{span_id}-{flags}"
    exec_id = _EXECUTION_ID_VAR.get() or _baggage_str(ATTR_EXECUTION_ID)
    cmd_type = _COMMAND_TYPE_VAR.get() or _baggage_str(ATTR_COMMAND_TYPE)
    step_id = _STEP_ID_VAR.get() or _baggage_str(ATTR_STEP_ID)
    return ExecutionTraceContext(
        execution_id=exec_id,
        command_type=cmd_type,
        step_id=step_id,
        source_owner=_SOURCE_OWNER_VAR.get(),
        source_table=_SOURCE_TABLE_VAR.get(),
        trace_id=trace_id,
        span_id=span_id,
        traceparent=traceparent,
    )


def format_w3c_traceparent() -> str | None:
    """Return the active W3C traceparent string or None if no valid span is active."""
    return get_current_trace_context().traceparent


def format_oracle_client_info() -> str | None:
    """Format active traceparent or execution_id for Oracle V$SESSION.CLIENT_INFO (max 64 bytes)."""
    snap = get_current_trace_context()
    if snap.traceparent:
        return snap.traceparent[:ORACLE_CLIENT_INFO_MAX_BYTES]
    if snap.execution_id:
        return f"goe:{snap.execution_id}"[:ORACLE_CLIENT_INFO_MAX_BYTES]
    return None


def format_sqlcommenter() -> str | None:
    """Format SQLCommenter key-value payload with traceparent and goe_exec."""
    snap = get_current_trace_context()
    parts: list[str] = []
    if snap.traceparent:
        parts.append(f"traceparent='{snap.traceparent}'")
    if snap.execution_id:
        parts.append(f"goe_exec='{snap.execution_id}'")
    if snap.step_id:
        parts.append(f"goe_step='{snap.step_id}'")
    if not parts:
        return None
    return ",".join(parts)


def inject_carrier(carrier: MutableMapping[str, str] | None = None) -> dict[str, str]:
    """Inject W3C traceparent and baggage into a string dictionary carrier."""
    target: dict[str, str] = dict(carrier) if carrier is not None else {}
    propagate.inject(target)
    exec_id = _EXECUTION_ID_VAR.get()
    if exec_id and "goe_execution_id" not in target:
        target["goe_execution_id"] = exec_id
    return target


def extract_carrier(carrier: Mapping[str, Any] | None) -> Context:
    """Extract an OpenTelemetry Context from a string dictionary carrier."""
    if not carrier:
        return context.get_current()
    clean = {str(k).lower(): str(v) for k, v in carrier.items() if isinstance(v, str)}
    return propagate.extract(clean)


def inject_env_context(base_env: Mapping[str, str] | None = None) -> dict[str, str]:
    """Inject TRACEPARENT, TRACESTATE, BAGGAGE, and GOE_EXECUTION_ID into a subprocess environment dict."""
    env_dict = dict(base_env) if base_env is not None else {}
    headers = inject_carrier()
    if "traceparent" in headers:
        env_dict["TRACEPARENT"] = headers["traceparent"]
    if "tracestate" in headers:
        env_dict["TRACESTATE"] = headers["tracestate"]
    if "baggage" in headers:
        env_dict["BAGGAGE"] = headers["baggage"]
    snap = get_current_trace_context()
    if snap.execution_id:
        env_dict["GOE_EXECUTION_ID"] = snap.execution_id
    return env_dict


def extract_env_context(env_map: Mapping[str, str] | None = None) -> Context:
    """Extract an OpenTelemetry Context from uppercase subprocess environment variables."""
    if not env_map:
        return context.get_current()
    carrier: dict[str, str] = {}
    if "TRACEPARENT" in env_map:
        carrier["traceparent"] = env_map["TRACEPARENT"]
    if "TRACESTATE" in env_map:
        carrier["tracestate"] = env_map["TRACESTATE"]
    if "BAGGAGE" in env_map:
        carrier["baggage"] = env_map["BAGGAGE"]
    return extract_carrier(carrier)
```
