# Task 1.4: Authenticate MCP Endpoints and Expose Read-Only Diagnostic Tools

## Objective

1. Remove `r"^/mcp"` from `SecurityConfig.exclude` in `src/goe/listener/security.py` so all `/mcp` requests authenticate through `ConsoleKeyAuthenticator` when `settings.shared_token` is configured.
2. Implement `ConfirmationGate` (short-lived HMAC-SHA256 confirmation tokens bound to `owner_table` and normalized payload digest) and register the five read-only MCP tools (`get_connectivity_report`, `get_table_status_report`, `list_offloaded_tables`, `preview_offload_plan`, and `get_job_status`) in `src/goe/listener/mcp.py`.

## Context

- `src/goe/listener/security.py:114` sets `exclude=[r"^/mcp", r"^/\.well-known"]`. Because `/mcp` is excluded from `SecurityPlugin`, unauthenticated callers can invoke MCP `initialize` and `tools/list`, while authenticated MCP tool sub-requests into `SecureController` endpoints lack a resolved `Principal`.
- `src/goe/listener/mcp.py:11-22` only configures `MCPConfig(name="GOE Listener MCP", before_tool_call=..., after_mcp_tool_call=...)` without registering standalone diagnostic tools or gating mutating tool calls.
- Read-only MCP tools allow AI assistants and operators to inspect connectivity (`ConnectRunner`), offloaded table metadata (`OracleRepoService` / `SystemService`), table status reports, dry-run offload plans (`OffloadPlan` with `execute=False`), and durable job queue status (`JobStoreProtocol.get_job_by_execution_id`) without mutating target databases.

```python
"""Model Context Protocol (MCP) configuration, confirmation gate, and diagnostic tools for GOE Listener."""

import hashlib
import hmac
import time
from typing import Any

from litestar_mcp import LitestarMCP, MCPConfig

from goe.listener.config import settings
from goe.listener.jobs.models import JobRecord
from goe.listener.jobs.store import JobStoreProtocol
from goe.listener.services.system import SystemService, get_system_service
from goe.listener.telemetry import after_mcp_tool_call, before_mcp_tool_call
from goe.typing import FrozenBaseStruct
from goe.util.serializers import encode_json_bytes

__all__ = (
    "ConfirmationChallenge",
    "ConfirmationGate",
    "build_mcp_config",
    "build_mcp_plugin",
    "get_connectivity_report",
    "get_job_status",
    "get_table_status_report",
    "list_offloaded_tables",
    "preview_offload_plan",
)


class ConfirmationChallenge(FrozenBaseStruct):
    """Confirmation token challenge returned when a mutating MCP tool is called without confirmation."""

    confirmation_required: bool
    tool_name: str
    owner_table: str
    confirmation_token: str
    expires_at_epoch: int


class ConfirmationGate:
    """HMAC-SHA256 confirmation token generator and verifier for mutating MCP tools."""

    def __init__(self, secret: str | None = None, ttl_seconds: int = 300) -> None:
        resolved_secret = secret or settings.shared_token or "goe-local-mcp-confirmation-key"
        self._secret_bytes = resolved_secret.encode("utf-8")
        self._ttl_seconds = ttl_seconds

    def issue_token(
        self,
        *,
        tool_name: str,
        owner_table: str,
        payload: dict[str, Any],
        now_epoch: int | None = None,
    ) -> ConfirmationChallenge:
        """Issue a time-bounded HMAC token bound to tool_name, owner_table, and payload."""
        issued_at = int(time.time()) if now_epoch is None else now_epoch
        expires_at = issued_at + self._ttl_seconds
        signature = self._sign(tool_name=tool_name, owner_table=owner_table, payload=payload, expires_at=expires_at)
        token = f"{expires_at}.{signature}"
        return ConfirmationChallenge(
            confirmation_required=True,
            tool_name=tool_name,
            owner_table=owner_table.upper(),
            confirmation_token=token,
            expires_at_epoch=expires_at,
        )

    def verify_token(
        self,
        *,
        token: str,
        tool_name: str,
        owner_table: str,
        payload: dict[str, Any],
        now_epoch: int | None = None,
    ) -> bool:
        """Verify that token is authentic and unexpired for the given tool call."""
        if "." not in token:
            return False
        raw_expiry, provided_sig = token.split(".", 1)
        try:
            expires_at = int(raw_expiry)
        except ValueError:
            return False
        current_time = int(time.time()) if now_epoch is None else now_epoch
        if current_time >= expires_at:
            return False
        expected_sig = self._sign(
            tool_name=tool_name,
            owner_table=owner_table,
            payload=payload,
            expires_at=expires_at,
        )
        return hmac.compare_digest(provided_sig, expected_sig)

    def _sign(
        self,
        *,
        tool_name: str,
        owner_table: str,
        payload: dict[str, Any],
        expires_at: int,
    ) -> str:
        filtered = {k: v for k, v in sorted(payload.items()) if k != "confirmation_token"}
        digest = hashlib.sha256(encode_json_bytes(filtered)).hexdigest()
        message = f"{tool_name}:{owner_table.strip().upper()}:{expires_at}:{digest}".encode("utf-8")
        return hmac.new(self._secret_bytes, message, hashlib.sha256).hexdigest()
```
