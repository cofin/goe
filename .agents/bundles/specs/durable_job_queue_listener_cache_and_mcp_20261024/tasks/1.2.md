# Task 1.2: Implement Oracle and SQLite Job Stores and Lease-Renewing Worker

## Objective

1. Implement `JobStoreProtocol`, `OracleJobStore`, `SqliteJobStore`, `DuplicateActiveTableJobError`, and `create_job_store()` auto-detection in `src/goe/listener/jobs/store.py`.
2. Implement `DurableJobWorker` in `src/goe/listener/jobs/worker.py` with 120-second leases, 30-second background heartbeat lease renewal for multi-hour offloads, an expired-lease reaper (`PENDING` retry when `attempt < max_attempts`, `DEAD_LETTER` when `attempt >= max_attempts`), and node heartbeat publishing.

## Context

- As documented in `.agents/bundles/specs/prd_10_staged_orchestration_durable_jobs_and_vertebra_listener_20261024/research/vertebra-and-runtime-evaluation.md:78-81`, offloads can run for multiple hours. A fixed 60-second visibility timeout without lease renewal causes running offloads to be reclaimed and re-executed.
- `OracleJobStore.claim_batch` executes `SELECT ... FOR UPDATE SKIP LOCKED` inside a transaction, calls `cursor.fetchmany(limit)` to bound the batch size in Python without triggering `ORA-02014`, increments `attempt = attempt + 1`, sets `status = 'CLAIMED'`, `worker_id`, `claimed_at = :now`, `lease_expires_at = :lease_expires_at`, and commits.
- When `enqueue` encounters a unique constraint violation (`ORA-00001` on `GOE_LISTENER_JOBS_ACTIVE_TBL_UQ` or `sqlite3.IntegrityError` on `goe_listener_jobs_active_tbl_uq`), the store raises `DuplicateActiveTableJobError(table_key)` so the HTTP/MCP layer can return `409 Conflict`.
- `create_job_store` probes whether `GOE_LISTENER_JOBS` exists in `ora_repo_user` when an `OracleDatabaseRegistry` is provided; if absent or in single-host mode without repository tables, it initializes `SqliteJobStore` at `$OFFLOAD_HOME/run/listener_jobs.sqlite3` with `PRAGMA journal_mode=WAL`, `PRAGMA synchronous=NORMAL`, and `PRAGMA busy_timeout=5000`.

```python
# SPDX-FileCopyrightText: 2026 The GOE Authors
# SPDX-License-Identifier: Apache-2.0

"""Oracle and SQLite durable job stores for the GOE Listener."""

from datetime import UTC, datetime, timedelta
import logging
import os
from pathlib import Path
import sqlite3
import threading
from typing import Any, Protocol
from uuid import uuid4

from sqlspec.loader import SQLFileLoader

from goe.exceptions import GOEError
from goe.listener.jobs.models import (
    JobRecord,
    JobState,
    NodeHeartbeat,
    normalize_table_key,
)

logger = logging.getLogger(__name__)

SQL_DIR = Path(__file__).resolve().parent / "sql"


class DuplicateActiveTableJobError(GOEError):
    """Raised when an active job already exists for the target frontend table."""

    def __init__(self, table_key: str, existing_execution_id: str | None = None) -> None:
        self.table_key = table_key
        self.existing_execution_id = existing_execution_id
        detail = f"Active offload job already exists for table {table_key}"
        if existing_execution_id:
            detail = f"{detail} (execution_id={existing_execution_id})"
        super().__init__(detail)


class JobStoreProtocol(Protocol):
    """Protocol implemented by OracleJobStore and SqliteJobStore."""

    backend_name: str

    def enqueue(
        self,
        *,
        execution_id: str,
        owner_table: str,
        payload: dict[str, Any],
        max_attempts: int = 3,
        now: datetime | None = None,
    ) -> JobRecord:
        """Insert a new PENDING job or raise DuplicateActiveTableJobError."""

    def claim_batch(
        self,
        *,
        worker_id: str,
        limit: int = 1,
        lease_seconds: int = 120,
        now: datetime | None = None,
    ) -> list[JobRecord]:
        """Atomically claim up to limit PENDING jobs for worker_id."""

    def mark_running(
        self,
        *,
        job_id: str,
        worker_id: str,
        lease_seconds: int = 120,
        now: datetime | None = None,
    ) -> bool:
        """Transition a CLAIMED job to RUNNING and extend its lease."""

    def renew_lease(
        self,
        *,
        job_id: str,
        worker_id: str,
        lease_seconds: int = 120,
        now: datetime | None = None,
    ) -> bool:
        """Extend lease_expires_at for an active job owned by worker_id."""

    def mark_succeeded(
        self,
        *,
        job_id: str,
        worker_id: str,
        now: datetime | None = None,
    ) -> bool:
        """Transition a running job to SUCCEEDED and clear its lease."""

    def mark_failed(
        self,
        *,
        job_id: str,
        worker_id: str,
        error_message: str,
        dead_letter: bool = False,
        now: datetime | None = None,
    ) -> bool:
        """Transition a running job to FAILED or DEAD_LETTER."""

    def reap_expired_leases(self, *, now: datetime | None = None) -> int:
        """Requeue expired leases with remaining attempts and dead-letter exhausted ones."""

    def get_job_by_execution_id(self, execution_id: str) -> JobRecord | None:
        """Return the JobRecord for execution_id, or None when absent."""

    def get_active_job_for_table(self, owner_table: str) -> JobRecord | None:
        """Return the active JobRecord for owner_table, or None when unlocked."""

    def list_jobs(self, *, limit: int = 100, status: JobState | None = None) -> list[JobRecord]:
        """List recent jobs ordered by created_at descending."""

    def upsert_heartbeat(self, heartbeat: NodeHeartbeat) -> None:
        """Insert or update a Listener/worker node heartbeat row."""

    def list_active_heartbeats(self, *, now: datetime | None = None) -> list[NodeHeartbeat]:
        """Return non-expired node heartbeat rows."""
```
