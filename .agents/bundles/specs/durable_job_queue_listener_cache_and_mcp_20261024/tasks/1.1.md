3. Register the new Oracle repository migration `sql/oracle/source/sql/create_offload_repo_24.sql` in `sql/oracle/source/sql/create_offload_repo.sql` and `sql/oracle/source/sql/upgrade_offload_repo_deltas.sql`.

## Context

- `src/goe/listener/jobs.py:1-43` currently defines four `@task` functions on top of `litestar-queues` (`QueueConfig(queue_backend="memory")` in `src/goe/listener/app.py:46-52`). In multi-worker or multi-host deployments, in-memory queues lose jobs on restart, isolate job state across workers, and lack table-level exclusivity.
- `sqlspec.loader.SQLFileLoader` parses `-- name: <query_name>` blocks from `.sql` files.
- Oracle `SELECT ... FOR UPDATE SKIP LOCKED` cannot be combined with `FETCH FIRST n ROWS ONLY` (`ORA-02014: cannot select FOR UPDATE from view with DISTINCT, GROUP BY, etc.`). Therefore, the Oracle claim query in `src/goe/listener/jobs/sql/job_queue.sql` selects eligible `PENDING` rows ordered by `created_at ASC` with `FOR UPDATE SKIP LOCKED` and no row-limiting SQL clause; the caller fetches at most `limit` rows via `cursor.fetchmany(limit)`.
- All value objects inherit from `FrozenBaseStruct` (`src/goe/typing.py`) and serialize JSON payloads via `goe.util.serializers`.

```python
# SPDX-FileCopyrightText: 2026 The GOE Authors
# SPDX-License-Identifier: Apache-2.0

"""Immutable domain models for the GOE Listener durable job queue."""

from datetime import datetime
from enum import StrEnum
from typing import Any

from goe.typing import FrozenBaseStruct
from goe.util.serializers import decode_json, encode_json

__all__ = (
    "JobRecord",
    "JobState",
    "NodeHeartbeat",
    "normalize_table_key",
)


class JobState(StrEnum):
    """Lifecycle states for a durable Listener offload job."""

    PENDING = "PENDING"
    CLAIMED = "CLAIMED"
    RUNNING = "RUNNING"
    SUCCEEDED = "SUCCEEDED"
    FAILED = "FAILED"
    CANCELLED = "CANCELLED"
    DEAD_LETTER = "DEAD_LETTER"

    def is_active(self) -> bool:
        """Return True when the job holds table exclusivity."""
        return self in (JobState.PENDING, JobState.CLAIMED, JobState.RUNNING)

    def is_terminal(self) -> bool:
        """Return True when the job has reached a final state."""
        return self in (
            JobState.SUCCEEDED,
            JobState.FAILED,
            JobState.CANCELLED,
            JobState.DEAD_LETTER,
        )


def normalize_table_key(owner_table: str) -> str:
    """Normalize an owner.table string into canonical uppercase TABLE_KEY form."""
    cleaned = owner_table.strip().upper()
    if "." not in cleaned:
        raise ValueError(f"Expected OWNER.TABLE format for table_key, got {owner_table!r}")
    owner, table = cleaned.split(".", 1)
    if not owner or not table:
        raise ValueError(f"Expected non-empty owner and table in {owner_table!r}")
    return f"{owner}.{table}"


class JobRecord(FrozenBaseStruct):
    """Immutable snapshot of a durable offload job row."""

    job_id: str
    execution_id: str
    table_key: str
    status: JobState
    payload_json: str
    created_at: datetime
    updated_at: datetime
    attempt: int = 0
    max_attempts: int = 3
    worker_id: str | None = None
    claimed_at: datetime | None = None
    lease_expires_at: datetime | None = None
    started_at: datetime | None = None
    finished_at: datetime | None = None
    error_message: str | None = None

    @classmethod
    def create_pending(
        cls,
        *,
        job_id: str,
        execution_id: str,
        owner_table: str,
        payload: dict[str, Any],
        now: datetime,
        max_attempts: int = 3,
    ) -> "JobRecord":
        """Construct a newly enqueued PENDING JobRecord."""
        return cls(
            job_id=job_id,
            execution_id=execution_id,
            table_key=normalize_table_key(owner_table),
            status=JobState.PENDING,
            payload_json=encode_json(payload),
            created_at=now,
            updated_at=now,
            attempt=0,
            max_attempts=max_attempts,
        )

    def decode_payload(self) -> dict[str, Any]:
        """Decode payload_json into a dictionary."""
        result = decode_json(self.payload_json)
        if not isinstance(result, dict):
            raise TypeError("JobRecord payload_json must decode to a dict")
        return result

    def is_lease_expired(self, now: datetime) -> bool:
        """Return True if the job holds a lease that has expired at now."""
        if self.status not in (JobState.CLAIMED, JobState.RUNNING):
            return False
        if self.lease_expires_at is None:
            return True
        return now >= self.lease_expires_at


class NodeHeartbeat(FrozenBaseStruct):
    """Immutable snapshot of an active Listener or worker node heartbeat row."""

    endpoint_id: str
    listener_group_id: str
    host_url: str
    worker_role: str
    updated_at: datetime
    expires_at: datetime
    version: str = "1.0.0"

    def is_expired(self, now: datetime) -> bool:
        """Return True when the node heartbeat has expired at now."""
        return now >= self.expires_at
```
