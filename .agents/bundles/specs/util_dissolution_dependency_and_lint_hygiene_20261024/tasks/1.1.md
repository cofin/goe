```python
# SPDX-FileCopyrightText: 2026 The GOE Authors
# SPDX-License-Identifier: Apache-2.0

"""Bounded concurrency execution helpers for GOE."""

from __future__ import annotations

from collections.abc import Callable, Sequence
from concurrent.futures import ThreadPoolExecutor
import contextvars
from typing import TypeVar

__all__ = ("run_bounded",)

T = TypeVar("T")
R = TypeVar("R")


def run_bounded(
    items: Sequence[T],
    worker_fn: Callable[[T], R],
    *,
    max_workers: int = 4,
) -> list[R]:
    """Execute worker_fn across items with bounded thread parallelism and ContextVar propagation."""
    if not items:
        return []
    effective_workers = max(1, min(max_workers, len(items)))
    if effective_workers == 1:
        return [worker_fn(item) for item in items]

    def _run_in_context(item: T) -> R:
        ctx = contextvars.copy_context()
        return ctx.run(worker_fn, item)

    with ThreadPoolExecutor(max_workers=effective_workers) as executor:
        futures = [executor.submit(_run_in_context, item) for item in items]
        return [future.result() for future in futures]
```
