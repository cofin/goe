# Task 1.3: Create golden dry-run (`execute=False`) characterization test suite for Oracle-to-BigQuery offloads

## Objective
Create a fast, deterministic, offline characterization test suite in `tests/unit/characterization/test_offload_bigquery_golden.py` that exercises GOE's Oracle-to-BigQuery offload pipeline in dry-run (`execute=False`) mode using `FakeOrchestrationRepoClient`, `TestBackedOffloadSourceTable`, and `BackendBigQueryTable`, locking in exact BigQuery DDL (`CREATE TABLE`, `PARTITION BY`, `CLUSTER BY`, `OPTIONS`), staging table DDL, `LOAD DATA` / external table merge SQL, predicate pushdown SQL, and repository metadata structure before Phase 2–4 structural refactorings begin.

## Context
- `src/goe/offload/bigquery/bigquery_backend_table.py`, `src/goe/offload/bigquery/bigquery_backend_api.py`, `src/goe/offload/oracle/oracle_source_table.py`, and `src/goe/offload/predicate_offload.py` generate complex SQL and DDL in dry-run (`execute=False`) mode without requiring live network connections.
- Existing unit tests in `tests/unit/offload/test_backend_table.py` and `tests/unit/offload/test_offload_source_table.py` only assert non-empty strings or partial fragments rather than locking in the exact SQL/DDL and canonical column mapping contracts for Oracle -> BigQuery.
- Having a golden characterization suite in `tests/unit/characterization/test_offload_bigquery_golden.py` provides an immediate regression gate for subsequent PRDs (config decomposition, backend trimming, and `goe.py` orchestration decomposition).

### Step 2: Create `tests/unit/characterization/test_offload_bigquery_golden.py`
Create `tests/unit/characterization/test_offload_bigquery_golden.py` covering five representative Oracle-to-BigQuery characterization scenarios in `execute=False` mode:
1. **RBO (`100/0`) Full Table Offload**: Non-partitioned Oracle table with `NUMBER`, `NUMBER(10)`, `NUMBER(18,2)`, `VARCHAR2`, `DATE`, and `TIMESTAMP(6) WITH TIME ZONE` columns mapped to BigQuery `BIGNUMERIC`, `INT64`, `NUMERIC`, `STRING`, `DATETIME`, and `TIMESTAMP`, asserting the exact `CREATE TABLE` DDL and staging load SQL.
2. **IPA `RANGE` Offload with Synthetic Partitioning & Clustering**: Range-partitioned Oracle table (`TIME_ID DATE`) with `--offload-partition-granularity=M` and `--sort-columns=PROD_ID,CUST_ID`, asserting the synthetic partition column (`GOE_PART_M_TIME_ID`), `PARTITION BY DATE_TRUNC(...)` / integer range partitioning DDL, `CLUSTER BY`, and high-water-mark extraction predicate (`WHERE TIME_ID < TO_DATE(...)`).
3. **IPA `LIST` Offload**: List-partitioned Oracle table (`CHANNEL_ID VARCHAR2`) asserting synthetic list partition mapping and incremental partition predicate generation.
4. **Predicate-Based Offload (PBO `90/10`) with Generic Predicate Pushdown**: Offload with `offload_predicate` (`Column(ID) > Constant(1000) AND Column(STATUS) = Constant('CLOSED')`) asserting both the Oracle source extraction predicate and BigQuery target predicate rendering (`predicate_offload.to_Predicate`).
5. **Schema Evolution / Column Addition**: Adding a new canonical column to an existing BigQuery backend table and asserting the exact `ALTER TABLE ... ADD COLUMN` DDL emitted in dry-run mode.

```python
# SPDX-FileCopyrightText: 2026 The GOE Authors
# SPDX-License-Identifier: Apache-2.0

"""Golden dry-run (execute=False) characterization tests for Oracle-to-BigQuery offload SQL and DDL."""

from __future__ import annotations

import pytest

from goe.offload.bigquery.bigquery_column import (
    BIGQUERY_TYPE_BIGNUMERIC,
    BIGQUERY_TYPE_DATETIME,
    BIGQUERY_TYPE_INT64,
    BIGQUERY_TYPE_NUMERIC,
    BIGQUERY_TYPE_STRING,
    BIGQUERY_TYPE_TIMESTAMP,
    BigQueryColumn,
)
from goe.offload.column_metadata import (
    CANONICAL_CHAR_SEMANTICS_BYTE,
    GOE_TYPE_DATE,
    GOE_TYPE_DECIMAL,
    GOE_TYPE_INTEGER_8,
    GOE_TYPE_TIMESTAMP_TZ,
    GOE_TYPE_VARIABLE_STRING,
    CanonicalColumn,
)
from goe.offload.factory.backend_api_factory import backend_api_factory
from goe.offload.factory.backend_table_factory import backend_table_factory
from goe.offload.offload_constants import DBTYPE_BIGQUERY
from goe.offload.offload_messages import OffloadMessages
from goe.offload.oracle.oracle_column import (
    ORACLE_TYPE_DATE,
    ORACLE_TYPE_NUMBER,
    ORACLE_TYPE_TIMESTAMP_TZ,
    ORACLE_TYPE_VARCHAR2,
    OracleColumn,
)
from goe.offload.predicate_offload import create_and_relation_predicate, predicate_to_where_clause
from tests.unit.test_functions import (
    FAKE_ORACLE_BQ_ENV,
    build_mock_options,
    build_test_backed_offload_source_table,
)


def _build_bq_dry_run_context(extra_env: dict[str, str] | None = None):
    """Build deterministic OrchestrationConfig, OffloadMessages, and dry-run BackendBigQueryApi."""
    env = dict(FAKE_ORACLE_BQ_ENV)
    if extra_env:
        env.update(extra_env)
    config = build_mock_options(env)
    messages = OffloadMessages()
    backend_api = backend_api_factory(
        DBTYPE_BIGQUERY,
        config,
        messages,
        dry_run=True,
        do_not_connect=True,
    )
    return config, messages, backend_api


def test_golden_bigquery_create_table_ddl_and_canonical_mapping() -> None:
    """Characterize Oracle-to-BigQuery canonical column mapping and CREATE TABLE DDL in dry-run mode."""
    config, messages, backend_api = _build_bq_dry_run_context()
    canonical_cols = [
        CanonicalColumn("ID", GOE_TYPE_INTEGER_8, nullable=False, data_precision=10, data_scale=0),
        CanonicalColumn("AMOUNT", GOE_TYPE_DECIMAL, nullable=True, data_precision=18, data_scale=2),
        CanonicalColumn("UNBOUNDED_NUM", GOE_TYPE_DECIMAL, nullable=True),
        CanonicalColumn(
            "DESCR",
            GOE_TYPE_VARIABLE_STRING,
            nullable=True,
            data_length=200,
            char_length=200,
            char_semantics=CANONICAL_CHAR_SEMANTICS_BYTE,
        ),
        CanonicalColumn("CREATED_DT", GOE_TYPE_DATE, nullable=False),
        CanonicalColumn("UPDATED_TSTZ", GOE_TYPE_TIMESTAMP_TZ, nullable=True, data_scale=6),
    ]
    bq_cols = [ backend_api.from_canonical_column(col) for col in canonical_cols ]
    assert [(c.name, c.data_type, c.nullable) for c in bq_cols] == [
        ("ID", BIGQUERY_TYPE_INT64, False),
        ("AMOUNT", BIGQUERY_TYPE_NUMERIC, True),
        ("UNBOUNDED_NUM", BIGQUERY_TYPE_BIGNUMERIC, True),
        ("DESCR", BIGQUERY_TYPE_STRING, True),
        ("CREATED_DT", BIGQUERY_TYPE_DATETIME, False),
        ("UPDATED_TSTZ", BIGQUERY_TYPE_TIMESTAMP, True),
    ]

    ddl_statements = backend_api.create_table(
        "target_ds",
        "sales_rbo",
        bq_cols,
        None,
        sort_column_names=["ID"],
        sync=True,
    )
    assert ddl_statements
    ddl_sql = "\n".join(ddl_statements)
    assert "CREATE TABLE `test-project.target_ds.sales_rbo`" in ddl_sql
    assert "CLUSTER BY `ID`" in ddl_sql


@pytest.mark.parametrize(
    ("predicate_dsl", "expected_ora_sql", "expected_bq_sql"),
    [
        (
            "Column(ID) > Constant(1000)",
            '("ID" > 1000)',
            "(`ID` > 1000)",
        ),
        (
            "(Column(STATUS) = Constant('CLOSED')) and (Column(AMOUNT) >= Constant(50.25))",
            '(("STATUS" = \'CLOSED\') AND ("AMOUNT" >= 50.25))',
            "((`STATUS` = 'CLOSED') AND (`AMOUNT` >= 50.25))",
        ),
    ],
)
def test_golden_predicate_pushdown_oracle_and_bigquery_rendering(
    predicate_dsl: str,
    expected_ora_sql: str,
    expected_bq_sql: str,
) -> None:
    """Characterize generic offload predicate compilation into Oracle and BigQuery SQL clauses."""
    ora_cols = [
        OracleColumn("ID", ORACLE_TYPE_NUMBER, data_precision=10, data_scale=0),
        OracleColumn("STATUS", ORACLE_TYPE_VARCHAR2, data_length=20),
        OracleColumn("AMOUNT", ORACLE_TYPE_NUMBER, data_precision=18, data_scale=2),
        OracleColumn("CREATED_DT", ORACLE_TYPE_DATE),
        OracleColumn("UPDATED_TSTZ", ORACLE_TYPE_TIMESTAMP_TZ),
    ]
    bq_cols = [
        BigQueryColumn("ID", BIGQUERY_TYPE_INT64),
        BigQueryColumn("STATUS", BIGQUERY_TYPE_STRING),
        BigQueryColumn("AMOUNT", BIGQUERY_TYPE_NUMERIC),
        BigQueryColumn("CREATED_DT", BIGQUERY_TYPE_DATETIME),
        BigQueryColumn("UPDATED_TSTZ", BIGQUERY_TYPE_TIMESTAMP),
    ]
    pred = create_and_relation_predicate([predicate_dsl])
    ora_clause = predicate_to_where_clause(ora_cols, pred)
    bq_clause = predicate_to_where_clause(bq_cols, pred, python_fns_override="bigquery")
    assert ora_clause == expected_ora_sql
    assert bq_clause == expected_bq_sql


def test_golden_bigquery_schema_evolution_add_columns_ddl() -> None:
    """Characterize BigQuery ALTER TABLE ADD COLUMN DDL generation in dry-run mode."""
    _, _, backend_api = _build_bq_dry_run_context()
    new_cols = [
        BigQueryColumn("NEW_COL_STR", BIGQUERY_TYPE_STRING, nullable=True),
        BigQueryColumn("NEW_COL_INT", BIGQUERY_TYPE_INT64, nullable=True),
    ]
    alter_stmts = backend_api.add_columns("target_ds", "sales_rbo", new_cols, sync=True)
    assert alter_stmts
    alter_sql = "\n".join(alter_stmts)
    assert "ALTER TABLE `test-project.target_ds.sales_rbo`" in alter_sql
    assert "ADD COLUMN `NEW_COL_STR` STRING" in alter_sql
    assert "ADD COLUMN `NEW_COL_INT` INT64" in alter_sql
```
