# Task 1.1: Fix integration test collection crash, repair stale test helper signatures, and add `tests/integration/conftest.py`

## Objective
Enable offline collection of `tests/integration` (`GOE_NO_AUTOLOAD_ENV=1 uv run pytest tests/integration --collect-only`) by moving `TestOrchestrationMetadata` config/DB initialization from `__init__` into `setUp()`, fixing stale helper bugs in `tests/integration/offload/test_offload_source_table.py` and `tests/testlib/test_framework/test_functions.py`, adding `pytest-databases[oracle]>=0.14.0` and integration markers to `pyproject.toml`, and creating `tests/integration/conftest.py`.

## Context
- `tests/integration/persistence/test_orchestration_metadata.py:139-146`: `TestOrchestrationMetadata.__init__` calls `build_current_options()`, `get_default_test_messages()`, and `get_testing_frontend_api()` during class instantiation at pytest collection time, crashing with `AssertionError: Required configuration missing: OFFLOAD_HOME` whenever `offload.env` is absent.
- `tests/integration/offload/test_offload_source_table.py:121,128`: Compares `if self.config == DBTYPE_ORACLE:` instead of `if self.config.db_type == DBTYPE_ORACLE:`.
- `tests/testlib/test_framework/test_functions.py:111-116`: `log()` passes `redis_publish=False` to `messages.log()`, which no longer accepts `redis_publish`.
- `tests/testlib/test_framework/test_functions.py:120-124`: `text_in_log(search_text, search_from_text="")` calls `get_line_from_log(search_text, search_from_text=search_from_text)` without the required leading `messages: OffloadMessages` parameter (`get_line_from_log` at lines `89-93`).
- `pyproject.toml:119-125` (`[dependency-groups].test`) does not include `pytest-databases[oracle]>=0.14.0`, and `pyproject.toml:338-348` (`[tool.pytest.ini_options]`) does not declare markers for `oracle`, `bigquery`, `gcs`, and `spark`.

```python
def log(
    messages: OffloadMessages,
    line: str,
    detail: int = NORMAL,
    ansi_code: str | None = None,
) -> None:
    """Write a line to OffloadMessages without deprecated Redis publish arguments."""
    messages.log(line, detail=detail, ansi_code=ansi_code)


def text_in_log(
    messages: OffloadMessages,
    search_text: str,
    search_from_text: str = "",
) -> bool:
    """Return True if search_text appears in the log file managed by messages."""
    matched_line = get_line_from_log(messages, search_text, search_from_text=search_from_text)
    return bool(matched_line)
```

```python
# SPDX-FileCopyrightText: 2026 The GOE Authors
# SPDX-License-Identifier: Apache-2.0

"""Pytest configuration and environment preflight fixtures for GOE integration tests."""

from __future__ import annotations

import os
from typing import TYPE_CHECKING

import pytest

if TYPE_CHECKING:
    from _pytest.config import Config
    from _pytest.nodes import Item


def pytest_collection_modifyitems(config: Config, items: list[Item]) -> None:
    """Attach integration markers and skip tests when required database/cloud env vars are absent."""
    has_oracle_env = bool(os.environ.get("OFFLOAD_HOME") and os.environ.get("ORA_ADM_USER"))
    has_gcp_env = bool(
        os.environ.get("GOOGLE_CLOUD_PROJECT")
        or os.environ.get("BIGQUERY_DATASET_PROJECT")
        or os.environ.get("GOOGLE_IMPERSONATE_SERVICE_ACCOUNT")
    )

    skip_oracle = pytest.mark.skip(
        reason="Integration test requires OFFLOAD_HOME and ORA_ADM_USER environment configuration."
    )
    skip_bigquery = pytest.mark.skip(
        reason="BigQuery integration test requires GOOGLE_CLOUD_PROJECT and ADC credentials."
    )

    for item in items:
        item.add_marker(pytest.mark.oracle)
        if "scenarios" in item.nodeid or "bigquery" in item.nodeid:
            item.add_marker(pytest.mark.bigquery)
            if not has_gcp_env:
                item.add_marker(skip_bigquery)
        if not has_oracle_env:
            item.add_marker(skip_oracle)
```
