# Task 1.1: Fix integration test collection crash, repair latent test framework bugs, and add `tests/integration/conftest.py`

## Objective

Fix the integration test collection crash in `TestOrchestrationMetadata.__init__` so `GOE_NO_AUTOLOAD_ENV=1 uv run pytest tests/integration --collect-only` succeeds without an active Oracle environment, fix `self.config == DBTYPE_ORACLE` comparisons in `tests/integration/offload/test_offload_source_table.py` and `tests/unit/offload/test_offload_source_table.py`, fix the missing `messages: OffloadMessages` argument in `text_in_log` (`tests/testlib/test_framework/test_functions.py`) and its caller (`tests/integration/orchestration/test_orchestration_runner.py`), and add pytest integration markers (`oracle`, `bigquery`, `gcs`, `spark`) and `tests/integration/conftest.py`.

  ```python
  # SPDX-FileCopyrightText: 2026 The GOE Authors
  # SPDX-License-Identifier: Apache-2.0

  """Pytest configuration and environment preflight fixtures for GOE integration tests."""

  from __future__ import annotations

  import os
  from typing import TYPE_CHECKING

  import pytest

  if TYPE_CHECKING:
      from _pytest.config import Config
      from _pytest.nodes import Item


  def pytest_collection_modifyitems(config: Config, items: list[Item]) -> None:
      """Attach integration markers and skip tests when required database/cloud env vars are absent."""
      has_oracle_env = bool(os.environ.get("ORA_CONN") and os.environ.get("ORA_ADM_USER"))
      has_gcp_env = bool(
          os.environ.get("GOOGLE_CLOUD_PROJECT")
          or os.environ.get("BIGQUERY_DATASET_PROJECT")
          or os.environ.get("GOOGLE_IMPERSONATE_SERVICE_ACCOUNT")
      )

      skip_oracle = pytest.mark.skip(
          reason="Integration test requires ORA_CONN and ORA_ADM_USER environment configuration."
      )
      skip_bigquery = pytest.mark.skip(
          reason="BigQuery integration test requires GOOGLE_CLOUD_PROJECT and ADC credentials."
      )

      for item in items:
          item.add_marker(pytest.mark.oracle)
          if "scenarios" in item.nodeid or "bigquery" in item.nodeid:
              item.add_marker(pytest.mark.bigquery)
              if not has_gcp_env:
                  item.add_marker(skip_bigquery)
          if not has_oracle_env:
              item.add_marker(skip_oracle)
  ```
