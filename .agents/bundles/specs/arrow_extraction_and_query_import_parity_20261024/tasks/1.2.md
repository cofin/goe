## Objective

1. Create `src/goe/offload/staging/arrow_writers.py` defining `ArrowParquetStagingWriter` and `ArrowAvroStagingWriter`.
2. Implement `ArrowParquetStagingWriter` registering a `pyarrow.RecordBatchReader` on a `DuckDBEngine` cursor (`cursor.register(view_name, reader)`), streaming it through `DuckDBParquetWriter` (`COPY (SELECT <cast_projection> FROM <view_name>) TO '<uri>' (FORMAT PARQUET, ...)`), and always calling `cursor.unregister(view_name)` inside a `finally:` block to prevent DuckDB 2.0 `nanobind` reference leaks.
3. 4. Update `pyproject.toml` to replace `avro` with `fastavro>=1.9.7` (available in `[project.optional-dependencies]` / dev dependencies).

## Context

- `src/goe/util/parquet_encoder.py:81-170` and `src/goe/util/avro_encoder.py:1-190` currently encode rows in Python one batch or row at a time.
- Verified in `.venv`: DuckDB 2.0's `cursor.register(view_name, reader)` accepts a `pyarrow.RecordBatchReader` directly and streams batches on demand during `COPY ... TO (FORMAT PARQUET)`, but holds a `nanobind` reference to the reader until `cursor.unregister(view_name)` is called.

```python
# SPDX-FileCopyrightText: 2026 The GOE Authors
# SPDX-License-Identifier: Apache-2.0

"""Streaming Arrow-to-Parquet (DuckDB 2.0) and Arrow-to-Avro (fastavro) staging writers."""

import base64
from collections.abc import Iterator, Sequence
from pathlib import Path
from typing import Any
import uuid

import pyarrow as pa

from goe.offload.column_metadata import ColumnMetadataInterface
from goe.offload.transport.duckdb.engine import DuckDBEngine, build_cast_projection
from goe.offload.transport.duckdb.parquet_writer import (
    DuckDBParquetWriter,
    ParquetWriteOptions,
    StagedFileManifest,
    StagedParquetFile,
)
from goe.util.module_loader import ensure_module
from goe.util.serializers import decode_json


class ArrowParquetStagingWriter:
    """Streams a pyarrow.RecordBatchReader through a DuckDBEngine cursor into Parquet files."""

    def __init__(
        self,
        engine: DuckDBEngine,
        *,
        write_options: ParquetWriteOptions | None = None,
    ) -> None:
        self._engine = engine
        self._writer = DuckDBParquetWriter(write_options)

    def write(
        self,
        reader: pa.RecordBatchReader,
        staging_columns: Sequence[ColumnMetadataInterface],
        target_uri: str,
    ) -> StagedFileManifest:
        """Register reader on a cursor, stream COPY ... TO (FORMAT PARQUET), and always unregister."""
        view_name = f"goe_arrow_batch_{uuid.uuid4().hex}"
        cast_projection = build_cast_projection(staging_columns)
        with self._engine.cursor() as cursor:
            registered = False
            try:
                cursor.register(view_name, reader)
                registered = True
                relation_sql = f"SELECT {cast_projection} FROM {view_name}"
                return self._writer.write_relation(cursor, relation_sql, target_uri)
            finally:
                if registered:
                    cursor.unregister(view_name)


class ArrowAvroStagingWriter:
    """Streams a pyarrow.RecordBatchReader into an Avro container file using fastavro."""

    def __init__(
        self,
        schema_json: str | dict[str, Any],
        *,
        compress: bool = True,
        base64_columns: Sequence[str] = (),
    ) -> None:
        self._raw_schema = decode_json(schema_json) if isinstance(schema_json, str) else schema_json
        self._compress = compress
        self._base64_columns = frozenset(base64_columns)

    def write(self, reader: pa.RecordBatchReader, target_path: str | Path) -> StagedFileManifest:
        """Stream all RecordBatches from reader into target_path and return a StagedFileManifest."""
        fastavro = ensure_module("fastavro", extra="avro")
        parsed_schema = fastavro.parse_schema(self._raw_schema)
        codec = "deflate" if self._compress else "null"
        out_path = Path(target_path)
        out_path.parent.mkdir(parents=True, exist_ok=True)
        row_count = 0

        def _record_iterator() -> Iterator[dict[str, Any]]:
            nonlocal row_count
            for batch in reader:
                records = batch.to_pylist()
                for record in records:
                    if self._base64_columns:
                        for col_name in self._base64_columns:
                            val = record.get(col_name)
                            if val is not None:
                                record[col_name] = base64.b64decode(val)
                    row_count += 1
                    yield record

        with out_path.open("wb") as fh:
            fastavro.writer(fh, parsed_schema, _record_iterator(), codec=codec)

        size_bytes = out_path.stat().st_size if out_path.is_file() else 0
        return StagedFileManifest(
            target_uri=str(out_path),
            compression=codec.upper(),
            files=(StagedParquetFile(uri=str(out_path), row_count=row_count, size_bytes=size_bytes),),
            total_rows=row_count,
            total_bytes=size_bytes,
        )
```
