# Task 1.1: Implement OracleArrowStreamExtractor lazy RecordBatchReader over fetch_df_batches

## Objective

1. Create `src/goe/offload/transport/arrow_extractor.py` defining `ArrowExtractionConfig(FrozenBaseStruct)` and `OracleArrowStreamExtractor`.
2. Stream `conn.fetch_df_batches(statement=..., parameters=..., size=batch_size, fetch_decimals=True)` inside a lazy `pyarrow.RecordBatchReader.from_batches(schema, batch_generator())` so at most one Arrow batch resides in memory at any point.
3. Provide deterministic output-type handlers and Arrow batch normalizers covering high-precision `NUMBER(p, s)` (`pa.decimal128(p, s)`), unbounded `NUMBER` (`TO_CHAR(..., 'TM')` UTF-8 string), `INTERVAL YEAR TO MONTH`, `INTERVAL DAY TO SECOND`, `TIMESTAMP WITH TIME ZONE`, `TIMESTAMP WITH LOCAL TIME ZONE`, `CLOB`/`NCLOB`/`BLOB` (arbitrary size >64 KiB), `RAW`, `BINARY_FLOAT`/`BINARY_DOUBLE` (with `nan_values_as_null` scrubbing), and nanosecond `TIMESTAMP(7..9)`.
4. Add `arrow_batch_extraction()` context manager to `src/goe/offload/oracle/oracle_offload_transport_rdbms_api.py` configuring the session parameters (`NLS_TIMESTAMP_FORMAT`, `NLS_TIMESTAMP_TZ_FORMAT`, `TIME_ZONE`, `MODULE`/`ACTION`) and returning a lazy `pa.RecordBatchReader`.

## Context

- `.venv/lib/python3.14/site-packages/sqlspec/adapters/oracledb/driver.py:1053-1085` (`OracleSyncDriver.select_to_arrow`) calls `collect_oracle_arrow_batches`, which eagerly materializes all batches into a Python `list` before returning a `RecordBatchReader`. For multi-gigabyte tables, eager buffering exhausts container memory.
- Verified against `oracledb==26.0.1` and `pyarrow==26.0.0` in `.venv`: `conn.fetch_df_batches(statement=sql, parameters=binds, size=batch_size, fetch_decimals=True)` yields `OracleDataFrame` objects implementing the Arrow PyCapsule `.__arrow_c_stream__()` protocol consumed by `pa.RecordBatchReader.from_stream(ora_df)`.
- When `fetch_df_batches` encounters an empty result set (or a type requiring cursor fallback such as `XMLTYPE` or custom output-type handler columns), `OracleArrowStreamExtractor` constructs a typed 0-row `pa.RecordBatchReader` matching the staging schema or streams `cursor.fetchmany(batch_size)` chunks directly into `pa.RecordBatch.from_pydict(...)` with zero full-table buffering.

```python
# SPDX-FileCopyrightText: 2026 The GOE Authors
# SPDX-License-Identifier: Apache-2.0

"""Lazy PyCapsule Arrow batch extractor over python-oracledb fetch_df_batches."""

from collections.abc import Iterator, Sequence
from decimal import Decimal
import math
from typing import Any

import oracledb
import pyarrow as pa
import pyarrow.compute as pc

from goe.offload.column_metadata import ColumnMetadataInterface
from goe.offload.staging.parquet.parquet_column import (
    PARQUET_TYPE_BINARY,
    PARQUET_TYPE_BOOLEAN,
    PARQUET_TYPE_DOUBLE,
    PARQUET_TYPE_FLOAT,
    PARQUET_TYPE_INT32,
    PARQUET_TYPE_INT64,
    PARQUET_TYPE_STRING,
)
from goe.typing import FrozenBaseStruct

DEFAULT_ARROW_BATCH_SIZE = 65_536
STAGING_TO_ARROW_TYPE: dict[str, pa.DataType] = {
    PARQUET_TYPE_STRING: pa.string(),
    PARQUET_TYPE_BINARY: pa.binary(),
    PARQUET_TYPE_INT32: pa.int32(),
    PARQUET_TYPE_INT64: pa.int64(),
    PARQUET_TYPE_FLOAT: pa.float32(),
    PARQUET_TYPE_DOUBLE: pa.float64(),
    PARQUET_TYPE_BOOLEAN: pa.bool_(),
}


class ArrowExtractionConfig(FrozenBaseStruct, kw_only=True):
    """Configuration for streaming Oracle rows as PyArrow RecordBatches."""

    batch_size: int = DEFAULT_ARROW_BATCH_SIZE
    fetch_decimals: bool = True
    nan_values_as_null: bool = False
    base64_columns: frozenset[str] = frozenset()


class OracleArrowStreamExtractor:
    """Streams Oracle extraction queries into a lazy pyarrow.RecordBatchReader."""

    def __init__(self, config: ArrowExtractionConfig | None = None) -> None:
        self._config = config or ArrowExtractionConfig()

    @property
    def config(self) -> ArrowExtractionConfig:
        """Return the extractor configuration."""
        return self._config

    @staticmethod
    def build_expected_arrow_schema(staging_columns: Sequence[ColumnMetadataInterface]) -> pa.Schema:
        """Build the target Arrow schema from GOE staging column metadata."""
        fields: list[pa.Field] = []
        for col in staging_columns:
            col_name = col.staging_file_column_name or col.name
            dtype = STAGING_TO_ARROW_TYPE.get(col.data_type, pa.string())
            nullable = bool(getattr(col, "nullable", True))
            fields.append(pa.field(col_name, dtype, nullable=nullable))
        return pa.schema(fields)

    @staticmethod
    def output_type_handler(cursor: oracledb.Cursor, metadata: Any) -> Any:
        """Normalize LOBs, intervals, and high-precision numbers when cursor fallback is used."""
        db_type = metadata.type_code
        if db_type in (oracledb.DB_TYPE_CLOB, oracledb.DB_TYPE_NCLOB, oracledb.DB_TYPE_LONG):
            return cursor.var(oracledb.DB_TYPE_LONG_VARCHAR, arraysize=cursor.arraysize)
        if db_type in (oracledb.DB_TYPE_BLOB, oracledb.DB_TYPE_LONG_RAW):
            return cursor.var(
                oracledb.DB_TYPE_LONG_NVARCHAR if False else oracledb.DB_TYPE_LONG_RAW, arraysize=cursor.arraysize
            )
        if db_type in (oracledb.DB_TYPE_INTERVAL_YM, oracledb.DB_TYPE_INTERVAL_DS):
            return cursor.var(oracledb.DB_TYPE_VARCHAR, 128, arraysize=cursor.arraysize)
        if db_type == oracledb.DB_TYPE_NUMBER:
            precision = getattr(metadata, "precision", 0) or 0
            scale = getattr(metadata, "scale", 0) or 0
            if precision > 18 or scale > 0:
                return cursor.var(
                    oracledb.DB_TYPE_VARCHAR,
                    256,
                    arraysize=cursor.arraysize,
                    outconverter=Decimal,
                )
        return None

    def extract_record_batch_reader(
        self,
        conn: oracledb.Connection,
        sql_text: str,
        binds: dict[str, Any] | None,
        staging_columns: Sequence[ColumnMetadataInterface],
    ) -> pa.RecordBatchReader:
        """Return a lazy pyarrow.RecordBatchReader streaming batches from conn.fetch_df_batches."""
        target_schema = self.build_expected_arrow_schema(staging_columns)
        df_iter = iter(
            conn.fetch_df_batches(
                statement=sql_text,
                parameters=binds or {},
                size=self._config.batch_size,
                fetch_decimals=self._config.fetch_decimals,
            )
        )
        first_df = next(df_iter, None)
        if first_df is None:
            return pa.RecordBatchReader.from_batches(target_schema, iter(()))
        first_stream = pa.RecordBatchReader.from_stream(first_df)
        first_batches = [self.normalize_batch(batch, target_schema) for batch in first_stream]
        effective_schema = first_batches[0].schema if first_batches else target_schema

        def _batch_generator() -> Iterator[pa.RecordBatch]:
            for batch in first_batches:
                yield batch
            for ora_df in df_iter:
                reader = pa.RecordBatchReader.from_stream(ora_df)
                for raw_batch in reader:
                    yield self.normalize_batch(raw_batch, effective_schema)

        return pa.RecordBatchReader.from_batches(effective_schema, _batch_generator())

    def normalize_batch(self, batch: pa.RecordBatch, target_schema: pa.Schema) -> pa.RecordBatch:
        """Align column names, scrub NaN/Inf floats when configured, and cast columns to target_schema."""
        arrays: list[pa.Array] = []
        fields: list[pa.Field] = []
        for idx, col_array in enumerate(batch.columns):
            target_field = target_schema[idx] if idx < len(target_schema) else batch.schema.field(idx)
            arr = col_array
            if self._config.nan_values_as_null and (pa.types.is_floating(arr.type) or pa.types.is_decimal(arr.type)):
                if pa.types.is_floating(arr.type):
                    mask = pc.or_(pc.is_nan(arr), pc.is_inf(arr))
                    arr = pc.if_else(mask, pa.scalar(None, type=arr.type), arr)
            if arr.type != target_field.type and not pa.types.is_decimal(arr.type):
                try:
                    arr = pc.cast(arr, target_field.type)
                except pa.ArrowInvalid:
                    arr = self._python_fallback_cast(arr, target_field.type)
            arrays.append(arr)
            fields.append(pa.field(target_field.name, arr.type, nullable=target_field.nullable))
        return pa.RecordBatch.from_arrays(arrays, schema=pa.schema(fields))

    def _python_fallback_cast(self, arr: pa.Array, target_type: pa.DataType) -> pa.Array:
        """Convert complex scalar objects (Decimal, memoryview, LOBs, float NaN) to target_type."""
        py_values = arr.to_pylist()
        converted: list[Any] = []
        for val in py_values:
            if val is None:
                converted.append(None)
            elif isinstance(val, float) and self._config.nan_values_as_null and (math.isnan(val) or math.isinf(val)):
                converted.append(None)
            elif pa.types.is_string(target_type) or pa.types.is_large_string(target_type):
                converted.append(val.decode("utf-8") if isinstance(val, bytes) else str(val))
            elif pa.types.is_binary(target_type) or pa.types.is_large_binary(target_type):
                converted.append(bytes(val) if not isinstance(val, str) else val.encode("utf-8"))
            elif pa.types.is_integer(target_type):
                converted.append(int(val))
            elif pa.types.is_floating(target_type):
                fval = float(val)
                if self._config.nan_values_as_null and (math.isnan(fval) or math.isinf(fval)):
                    converted.append(None)
                else:
                    converted.append(fval)
            else:
                converted.append(val)
        return pa.array(converted, type=target_type)
```
