- `FrozenBaseStruct` comes from `goe.typing` (Chapter 5.2): `msgspec.Struct, frozen=True, gc=False, omit_defaults=True, repr_omit_defaults=True` with `to_dict()`. Struct config keywords (`array_like=True`, `kw_only=True`) may be passed on subclasses. `msgspec.structs.replace(obj, **changes)` returns a new frozen instance without validation.
- `contextvars.ContextVar.set` returns a `Token` and `reset(token)` restores the previous value; `asyncio` tasks run in a copy of the creating context; values set inside a thread or task are invisible to others.
- `ExecutionId` is defined in `src/goe/orchestration/execution_id.py`.
- `gc=False` structs are not tracked by the cyclic garbage collector; sinks must never hold a reference back to an `ExecutionContext` (none below do).

```python
# SPDX-FileCopyrightText: 2026 The GOE Authors
# SPDX-License-Identifier: Apache-2.0

"""Execution context bound per orchestration command through a ContextVar."""

from collections.abc import Iterator
from contextlib import contextmanager
from contextvars import ContextVar, Token
from typing import Any

import msgspec

from goe.orchestration.execution_id import ExecutionId
from goe.orchestration.sinks import (
    NORMAL,
    ConsoleSink,
    ExecutionEvent,
    ExecutionSink,
    LogFileSink,
    RepositoryStepSink,
    TelemetrySink,
)
from goe.typing import FrozenBaseStruct

__all__ = (
    "ExecutionContext",
    "bind_execution_context",
    "current_execution_context",
    "reset_execution_context",
    "set_execution_context",
)

_CURRENT_CONTEXT: ContextVar["ExecutionContext | None"] = ContextVar("goe_execution_context", default=None)


class ExecutionContext(FrozenBaseStruct, kw_only=True):
    """Immutable per-command identity, console policy, log settings, and ordered event sinks."""

    detail: int = NORMAL
    ansi: bool = True
    execution_id: ExecutionId | str | None = None
    command_type: str | None = None
    log_dir: str | None = None
    dev_log: str | None = None
    dev_log_level: str | None = None
    sinks: tuple[ExecutionSink, ...] = ()

    @classmethod
    def for_console(
        cls,
        *,
        detail: int,
        ansi: bool,
        log_handle: Any = None,
        owns_log_handle: bool = True,
        log_dir: str | None = None,
        dev_log: str | None = None,
        dev_log_level: str | None = None,
    ) -> "ExecutionContext":
        """Build the standard sink set: optional log file, console that reports write errors to the file, telemetry."""
        file_sink = LogFileSink(log_handle, owns_handle=owns_log_handle) if log_handle is not None else None
        console = ConsoleSink(detail, ansi, on_error=file_sink.write_line if file_sink else None)
        sinks: tuple[ExecutionSink, ...] = (
            (file_sink, console, TelemetrySink()) if file_sink else (console, TelemetrySink())
        )
        return cls(
            detail=detail,
            ansi=ansi,
            log_dir=log_dir,
            dev_log=dev_log,
            dev_log_level=dev_log_level,
            sinks=sinks,
        )

    @property
    def log_file_sink(self) -> LogFileSink | None:
        """Return the first log file sink, or None when the context writes no log file."""
        return next((sink for sink in self.sinks if isinstance(sink, LogFileSink)), None)

    @property
    def repository_sink(self) -> RepositoryStepSink | None:
        """Return the repository step sink, or None when steps are not recorded."""
        return next((sink for sink in self.sinks if isinstance(sink, RepositoryStepSink)), None)

    def emit(self, event: ExecutionEvent) -> None:
        """Deliver an event to every sink in order; sink exceptions propagate to the caller."""
        for sink in self.sinks:
            sink.handle(event)

    def with_log_file(self, handle: Any, *, owns_handle: bool = True) -> "ExecutionContext":
        """Return a copy writing to handle, with console errors reported to the new file."""
        file_sink = LogFileSink(handle, owns_handle=owns_handle)
        console = ConsoleSink(self.detail, self.ansi, on_error=file_sink.write_line)
        others = tuple(sink for sink in self.sinks if not isinstance(sink, (LogFileSink, ConsoleSink)))
        return msgspec.structs.replace(self, sinks=(file_sink, console, *others))

    def with_repository(self, repo_client: Any) -> "ExecutionContext":
        """Return a copy whose repository step sink records through repo_client."""
        others = tuple(sink for sink in self.sinks if not isinstance(sink, RepositoryStepSink))
        return msgspec.structs.replace(self, sinks=(*others, RepositoryStepSink(repo_client)))

    def with_sinks(self, *sinks: ExecutionSink) -> "ExecutionContext":
        """Return a copy with sinks appended."""
        return msgspec.structs.replace(self, sinks=(*self.sinks, *sinks))

    def with_execution(self, execution_id: ExecutionId | str | None, command_type: str | None) -> "ExecutionContext":
        """Return a copy carrying the execution identity and command type."""
        return msgspec.structs.replace(self, execution_id=execution_id, command_type=command_type)

    def close(self) -> None:
        """Close every sink."""
        for sink in self.sinks:
            sink.close()


def current_execution_context() -> ExecutionContext | None:
    """Return the context bound in the current thread or task, or None."""
    return _CURRENT_CONTEXT.get()


def set_execution_context(context: ExecutionContext | None) -> Token[ExecutionContext | None]:
    """Bind a context for the rest of the current thread or task and return the reset token."""
    return _CURRENT_CONTEXT.set(context)


def reset_execution_context(token: Token[ExecutionContext | None]) -> None:
    """Restore the context that was bound before set_execution_context returned token."""
    _CURRENT_CONTEXT.reset(token)


@contextmanager
def bind_execution_context(context: ExecutionContext) -> Iterator[ExecutionContext]:
    """Bind context for the duration of the with block and restore the previous context afterwards."""
    token = _CURRENT_CONTEXT.set(context)
    try:
        yield context
    finally:
        _CURRENT_CONTEXT.reset(token)
```
