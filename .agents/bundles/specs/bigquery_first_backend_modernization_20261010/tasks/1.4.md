# Task 1.4: Use `if_generation_match=0` uploads and single-pass deletes in `GOEGcs`

## Objective

Remove redundant GCS round trips from `GOEGcs` (`src/goe/filesystem/goe_gcs.py`) while preserving its observable contract: no-overwrite uploads use the `ifGenerationMatch=0` precondition instead of `blob.exists()` (TOCTOU), and deletes drop per-blob `exists()` calls, the `_post_cloud_delete_wait` sleep and the re-list/`GOEDfsDeleteNotComplete` retry, still accepting concurrently vanished blobs.

## Context

- Baseline `46890a8`. `gcp_adc_credential_unification_20261010:1.2` changes `GOEGcs.__init__` (credential injection) and adds `tests/unit/filesystem/test_goe_gcs.py`; do not modify `__init__`.
- Anchors: imports `L8-27` (`GOEDfsDeleteNotComplete` imported at `L24`, used only by `delete`); `copy_from_local` (`L97-111`, `blob.exists()` check at `L109-110`); `delete` (`L129-184`: `@retry.Retry` predicate includes `GOEDfsDeleteNotComplete`; single-file branch `L155-165` calls `blob.exists()`, `pragmatic_delete`, `self._post_cloud_delete_wait(scheme)` and re-checks `exists()`; recursive branch `L166-182` calls `blob.exists()` per listed blob, waits, re-lists and raises `GOEDfsDeleteNotComplete`); `write` (`L295-307`, `blob.exists()` at `L305-306`).
- `GOEDfs._post_cloud_delete_wait` (`goe_dfs.py:305-309`) sleeps `POST_CLOUD_DELETE_WAIT_SECONDS = 0.2` (`L113`).
- Observable contract to preserve (characterized in Step 1): no-overwrite copy onto an existing object raises `GOEDfsException("Cannot copy file over existing file: <path>")`; `write` raises `GOEDfsException("Cannot write to existing file: <path>")`; single delete returns `True`, or `None` when the object is absent; recursive delete returns `True` (also when the prefix is empty) and removes every object under `<prefix>/`; NotFound during delete is accepted.
- Verified APIs (`google-cloud-storage==3.13.1`): `Blob.upload_from_filename(filename, ..., if_generation_match=None, ...)`, `Blob.upload_from_string(data, ..., if_generation_match=None, ...)`; `ifGenerationMatch=0` fails with HTTP 412 mapped to `google.api_core.exceptions.PreconditionFailed` when a live object exists (https://cloud.google.com/storage/docs/request-preconditions); `Bucket.delete_blob(blob_name, ...)` raises `NotFound`; `Bucket.delete_blobs(blobs, on_error=None, ...)` deletes per blob and calls `on_error(blob)` on NotFound. GCS listing is strongly consistent (https://cloud.google.com/storage/docs/consistency). `transfer_manager` has no delete function, so it is not used (see spec `BQ-D2`).
- Construct the client the same way `tests/unit/filesystem/test_goe_dfs.py::TestGOEGcs.setUp` (`L324`) does at execution time (baseline: `GOEGcs(messages, dry_run=True, do_not_connect=True)`), but with `dry_run=False`, then assign `gcs._client` to the fake.
- Strategy `characterization`: behavior-preserving. Initial evidence: the Step 1 contract tests pass against the unmodified module (the fake answers `exists()` and enforces `if_generation_match=0` from the same object store).

"""Characterization of GOEGcs upload and delete contracts against an in-memory GCS fake."""

from unittest.mock import patch

import pytest
from google.api_core import exceptions as google_exceptions

from goe.filesystem.goe_dfs import GOEDfsException
from goe.filesystem.goe_gcs import GOEGcs
from goe.offload.offload_messages import OffloadMessages


class FakeBlob:
    """Blob stand-in backed by FakeBucket.objects; counts exists() calls."""

    def __init__(self, bucket: "FakeBucket", name: str) -> None:
        """Bind the blob to its bucket."""
        self.bucket = bucket
        self.name = name
        self.size = 1

    def exists(self) -> bool:
        """Report whether the object is live."""
        self.bucket.exists_calls += 1
        return self.name in self.bucket.objects

    def upload_from_filename(self, filename: str, if_generation_match: int | None = None) -> None:
        """Upload from a local file."""
        self._store(if_generation_match)

    def upload_from_string(self, data: object, if_generation_match: int | None = None) -> None:
        """Upload from memory."""
        self._store(if_generation_match)

    def delete(self) -> None:
        """Delete the object or raise NotFound."""
        if self.name not in self.bucket.objects:
            raise google_exceptions.NotFound("missing")
        self.bucket.objects.discard(self.name)

    def _store(self, if_generation_match: int | None) -> None:
        """Store the object honouring ifGenerationMatch=0."""
        if if_generation_match == 0 and self.name in self.bucket.objects:
            raise google_exceptions.PreconditionFailed("object exists")
        self.bucket.objects.add(self.name)


class FakeBucket:
    """Bucket stand-in holding object names."""

    def __init__(self, objects: set[str]) -> None:
        """Seed the bucket."""
        self.objects = set(objects)
        self.exists_calls = 0

    def blob(self, name: str) -> FakeBlob:
        """Return a blob handle."""
        return FakeBlob(self, name)

    def delete_blob(self, name: str) -> None:
        """Delete one object by name."""
        FakeBlob(self, name).delete()

    def delete_blobs(self, blobs: list[FakeBlob], on_error=None) -> None:
        """Delete blobs, calling on_error for NotFound."""
        for blob in blobs:
            try:
                blob.delete()
            except google_exceptions.NotFound:
                if on_error is None:
                    raise
                on_error(blob)


class FakeClient:
    """Storage client stand-in."""

    def __init__(self, bucket: FakeBucket) -> None:
        """Wrap one bucket."""
        self.bucket = bucket

    def get_bucket(self, name: str) -> FakeBucket:
        """Return the bucket."""
        return self.bucket

    def list_blobs(self, bucket: FakeBucket, prefix: str = "", delimiter: str | None = None) -> list[FakeBlob]:
        """List blobs by prefix."""
        return [FakeBlob(bucket, n) for n in sorted(bucket.objects) if n.startswith(prefix)]


def test_copy_without_overwrite_onto_existing_raises() -> None:
    """No-overwrite copy onto a live object raises GOEDfsException."""
    gcs, _ = _gcs({"stage/part-0.parquet"})
    with pytest.raises(GOEDfsException, match="Cannot copy file over existing file"):
        gcs.copy_from_local("/tmp/part-0.parquet", "gs://b/stage/part-0.parquet", overwrite=False)


def test_copy_creates_and_overwrites() -> None:
    """Copies create new objects and overwrite when requested."""
    gcs, bucket = _gcs({"stage/part-0.parquet"})
    gcs.copy_from_local("/tmp/part-1.parquet", "gs://b/stage/part-1.parquet", overwrite=False)
    gcs.copy_from_local("/tmp/part-0.parquet", "gs://b/stage/part-0.parquet", overwrite=True)
    assert bucket.objects == {"stage/part-0.parquet", "stage/part-1.parquet"}


def test_write_without_overwrite_onto_existing_raises() -> None:
    """No-overwrite write onto a live object raises GOEDfsException."""
    gcs, _ = _gcs({"stage/ddl.sql"})
    with pytest.raises(GOEDfsException, match="Cannot write to existing file"):
        gcs.write("gs://b/stage/ddl.sql", "SELECT 1", overwrite=False)


def test_single_delete_returns_true_or_none() -> None:
    """Single-object delete returns True, or None when the object is absent."""
    gcs, bucket = _gcs({"stage/part-0.parquet"})
    assert gcs.delete("gs://b/stage/part-0.parquet") is True
    assert bucket.objects == set()
    assert gcs.delete("gs://b/stage/part-0.parquet") is None


def test_recursive_delete_removes_prefix() -> None:
    """Recursive delete removes every object under the prefix and returns True."""
    gcs, bucket = _gcs({"stage/t/part-0.parquet", "stage/t/part-1.parquet", "stage/other.parquet"})
    assert gcs.delete("gs://b/stage/t", recursive=True) is True
    assert bucket.objects == {"stage/other.parquet"}
    assert gcs.delete("gs://b/stage/empty", recursive=True) is True


def _gcs(objects: set[str]) -> tuple[GOEGcs, FakeBucket]:
    """Return an executing GOEGcs bound to a fake bucket."""
    gcs = GOEGcs(OffloadMessages(), dry_run=False, do_not_connect=True)
    bucket = FakeBucket(objects)
    gcs._client = FakeClient(bucket)
    return gcs, bucket

    @retry.Retry(
        predicate=retry.if_exception_type(ConnectionError, google_exceptions.ServiceUnavailable),
        deadline=DFS_RETRY_TIMEOUT,
    )
    def delete(self, dfs_path, recursive=False):
        """Delete a file, or a directory prefix and its contents, in a single pass.

        GCS listing is strongly consistent, so no post-delete wait or re-list is needed. Objects that vanish
        concurrently (404 NotFound) are accepted as deleted.
        """

        def accept_not_found(blob):
            self.debug("delete(%s) returned 404 (NotFound), accepting as deleted" % blob.name)

        assert dfs_path
        assert isinstance(dfs_path, str)
        logger.info("delete(%s, %s)" % (dfs_path, recursive))
        scheme, container, path = self._uri_component_split(dfs_path)
        self.debug("Delete scheme/container/path: %s" % str([scheme, container, path]))
        if self._dry_run:
            return None
        bucket = self._client.get_bucket(container)
        if not recursive or (self.stat(dfs_path) or {}).get("type") == DFS_TYPE_FILE:
            self.debug("delete(%s)" % path)
            try:
                bucket.delete_blob(path)
            except google_exceptions.NotFound:
                return None
            return True
        self.debug("Recursive delete using directory prefix: %s" % path)
        if not path.endswith(URI_SEP):
            path += URI_SEP
        bucket.delete_blobs(list(self._client.list_blobs(bucket, prefix=path)), on_error=accept_not_found)
        return True
