# Task 1.2: Adopt Parquet-default staging and native BigQuery load jobs in `BackendBigQueryTable`

## Objective

Default BigQuery staging format to `PARQUET` (while preserving `AVRO` when explicitly configured) and add native BigQuery batch load jobs (`load_table_from_uri` with `WriteDisposition.WRITE_TRUNCATE`, `decimal_target_types=["NUMERIC", "BIGNUMERIC", "STRING"]`, and CMEK `DestinationEncryptionConfiguration`) so `validate_type_conversions` and `load_final_table` query native columnar storage rather than re-scanning GCS through an external table, while preserving 100% of existing BigQuery features (external-table cast fallback, CMEK/KMS, partitioning, clustering, `BIGNUMERIC`, and dry-run logging).

## Context

1. **External Staging Table Double-Scan (`src/goe/offload/bigquery/bigquery_backend_table.py:105-138`, `L456-473`, `L512-514`, `L549-554`):**
   - Currently, `BackendBigQueryTable._create_load_table` (`L105-138`) calls `self._db_api.create_table(..., external=True)` (which delegates to `BackendBigQueryApi._create_external_table` at `src/goe/offload/bigquery/bigquery_backend_api.py:238-262`).
   - Subsequent `validate_type_conversions` (`L549-554`) and `load_final_table` (`L456-473`) execute `SELECT ... FROM <load_table>`, causing BigQuery to re-scan the staged GCS files multiple times and bill external table scan bytes under on-demand pricing.
   - BigQuery batch load jobs (`client.load_table_from_uri`) run on the free shared slot pool and populate a native BigQuery staging table. Because GOE's staging files encode decimals, dates, and timestamps as `STRING` columns (`parquet_staging_file.py:151-163`), loading the staged files into a native table with an explicit schema matching `staging_file.get_staging_columns()` keeps every string-staged column as `STRING` in the load table—meaning `validate_type_conversions` and `_staging_to_backend_cast` (`bigquery_backend_table.py:269-399`) require zero SQL changes and zero feature loss.
2. **Parquet Default & Decimal Target Types (`src/goe/offload/offload_constants.py:369-424`):**
   - `BIGQUERY_BACKEND_CAPABILITIES` (`src/goe/offload/offload_constants.py:370-424`) defines `CAPABILITY_STAGING_FILE_FORMATS: [FILE_STORAGE_FORMAT_AVRO, FILE_STORAGE_FORMAT_PARQUET]`. Placing `FILE_STORAGE_FORMAT_PARQUET` first in `BIGQUERY_BACKEND_CAPABILITIES[CAPABILITY_STAGING_FILE_FORMATS]` and adding `CAPABILITY_NATIVE_STAGE_LOAD = "CAPABILITY_NATIVE_STAGE_LOAD"` (defaulting to `False` via `.get()` in `BackendApiInterface`) makes Parquet and native staging load the BigQuery default without touching Snowflake, Synapse, Hadoop, or Teradata.
3. **Non-Abstract Base Interface Rule (`src/goe/offload/backend_api.py` & `src/goe/offload/backend_table.py`):**
   - Adding a new `@abstractmethod` on `BackendApiInterface` or `BackendTableInterface` would break all five deprioritized backends. Instead, `load_table_from_staging_uri` on `BackendApiInterface` must be a concrete method raising `NotImplementedError`, gated by `CAPABILITY_NATIVE_STAGE_LOAD`.

```python
def _build_api(dry_run: bool = False, kms_key: bool = False) -> tuple[BackendBigQueryApi, MagicMock]:
    """Construct a minimally initialized BackendBigQueryApi with a mocked BigQuery client."""
    api = object.__new__(BackendBigQueryApi)
    api._connection_options = MagicMock(
        bigquery_dataset_project="test-proj",
        bigquery_dataset_location="US",
        bigquery_kms_key_ring_location="us" if kms_key else None,
        bigquery_kms_key_ring_name="ring1" if kms_key else None,
        bigquery_kms_key_name="key1" if kms_key else None,
    )
    api._backend_type = DBTYPE_BIGQUERY
    api._dry_run = dry_run
    api._messages = MagicMock()
    api._repo_client = MagicMock()
    api._Context = None
    api._bigquery_dataset_project = "test-proj"
    api._bigquery_dataset_location = "US"
    api._sql_engine_name = "BigQuery"
    api._target_version = None
    api._session_user = "tester@example.com"
    client_mock = MagicMock(spec=bigquery.Client)
    api._client = client_mock
    return api, client_mock


def test_bigquery_capabilities_default_to_parquet_and_native_stage_load() -> None:
    """Verify Parquet is the primary staging format for BigQuery and native stage load is enabled."""
    formats = BIGQUERY_BACKEND_CAPABILITIES[CAPABILITY_STAGING_FILE_FORMATS]
    assert formats[0] == FILE_STORAGE_FORMAT_PARQUET
    assert FILE_STORAGE_FORMAT_AVRO in formats
    assert BIGQUERY_BACKEND_CAPABILITIES[CAPABILITY_NATIVE_STAGE_LOAD] is True


def test_load_table_from_staging_uri_parquet_with_cmek() -> None:
    """Verify native Parquet load job config sets WRITE_TRUNCATE, decimal_target_types, and CMEK."""
    api, client_mock = _build_api(dry_run=False, kms_key=True)
    load_job = MagicMock()
    load_job.output_rows = 250
    client_mock.load_table_from_uri.return_value = load_job

    columns = [
        BigQueryColumn("ID", "STRING"),
        BigQueryColumn("AMOUNT", "NUMERIC", data_precision=18, data_scale=2),
    ]
    rows = api.load_table_from_staging_uri(
        "load_ds",
        "orders_load",
        "gs://my-bucket/stage/orders/part*",
        FILE_STORAGE_FORMAT_PARQUET,
        columns=columns,
    )

    assert rows == 250
    client_mock.load_table_from_uri.assert_called_once()
    uri_arg, dest_arg = client_mock.load_table_from_uri.call_args.args[:2]
    job_config: bigquery.LoadJobConfig = client_mock.load_table_from_uri.call_args.kwargs["job_config"]
    assert uri_arg == "gs://my-bucket/stage/orders/part*"
    assert dest_arg == "test-proj.load_ds.orders_load"
    assert job_config.source_format == bigquery.SourceFormat.PARQUET
    assert job_config.write_disposition == bigquery.WriteDisposition.WRITE_TRUNCATE
    assert list(job_config.decimal_target_types) == ["NUMERIC", "BIGNUMERIC", "STRING"]
    assert job_config.destination_encryption_configuration is not None
    assert "keyRings/ring1/cryptoKeys/key1" in job_config.destination_encryption_configuration.kms_key_name
    assert [f.name for f in job_config.schema] == ["ID", "AMOUNT"]
    load_job.result.assert_called_once()


def test_load_table_from_staging_uri_dry_run_skips_client_call() -> None:

        columns=columns,
    )

    assert rows is None
    client_mock.load_table_from_uri.assert_not_called()
```

```python
# SPDX-FileCopyrightText: 2026 The GOE Authors
# SPDX-License-Identifier: Apache-2.0

    def load_table_from_staging_uri(
        self,
        db_name: str,
        table_name: str,
        staging_uri: str,
        staging_format: str,
        columns: list[Any] | None = None,
    ) -> int | None:
        """Load staged Parquet or Avro files from GCS into a native BigQuery staging table."""
        assert db_name and table_name and staging_uri and staging_format
        destination = (
            f"{self._bigquery_dataset_project}.{db_name}.{table_name}"
            if self._bigquery_dataset_project
            else f"{db_name}.{table_name}"
        )
        job_config = bigquery.LoadJobConfig(
            write_disposition=bigquery.WriteDisposition.WRITE_TRUNCATE,
        )
        if staging_format == FILE_STORAGE_FORMAT_PARQUET:
            job_config.source_format = bigquery.SourceFormat.PARQUET
            job_config.decimal_target_types = ["NUMERIC", "BIGNUMERIC", "STRING"]
        elif staging_format == FILE_STORAGE_FORMAT_AVRO:
            job_config.source_format = bigquery.SourceFormat.AVRO
            job_config.use_avro_logical_types = True
        else:
            raise BackendApiException(f"Unsupported BigQuery staging format: {staging_format}")

        if columns:
            job_config.schema = [
                bigquery.SchemaField(col.name, col.format_data_type()) for col in columns
            ]
        if self.bigquery_kms_key_name():
            job_config.destination_encryption_configuration = bigquery.EncryptionConfiguration(
                kms_key_name=self.bigquery_kms_key_name()
            )

        self._log(
            f"BigQuery LoadJob: {staging_uri} -> {destination} (format={staging_format})",
            detail=VERBOSE_LEVEL,
        )
        if self._dry_run:
            return None

        load_job = self._client.load_table_from_uri(
            staging_uri,
            destination,
            location=self._bigquery_dataset_location,
            job_config=job_config,
        )
        load_job.result()
        return load_job.output_rows
```
