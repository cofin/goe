# Task 1.2: Replace Legacy SQL partition metadata and modernize `BackendBigQueryApi`

## Objective

In `BackendBigQueryApi` (`src/goe/offload/bigquery/bigquery_backend_api.py`): replace Legacy SQL `$__PARTITIONS_SUMMARY__` metadata with parameterized GoogleSQL `INFORMATION_SCHEMA.PARTITIONS`, delete the `use_legacy_sql` branch, delete the redundant `__del__`, make `close()` idempotent with `__enter__`/`__exit__`, run non-profiled reads through `client.query_and_wait()`, attach sanitized GOE job labels, send `ClientInfo(user_agent="goe/<version>")`, extend the session allow-list, and set `default_kms_key_name` on datasets GOE creates.

    def get_table_partition_count(self, db_name, table_name):
        """Return the number of distinct partition ids from INFORMATION_SCHEMA.PARTITIONS.

        __NULL__ and __UNPARTITIONED__ are counted to keep the semantics of the former Legacy SQL
        __PARTITIONS_SUMMARY__ meta-table; DISTINCT covers __UNPARTITIONED__ rows reported per storage tier.
        """
        assert db_name and table_name
        if not self._table_is_partitioned(db_name, table_name):

        sql = (
            "SELECT COUNT(DISTINCT partition_id) FROM %s WHERE table_name = @table_name AND partition_id IS NOT NULL"
            % self._information_schema_partitions_reference(db_name)
        )
        row = self.execute_query_fetch_one(sql, query_params=[("table_name", "STRING", table_name)], log_level=VVERBOSE)
        return row[0] if row else None

    def get_table_partitions(self, db_name, table_name):
        """Return {partition_id: partition info} for every distinct partition id in INFORMATION_SCHEMA.PARTITIONS."""
        if not self._table_is_partitioned(db_name, table_name):
            return None
        sql = (
            "SELECT DISTINCT partition_id FROM %s WHERE table_name = @table_name AND partition_id IS NOT NULL"
            " ORDER BY partition_id" % self._information_schema_partitions_reference(db_name)
        )
        rows = self.execute_query_fetch_all(sql, query_params=[("table_name", "STRING", table_name)], log_level=VVERBOSE)
        return {str(_[0]): self._table_partition_info(partition_id=_[0]) for _ in rows}
