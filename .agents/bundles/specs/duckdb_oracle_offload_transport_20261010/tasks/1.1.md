# Task 1.1: Add `OFFLOAD_TRANSPORT_DUCKDB_*` orchestration options

## Objective

Expose eight `OrchestrationConfig` attributes that configure the opt-in DuckDB transport, with environment defaults and validation: `offload_transport_duckdb_version`, `_extension_path`, `_extension_dir`, `_extension_repository`, `_threads`, `_memory_limit`, `_temp_directory`, `_max_bytes`.

## Context

- Baseline `46890a8`; earlier chapters (`gcp_adc_credential_unification_20261010:1.1`, `legacy_shim_removal_and_repo_hygiene_20261010:1.2`, `bigquery_first_backend_modernization_20261010:1.1`) add/remove entries in the same files; anchor by symbol.
- `src/goe/config/orchestration_defaults.py`: transport defaults block `L670-821`; `offload_transport_dsn_default` (`L806-807`); `goe.util.misc_functions.is_pos_int(value, allow_zero=False)` validates positive integers (used by `posint_option_from_string`, `L67-70`, which raises `OrchestrationDefaultsException`, so it is not used here).
- `src/goe/config/orchestration_config.py`: `EXPECTED_CONFIG_ARGS` (`"offload_transport_dsn",` at `L124`), class annotation `offload_transport: str` (`L273`), `from_dict` keyword `offload_transport_dsn=config_dict.get(...)` (`L543-546`).
- `src/goe/config/config_validation_functions.py::normalise_offload_transport_config` (`L204-300`); its "Generic transport config" section ends with the `offload_transport_user` check (`L297-300`). `normalise_size_option(opt_val, binary_sizes=False, strict_name=None, exc_cls=...)` (`L399-411`); `human_size_to_bytes("20G", binary_sizes=True) == 21474836480` (verified locally).
- Option semantics (see `research/duckdb_integration.md` §3-§4):
  - `OFFLOAD_TRANSPORT_DUCKDB_VERSION` default `1.5.5` — duckdb-oracle v0.2.2 is built only for DuckDB 1.5.5 (https://github.com/cofin/duckdb-oracle/releases); GOE locks `duckdb==1.5.5` transitively (`uv.lock:886-887`).
  - `OFFLOAD_TRANSPORT_DUCKDB_EXTENSION_PATH` (unset) — local `oracle.duckdb_extension` file loaded with `LOAD '<path>'`.
  - `OFFLOAD_TRANSPORT_DUCKDB_EXTENSION_DIR` (unset) — DuckDB `extension_directory` with pre-installed `oracle`/`gcs`.
  - `OFFLOAD_TRANSPORT_DUCKDB_EXTENSION_REPOSITORY` default `https://cofin.github.io/duckdb-oracle`.
  - `OFFLOAD_TRANSPORT_DUCKDB_THREADS` default `2` (positive int), `OFFLOAD_TRANSPORT_DUCKDB_MEMORY_LIMIT` default `2GB` (DuckDB `memory_limit` string), per batch connection.
  - `OFFLOAD_TRANSPORT_DUCKDB_TEMP_DIRECTORY` default `$OFFLOAD_HOME/run/duckdb`.
  - `OFFLOAD_TRANSPORT_DUCKDB_MAX_BYTES` default `20G`, normalised to bytes (binary units).
- No call-timeout option: duckdb-oracle hard-codes a 30 s OCI call timeout and exposes no setting for it.
- Strategy `behavior_tdd`. Initial failing symptom: `AttributeError: 'OrchestrationConfig' object has no attribute 'offload_transport_duckdb_version'`.

```python
# SPDX-FileCopyrightText: 2026 The GOE Authors

"""OFFLOAD_TRANSPORT_DUCKDB_* configuration defaults and validation."""

import pytest

from goe.config.config_validation_functions import OrchestrationConfigException
from tests.unit.test_functions import FAKE_ORACLE_BQ_ENV, build_mock_options


def test_duckdb_transport_defaults() -> None:
    """Defaults match the duckdb-oracle v0.2.2 / DuckDB 1.5.5 build and conservative per-batch resources."""
    config = build_mock_options(FAKE_ORACLE_BQ_ENV)
    assert config.offload_transport_duckdb_version == "1.5.5"
    assert config.offload_transport_duckdb_extension_path is None
    assert config.offload_transport_duckdb_extension_dir is None
    assert config.offload_transport_duckdb_extension_repository == "https://cofin.github.io/duckdb-oracle"
    assert config.offload_transport_duckdb_threads == 2
    assert config.offload_transport_duckdb_memory_limit == "2GB"
    assert config.offload_transport_duckdb_temp_directory == "/tmp/offload/run/duckdb"
    assert config.offload_transport_duckdb_max_bytes == 20 * 1024**3


def test_duckdb_transport_overrides() -> None:
    """Environment overrides are honoured and sizes are normalised to bytes."""
    env = {
        **FAKE_ORACLE_BQ_ENV,
        "OFFLOAD_TRANSPORT_DUCKDB_THREADS": "4",
        "OFFLOAD_TRANSPORT_DUCKDB_MAX_BYTES": "512M",
        "OFFLOAD_TRANSPORT_DUCKDB_EXTENSION_DIR": "/opt/goe/duckdb_extensions",
        "OFFLOAD_TRANSPORT_DUCKDB_TEMP_DIRECTORY": "/var/tmp/goe_duckdb",
    }
    config = build_mock_options(env)
    assert config.offload_transport_duckdb_threads == 4
    assert config.offload_transport_duckdb_max_bytes == 512 * 1024**2
    assert config.offload_transport_duckdb_extension_dir == "/opt/goe/duckdb_extensions"
    assert config.offload_transport_duckdb_temp_directory == "/var/tmp/goe_duckdb"


@pytest.mark.parametrize(
    ("name", "value"), [("OFFLOAD_TRANSPORT_DUCKDB_THREADS", "0"), ("OFFLOAD_TRANSPORT_DUCKDB_MAX_BYTES", "lots")]
)
def test_duckdb_transport_invalid_values(name: str, value: str) -> None:
    """Non-positive thread counts and malformed sizes are rejected."""
    with pytest.raises(OrchestrationConfigException, match=name):
        build_mock_options({**FAKE_ORACLE_BQ_ENV, name: value})
```

```python
def offload_transport_duckdb_version_default() -> str:
    """Return the DuckDB version required by the configured duckdb-oracle extension build."""
    return os.environ.get("OFFLOAD_TRANSPORT_DUCKDB_VERSION") or "1.5.5"


def offload_transport_duckdb_extension_path_default() -> str | None:
    """Return an optional local path to an oracle.duckdb_extension file."""
    return os.environ.get("OFFLOAD_TRANSPORT_DUCKDB_EXTENSION_PATH") or None


def offload_transport_duckdb_extension_dir_default() -> str | None:
    """Return an optional DuckDB extension_directory containing pre-installed oracle and gcs extensions."""
    return os.environ.get("OFFLOAD_TRANSPORT_DUCKDB_EXTENSION_DIR") or None


def offload_transport_duckdb_extension_repository_default() -> str:
    """Return the custom repository serving unsigned duckdb-oracle builds."""
    return os.environ.get("OFFLOAD_TRANSPORT_DUCKDB_EXTENSION_REPOSITORY") or "https://cofin.github.io/duckdb-oracle"


def offload_transport_duckdb_threads_default() -> str:
    """Return the DuckDB thread count for each batch connection."""
    return os.environ.get("OFFLOAD_TRANSPORT_DUCKDB_THREADS") or "2"


def offload_transport_duckdb_memory_limit_default() -> str:
    """Return the DuckDB memory_limit for each batch connection."""
    return os.environ.get("OFFLOAD_TRANSPORT_DUCKDB_MEMORY_LIMIT") or "2GB"


def offload_transport_duckdb_temp_directory_default() -> str | None:
    """Return the DuckDB spill directory, defaulting to $OFFLOAD_HOME/run/duckdb."""
    if os.environ.get("OFFLOAD_TRANSPORT_DUCKDB_TEMP_DIRECTORY"):
        return os.environ["OFFLOAD_TRANSPORT_DUCKDB_TEMP_DIRECTORY"]
    if os.environ.get("OFFLOAD_HOME"):
        return os.path.join(os.environ["OFFLOAD_HOME"], "run", "duckdb")
    return None


def offload_transport_duckdb_max_bytes_default() -> str:
    """Return the maximum source table size eligible for DuckDB transport as a human-readable size."""
    return os.environ.get("OFFLOAD_TRANSPORT_DUCKDB_MAX_BYTES") or "20G"
```
