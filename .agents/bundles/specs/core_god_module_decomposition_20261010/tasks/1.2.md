```python
# SPDX-License-Identifier: Apache-2.0

"""HTML Jinja2 renderer for Offload Status Report."""

from pathlib import Path
from typing import Any

from jinja2 import Environment, FileSystemLoader

from goe.offload.offload_messages import VVERBOSE
from goe.util.goe_version import GOEVersion

TEMPLATES_DIR = Path(__file__).resolve().parents[2] / "templates"


def render_html_report(report: Any) -> str:
    """Render the Offload Status Report as HTML using the Jinja2 template."""
    report._messages.log("Generating HTML report", detail=VVERBOSE)
    env = Environment(loader=FileSystemLoader(str(TEMPLATES_DIR)))
    template = env.get_template("offload_status_report/goe_offload_status_report.html")
    return template.render(
        title="Offload Status Report",
        version=str(GOEVersion()),
        output_level=report._output_level,
        report=report._report_data,
    )
```
