## Objective

1. Create `src/goe/sync/__init__.py` and `src/goe/sync/differ.py` defining `ColumnAddition(FrozenBaseStruct)`, `TypeWidening(FrozenBaseStruct)`, `NullabilityRelaxation(FrozenBaseStruct)`, `PartitionDrop(FrozenBaseStruct)`, `SchemaDiff(FrozenBaseStruct)`, and `SchemaSyncService`.
2. Compute schema drift in a single pass between `OracleTableSnapshot` (`OracleCatalogService`) and backend `ColumnMetadata` (`CachedBackendCatalog`), and emit `ALTER TABLE` (`alter_add_columns`, `alter_column_set_data_type`, `alter_column_drop_not_null`) and hybrid view DDL through `DialectSqlEngine`.
3. Execute Oracle hybrid DDL via `OracleDatabaseRegistry.proxy_ddl_session` (fixing the `ORA-01017` proxy password defect and connection leak in `src/goe/schema_sync/steps/add_oracle_column.py:55-65`), acquire per-table locks via `OrchestrationLockInterface`, and write clean command files without trailing `}');` syntax errors.

  ```python
  # SPDX-FileCopyrightText: 2026 The GOE Authors
  # SPDX-License-Identifier: Apache-2.0

  """Single-pass schema drift analyzer and DDL synchronization service."""

  from collections.abc import Callable, Sequence
  from contextlib import AbstractContextManager
  from pathlib import Path
  from typing import Any

  from goe.offload.sql_ast.dialect_engine import SqlDialect, TableRef, get_dialect_engine
  from goe.typing import FrozenBaseStruct

  __all__ = (
      "ColumnAddition",
      "NullabilityRelaxation",
      "PartitionDrop",
      "SchemaDiff",
      "SchemaSyncService",
      "SourceColumnSnapshot",
      "TargetColumnSnapshot",
      "TypeWidening",
  )


  class SourceColumnSnapshot(FrozenBaseStruct):
      """Normalized source column metadata for schema drift comparison."""

      name: str
      canonical_type: str
      backend_sql_type: str
      data_precision: int | None = None
      data_scale: int | None = None
      char_length: int | None = None
      nullable: bool = True


  class TargetColumnSnapshot(FrozenBaseStruct):
      """Normalized backend column metadata for schema drift comparison."""

      name: str
      canonical_type: str
      backend_sql_type: str
      data_precision: int | None = None
      data_scale: int | None = None
      char_length: int | None = None
      nullable: bool = True


  class ColumnAddition(FrozenBaseStruct):
      """Represents a new column present in Oracle that must be added to the backend table."""

      column_name: str
      backend_sql_type: str
      nullable: bool = True


  class TypeWidening(FrozenBaseStruct):
      """Represents a compatible precision, scale, or length widening on an existing column."""

      column_name: str
      previous_sql_type: str
      target_sql_type: str


  class NullabilityRelaxation(FrozenBaseStruct):
      """Represents a NOT NULL to NULL relaxation on an existing column."""

      column_name: str


  class PartitionDrop(FrozenBaseStruct):
      """Represents a source partition dropped after offload that requires hybrid view refresh."""

      partition_name: str


  class SchemaDiff(FrozenBaseStruct):
      """Aggregated schema drift vector between an Oracle table snapshot and backend metadata."""

      source_owner: str
      source_table: str
      target_table: TableRef
      column_additions: tuple[ColumnAddition, ...] = ()
      type_widenings: tuple[TypeWidening, ...] = ()
      nullability_relaxations: tuple[NullabilityRelaxation, ...] = ()
      partition_drops: tuple[PartitionDrop, ...] = ()
      hybrid_view: TableRef | None = None
      view_select_tuples: tuple[tuple[str, str], ...] = ()

      @property
      def has_changes(self) -> bool:
          """Return True when at least one schema drift action is required."""
          return bool(
              self.column_additions or self.type_widenings or self.nullability_relaxations or self.partition_drops
          )


  class SchemaSyncService:
      """Computes SchemaDiff vectors and applies backend and hybrid DDL via DialectSqlEngine."""

      def __init__(
          self,
          *,
          dialect: SqlDialect,
          backend_ddl_executor: Callable[[Sequence[str]], None] | None = None,
          proxy_ddl_runner: Callable[[str, Sequence[str]], None] | None = None,
          table_lock_factory: Callable[[str, str, bool], AbstractContextManager[Any]] | None = None,
      ) -> None:
          self._dialect = dialect
          self._backend_ddl_executor = backend_ddl_executor
          self._proxy_ddl_runner = proxy_ddl_runner
          self._table_lock_factory = table_lock_factory

      def compute_diff(
          self,
          *,
          source_owner: str,
          source_table: str,
          target_table: TableRef,
          source_columns: Sequence[SourceColumnSnapshot],
          target_columns: Sequence[TargetColumnSnapshot],
          source_partitions: Sequence[str] = (),
          offloaded_partitions: Sequence[str] = (),
          hybrid_view: TableRef | None = None,
      ) -> SchemaDiff:
          """Compute a single-pass SchemaDiff between source snapshot and backend metadata."""
          target_by_name = {col.name.upper(): col for col in target_columns}
          additions: list[ColumnAddition] = []
          widenings: list[TypeWidening] = []
          relaxations: list[NullabilityRelaxation] = []

          for src in source_columns:
              tgt = target_by_name.get(src.name.upper())
              if tgt is None:
                  additions.append(
                      ColumnAddition(
                          column_name=src.name,
                          backend_sql_type=src.backend_sql_type,
                          nullable=src.nullable,
                      )
                  )
                  continue
              if self._is_widened(src, tgt):
                  widenings.append(
                      TypeWidening(
                          column_name=src.name,
                          previous_sql_type=tgt.backend_sql_type,
                          target_sql_type=src.backend_sql_type,
                      )
                  )
              if src.nullable and not tgt.nullable:
                  relaxations.append(NullabilityRelaxation(column_name=src.name))

          active_parts = {p.upper() for p in source_partitions}
          dropped = tuple(
              PartitionDrop(partition_name=part) for part in offloaded_partitions if part.upper() not in active_parts
          )
          select_tuples = tuple(
              (f"`{c.name}`" if self._dialect == "bigquery" else f'"{c.name}"', c.name) for c in source_columns
          )
          return SchemaDiff(
              source_owner=source_owner,
              source_table=source_table,
              target_table=target_table,
              column_additions=tuple(additions),
              type_widenings=tuple(widenings),
              nullability_relaxations=tuple(relaxations),
              partition_drops=dropped,
              hybrid_view=hybrid_view,
              view_select_tuples=select_tuples,
          )

      def generate_ddl(self, diff: SchemaDiff) -> tuple[str, ...]:
          """Render backend ALTER TABLE and hybrid CREATE OR REPLACE VIEW DDL statements."""
          if not diff.has_changes:
              return ()
          engine = get_dialect_engine(self._dialect)
          statements: list[str] = []
          if diff.column_additions:
              statements.append(
                  engine.alter_add_columns(
                      table=diff.target_table,
                      columns=[(col.column_name, col.backend_sql_type) for col in diff.column_additions],
                  )
              )
          for widening in diff.type_widenings:
              statements.append(
                  engine.alter_column_set_data_type(
                      table=diff.target_table,
                      column=widening.column_name,
                      data_type=widening.target_sql_type,
                  )
              )
          for relaxation in diff.nullability_relaxations:
              statements.append(
                  engine.alter_column_drop_not_null(
                      table=diff.target_table,
                      column=relaxation.column_name,
                  )
              )
          if diff.hybrid_view is not None and diff.view_select_tuples and self._dialect == "bigquery":
              from_table_sql = self._format_table_ref(diff.target_table)
              statements.append(
                  engine.create_view(
                      project=diff.hybrid_view.catalog,
                      db_name=diff.hybrid_view.schema or diff.source_owner,
                      view_name=diff.hybrid_view.name,
                      select_expr_tuples=list(diff.view_select_tuples),
                      from_tables=from_table_sql,
                      where_clauses=[],
                      replace=True,
                  )
              )
          return tuple(statements)

      def apply_diff(
          self,
          diff: SchemaDiff,
          *,
          execute: bool = False,
          command_file: Path | None = None,
          oracle_hybrid_ddl: Sequence[str] = (),
      ) -> tuple[str, ...]:
          """Execute or preview DDL for a SchemaDiff under table lock and optionally write a command file."""
          backend_stmts = self.generate_ddl(diff)
          all_stmts = (*backend_stmts, *oracle_hybrid_ddl)
          if not all_stmts:
              return ()

          if self._table_lock_factory is not None:
              with self._table_lock_factory(diff.source_owner, diff.source_table, not execute):
                  self._execute_statements(diff.source_owner, backend_stmts, oracle_hybrid_ddl, execute=execute)
          else:
              self._execute_statements(diff.source_owner, backend_stmts, oracle_hybrid_ddl, execute=execute)

          if command_file is not None:
              text = "\n".join(f"{stmt.rstrip(';')};" for stmt in all_stmts) + "\n"
              command_file.write_text(text, encoding="utf-8")
          return all_stmts

      def _execute_statements(
          self,
          source_owner: str,
          backend_stmts: Sequence[str],
          oracle_hybrid_ddl: Sequence[str],
          *,
          execute: bool,
      ) -> None:
          if not execute:
              return
          if backend_stmts and self._backend_ddl_executor is not None:
              self._backend_ddl_executor(backend_stmts)
          if oracle_hybrid_ddl and self._proxy_ddl_runner is not None:
              self._proxy_ddl_runner(source_owner, oracle_hybrid_ddl)

      @staticmethod
      def _is_widened(src: SourceColumnSnapshot, tgt: TargetColumnSnapshot) -> bool:
          if src.backend_sql_type == tgt.backend_sql_type:
              return False
          if (
              src.data_precision is not None
              and tgt.data_precision is not None
              and src.data_precision > tgt.data_precision
          ):
              return True
          if src.data_scale is not None and tgt.data_scale is not None and src.data_scale > tgt.data_scale:
              return True
          return src.char_length is not None and tgt.char_length is not None and src.char_length > tgt.char_length

      def _format_table_ref(self, ref: TableRef) -> str:
          parts = [p for p in (ref.catalog, ref.schema, ref.name) if p]
          if self._dialect == "bigquery":
              return ".".join(f"`{p}`" for p in parts)
          return ".".join(f'"{p}"' for p in parts)
  ```
