# Task 1.2: Migrate `OracleOrchestrationRepoClient` and `schemas.py` to `OracleGateway` and `msgspec.Struct` and Delete Inline SQL

## Objective

Expand `src/goe/persistence/schemas.py` (`L1-149`) with `msgspec.Struct` repository projection models, refactor `src/goe/persistence/oracle/oracle_orchestration_repo_client.py` (`L83-1035`) to execute all queries and PL/SQL calls via `OracleGateway` (`src/goe/persistence/oracle_gateway.py`) and `src/goe/persistence/sql/oracle_repo.sql`, and **delete all 550+ lines of replaced inline f-string SQL and raw cursor `OUT`-bind boilerplate**.

## Context

- `src/goe/persistence/schemas.py` (`L68-132`) currently defines `ColumnDetail`, `PartitionDetail`, and `SubPartitionDetail` with `.from_orm(cls, obj)` classmethods that expect attribute access on namedtuple rows returned by `OracleFrontendApi.execute_query_fetch_all(..., as_named_tuple=True)`.
- `src/goe/persistence/oracle/oracle_orchestration_repo_client.py` currently:
  - Calls `self._frontend_api.get_oracle_connection_object()` directly in `start_command` (`L270`), `start_command_step` (`L311`), and `start_offload_chunk` (`L354`) to allocate `conn_obj.cursor().var(int)` `OUT` binds.
  - Calls `self._frontend_api.get_ora_type_object(...)` in `_metadata_dict_to_ora_object` (`L145`) and `_offload_partitions_to_ora_object` (`L197-198`).
  - Embeds 550+ lines of raw f-string SQL across `get_offloadable_schemas` (`L430-510`), `get_schema_tables` (`L511-876`), `get_table_columns` (`L878-905`), `get_table_partitions` (`L907-923`), `get_table_subpartitions` (`L925-940`), `get_command_step_codes` (`L942-945`), `get_command_executions` (`L947-973`), `get_command_execution` (`L975-1001`), and `get_command_execution_steps` (`L1003-1035`).

# SPDX-FileCopyrightText: 2026 The GOE Authors

import datetime
from unittest.mock import MagicMock
import uuid

from goe.offload.offload_messages import OffloadMessages
from goe.persistence.oracle.oracle_orchestration_repo_client import (
    OracleOrchestrationRepoClient,
)
from goe.persistence.schemas import ColumnDetail, PartitionDetail, SubPartitionDetail


def _make_mock_options() -> MagicMock:
    """Build a mock connection options object for OracleOrchestrationRepoClient."""
    opts = MagicMock()
    opts.ora_repo_user = "GOE_REPO"
    opts.ora_adm_user = "GOE_ADM"
    opts.ora_app_user = "GOE_APP"
    opts.rdbms_app_user = "GOE_APP"
    opts.backend_distribution = None
    opts.target = "bigquery"
    return opts


def test_column_and_partition_detail_from_row_mapping() -> None:
    """Verify ColumnDetail, PartitionDetail, and SubPartitionDetail hydrate from dict rows."""
    col = ColumnDetail.from_row(
        {
            "column_id": 1,
            "column_name": "ID",
            "data_type": "NUMBER",
            "nullable": "N",
            "data_default": None,
            "hidden_column": "NO",
            "virtual_column": "NO",
            "avg_col_len": 5,
            "num_distinct": 100,
        }
    )
    assert col.column_name == "ID"
    assert col.column_id == 1

    part = PartitionDetail.from_row(
        {
            "partition_name": "P_2025",
            "partition_position": 1,
            "high_value": "TO_DATE(' 2026-01-01 00:00:00', 'SYYYY-MM-DD HH24:MI:SS')",
            "subpartition_count": 0,
            "num_rows": 500,
            "avg_row_len": 32,
            "part_size": 16384,
        }
    )
    assert part.partition_name == "P_2025"
    assert part.subpartitions == []

    subpart = SubPartitionDetail.from_row(
        {
            "partition_name": "P_2025",
            "subpartition_name": "SP_1",
            "subpartition_position": 1,
            "high_value": "'US'",
            "num_rows": 250,
            "avg_row_len": 32,
            "subpart_size": 8192,
        }
    )
    assert subpart.subpartition_name == "SP_1"


def test_get_offloadable_schemas_uses_named_binds() -> None:
    """Verify get_offloadable_schemas passes adm_user, app_user, and repo_user as named binds."""
    opts = _make_mock_options()
    messages = OffloadMessages()
    client = OracleOrchestrationRepoClient(opts, messages, dry_run=False)
    mock_gateway = MagicMock()
    mock_gateway.select_dicts.return_value = [
        {
            "schema_name": "SH",
            "created": datetime.datetime(2026, 1, 1, 12, 0, 0),
            "User_type": "A",
            "schema_size": 1024,
            "tot_tables": 10,
            "tot_iot_tables": 0,
            "offloadable_tables": 10,
            "offloaded_tables": 5,
            "incremental_tables": 2,
            "imd_tables": 1,
            "hybrid_objects": 5,
            "offloaded_Attempted_tables": 5,
            "reclaimable_space": 512,
            "Unsupported_data_types": "N",
        }
    ]
    client._gateway = mock_gateway

    rows = client.get_offloadable_schemas()
    assert len(rows) == 1
    assert rows[0]["schema_name"] == "SH"
    assert rows[0]["created"] == "2026-01-01T12:00:00"
    call_kwargs = mock_gateway.select_dicts.call_args.kwargs
    assert call_kwargs["params"] == {
        "adm_user": "GOE_ADM",
        "app_user": "GOE_APP",
        "repo_user": "GOE_REPO",
    }


def test_get_command_execution_hydrates_uuid_and_json_parameters() -> None:

        "offloaded_rows": 100,
        "transport_bytes": 2048,
        "source_owner": "SH",
        "source_table": "SALES",
        "backend_type": "GOOGLE",
        "backend_Owner": "sh_dataset",
        "backend_table": "sales",

    client._gateway = mock_gateway

    record = client.get_command_execution(exec_uuid)
    assert record["execution_id"] == exec_uuid
    assert record["parameters"] == {"owner_table": "SH.SALES"}
