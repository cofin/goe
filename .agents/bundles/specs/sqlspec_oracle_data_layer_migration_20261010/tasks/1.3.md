# Task 1.3: Move Repository SQL to `oracle_repo.sql` with `msgspec.Struct` Hydration

## Objective

Move the 7 repository statements embedded in `OracleOrchestrationRepoClient` into `src/goe/persistence/sql/oracle_repo.sql`, execute them through `OracleGateway` with `msgspec.Struct` row validation, bind the three user names instead of interpolating them, validate the schema-name and column-name template slots, and replace the three `conn_obj.cursor().var(int)` OUT-bind allocations with `OracleGateway.new_out_var(int)`. Returned shapes stay identical (`list[dict]` / `dict | None` with lower-case keys and raw values).

```bash
    "upper('{self._connection_options.ora_repo_user}')"

def method_sql(tree, method):
    for node in ast.walk(tree):
        if isinstance(node, ast.FunctionDef) and node.name == method:
            for stmt in node.body:
                if (
                    isinstance(stmt, ast.Assign)
                    and isinstance(stmt.targets[0], ast.Name)
                    and stmt.targets[0].id == "sql"
                    and isinstance(stmt.value, ast.JoinedStr)
                ):
                    return "".join(
                        v.value if isinstance(v, ast.Constant) else "{" + ast.unparse(v.value) + "}"
                        for v in stmt.value.values
                    )
    raise SystemExit(f"No f-string sql assignment found in {method}")


def dedent_block(text):
    lines = [line.rstrip() for line in text.strip("\n").splitlines()]
    indent = min(len(line) - len(line.lstrip()) for line in lines if line.strip())
    return "\n".join(line[indent:] for line in lines).strip()


tree = ast.parse(SOURCE.read_text())
offloadable = method_sql(tree, "get_offloadable_schemas")
if LEGACY_USERS not in offloadable:
    raise SystemExit("get_offloadable_schemas user exclusion anchor not found")
offloadable = offloadable.replace(LEGACY_USERS, "UPPER(:adm_user), UPPER(:app_user), UPPER(:repo_user)")
schema_tables = (
    method_sql(tree, "get_schema_tables")
    .replace(REPO_USER, "{repo_user}")
    .replace("uc.unsupported_col_name_{unsupported_column_pattern}_count", "uc.{unsupported_col_name_count}")
    .replace("uc.unsupported_col_name_{unsupported_column_pattern}", "uc.{unsupported_col_name}")
)
steps = dedent_block(method_sql(tree, "get_command_execution_steps").replace(REPO_USER, "{repo_user}"))
statements = {
    "get_offloadable_schemas": dedent_block(offloadable),
    "get_schema_tables": dedent_block(schema_tables),
    "get_command_step_codes": dedent_block(method_sql(tree, "get_command_step_codes").replace(REPO_USER, "{repo_user}")),
    "get_command_executions": dedent_block(method_sql(tree, "get_command_executions").replace(REPO_USER, "{repo_user}")),
    "get_command_execution": dedent_block(method_sql(tree, "get_command_execution").replace(REPO_USER, "{repo_user}")),
    "get_command_execution_steps_all": steps,
    "get_command_execution_steps_by_id": steps + "\nWHERE CE.UUID = :execution_id",
}
for name, text in statements.items():
    if "{self." in text or "{unsupported_column_pattern}" in text:
        raise SystemExit(f"Unconverted interpolation left in {name}")
TARGET.parent.mkdir(parents=True, exist_ok=True)
TARGET.write_text("\n\n".join(f"-- name: {name}\n{text};" for name, text in statements.items()) + "\n")
print(f"Wrote {len(statements)} statements to {TARGET}")
```

```python
    @property
    def _gateway(self) -> OracleGateway:
        """Return the lazily created gateway for ``oracle_repo.sql`` statements.

        Created on first use so constructing the client never validates ``ora_repo_user`` or touches a connection.
        """
        if self._repo_gateway is None:
            self._repo_gateway = OracleGateway(
                lambda: self._frontend_api.get_validated_oracle_connection(),
                load_sql_file(ORACLE_REPO_SQL_PATH),
                slots={"repo_user": self._repo_user},
            )
        return self._repo_gateway

    def get_offloadable_schemas(self):
        """Return offloadable schema rows as dicts.

        User names are bound as ``str()`` values so a missing option keeps the legacy ``upper('None')`` comparison
        instead of a ``NOT IN (..., NULL)`` predicate that would exclude every schema.
        """
        rows = self._gateway.select_structs(
            "get_offloadable_schemas",
            OffloadableSchemaRow,
            {
                "adm_user": str(self._connection_options.ora_adm_user),
                "app_user": str(self._connection_options.ora_app_user),
                "repo_user": str(self._connection_options.ora_repo_user),
            },
        )
        return [msgspec.structs.asdict(row) for row in rows or []]

    def get_schema_tables(self, schema_name):
        """Return schema tables for the given schema using the backend's unsupported column pattern."""
        backend = self._connection_options.backend_distribution
        if backend == BACKEND_DISTRO_CDH:
            unsupported_column_pattern = "1"
        elif backend == BACKEND_DISTRO_GCP:
            unsupported_column_pattern = "2"
        elif backend == BACKEND_DISTRO_SNOWFLAKE:
            unsupported_column_pattern = "3"
        elif backend == BACKEND_DISTRO_MSAZURE:
            unsupported_column_pattern = "4"
        rows = self._gateway.select_structs(
            "get_schema_tables",
            SchemaTableRow,
            {"schema_name": schema_name},
            slots={
                "unsupported_col_name": f"unsupported_col_name_{unsupported_column_pattern}",
                "unsupported_col_name_count": f"unsupported_col_name_{unsupported_column_pattern}_count",
            },
        )
        return [msgspec.structs.asdict(row) for row in rows or []]
```
