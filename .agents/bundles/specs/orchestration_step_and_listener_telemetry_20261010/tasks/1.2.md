# Task 1.2: `TraceContextLogFilter`, `GOELogFileHandle` JSONL Enrichment & Warning Span Events

## Objective

Enrich GOE's Python `logging`, Structlog (Listener), and `GOELogFileHandle` / `OffloadMessages` output with active `trace_id`, `span_id`, `execution_id`, and `step_id` correlation while preserving 100% byte-for-byte CLI stdout compatibility in default text mode:
1. Create `src/goe/observability/logging.py` with:
   - `StructuredLogEntry(msgspec.Struct, frozen=True, kw_only=True)` for fast `msgspec.json.encode()` serialization.
   - `TraceContextLogFilter(logging.Filter)` that injects `trace_id`, `span_id`, `execution_id`, and `step_id` onto every `logging.LogRecord`.
   - `structlog_trace_context_processor(logger, method_name, event_dict)` for `src/goe/listener/config/logging.py`.
   - `record_log_span_event(level, message)` that attaches `log.warning` / `log.error` events onto the current recording span.
2. Enhance `GOELogFileHandle` (`src/goe/util/goe_log_fh.py:28-74`) to support:
   - Default `log_format="text"`: identical plain-text output to the `.log` file.
   - `log_format="json"`: writes `StructuredLogEntry` JSON lines to the log file.
   - `log_json_sidecar=True`: writes plain text to `<path>.log` and simultaneously writes `StructuredLogEntry` JSON lines to `<path>.log.jsonl`.
3. Update `OffloadMessages.warning()` (`src/goe/offload/offload_messages.py:345-351`) to invoke `record_log_span_event("WARNING", line)`.

```python
# SPDX-FileCopyrightText: 2026 The GOE Authors
# SPDX-License-Identifier: Apache-2.0

"""Structured log correlation filter, Structlog processor, and JSONL encoder for GOE."""

from __future__ import annotations

import logging
from collections.abc import MutableMapping
from datetime import datetime, timezone
from typing import Any

import msgspec
from opentelemetry import trace

from goe.observability.context import get_current_trace_context

__all__ = [
    "StructuredLogEntry",
    "TraceContextLogFilter",
    "encode_structured_log_line",
    "record_log_span_event",
    "structlog_trace_context_processor",
]


class StructuredLogEntry(msgspec.Struct, frozen=True, kw_only=True):
    """Structured JSONL log record correlated with OpenTelemetry and GOE ExecutionId."""

    timestamp: str
    level: str
    message: str
    execution_id: str | None = None
    command_type: str | None = None
    step_id: str | None = None
    trace_id: str | None = None
    span_id: str | None = None


class TraceContextLogFilter(logging.Filter):
    """Logging filter that attaches trace_id, span_id, execution_id, and step_id to LogRecords."""

    def filter(self, record: logging.LogRecord) -> bool:
        """Enrich LogRecord with active trace and execution context attributes."""
        snap = get_current_trace_context()
        record.trace_id = snap.trace_id or ""
        record.span_id = snap.span_id or ""
        record.execution_id = snap.execution_id or ""
        record.step_id = snap.step_id or ""
        record.command_type = snap.command_type or ""
        return True


def structlog_trace_context_processor(
    logger: Any,
    method_name: str,
    event_dict: MutableMapping[str, Any],
) -> MutableMapping[str, Any]:
    """Structlog processor injecting active trace_id, span_id, execution_id, and step_id."""
    snap = get_current_trace_context()
    if snap.trace_id:
        event_dict["trace_id"] = snap.trace_id
    if snap.span_id:
        event_dict["span_id"] = snap.span_id
    if snap.execution_id:
        event_dict["execution_id"] = snap.execution_id
    if snap.step_id:
        event_dict["step_id"] = snap.step_id
    if snap.command_type:
        event_dict["command_type"] = snap.command_type
    return event_dict


def encode_structured_log_line(message: str, level: str = "INFO") -> bytes:
    """Encode a log message with current trace context as a UTF-8 JSONL byte string."""
    snap = get_current_trace_context()
    entry = StructuredLogEntry(
        timestamp=datetime.now(timezone.utc).isoformat(),
        level=level,
        message=message.rstrip("\n"),
        execution_id=snap.execution_id,
        command_type=snap.command_type,
        step_id=snap.step_id,
        trace_id=snap.trace_id,
        span_id=snap.span_id,
    )
    return msgspec.json.encode(entry) + b"\n"


def record_log_span_event(level: str, message: str) -> None:
    """Attach a warning or error log line as an event on the currently recording span."""
    span = trace.get_current_span()
    if span is None or not span.is_recording():
        return
    snap = get_current_trace_context()
    attrs: dict[str, str] = {
        "log.severity": level.upper(),
        "log.message": message,
    }
    if snap.step_id:
        attrs["goe.step.id"] = snap.step_id
    if snap.execution_id:
        attrs["goe.execution.id"] = snap.execution_id
    span.add_event(f"log.{level.lower()}", attributes=attrs)
```
