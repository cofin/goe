# Task 1.1: `OrchestrationRunner` Command Spans & `OffloadMessages.offload_step()` Telemetry

## Objective

1. Create `src/goe/observability/spans.py` providing context managers `traced_command(...)` and `traced_step(...)`:
   - `traced_command` binds `ExecutionId`, starts `goe.command.<command_type_lower>` (`SpanKind.INTERNAL`), sets `goe.execution.id`, `goe.command.type`, `goe.command.dry_run`, `goe.source.owner`, `goe.source.table`, and `goe.backend.target`, records exceptions and `Status(StatusCode.ERROR)` on failure, and emits `goe.commands.total` and `goe.command.duration` via `GOEMetrics.record_command()`.

```python
# SPDX-FileCopyrightText: 2026 The GOE Authors
# SPDX-License-Identifier: Apache-2.0

"""Command and step span helpers for GOE orchestration."""

from __future__ import annotations

import time
from collections.abc import Iterator, Mapping
from contextlib import contextmanager
from typing import TYPE_CHECKING, Any

from opentelemetry.trace import Span, SpanKind, Status, StatusCode

from goe.observability.context import bind_execution_context, bind_step_context
from goe.observability.runtime import get_observability_runtime, get_tracer

if TYPE_CHECKING:
    from goe.orchestration.execution_id import ExecutionId

__all__ = [
    "record_step_details_on_span",
    "traced_command",
    "traced_step",
]


@contextmanager
def traced_command(
    command_type: str,
    execution_id: ExecutionId | str | None,
    backend: str = "bigquery",
    dry_run: bool = False,
    source_owner: str | None = None,
    source_table: str | None = None,
) -> Iterator[Span]:
    """Start a root or child command span, bind ExecutionId context, and record command metrics."""
    cmd_clean = (command_type or "unknown").lower()
    span_name = f"goe.command.{cmd_clean}"
    tracer = get_tracer()
    runtime = get_observability_runtime()
    start_ts = time.perf_counter()
    status_label = "success"

    with bind_execution_context(
        execution_id=execution_id,
        command_type=command_type,
        source_owner=source_owner,
        source_table=source_table,
    ):
        with tracer.start_as_current_span(span_name, kind=SpanKind.INTERNAL) as span:
            if span.is_recording():
                span.set_attribute("goe.command.type", command_type)
                span.set_attribute("goe.command.dry_run", dry_run)
                span.set_attribute("goe.backend.target", backend or "unknown")
                if execution_id is not None:
                    span.set_attribute("goe.execution.id", str(execution_id))
                if source_owner:
                    span.set_attribute("goe.source.owner", source_owner)
                if source_table:
                    span.set_attribute("goe.source.table", source_table)
            try:
                yield span
                if span.is_recording():
                    span.set_status(Status(StatusCode.OK))
            except Exception as exc:
                status_label = "error"
                if span.is_recording():
                    span.record_exception(exc)
                    span.set_status(Status(StatusCode.ERROR, str(exc)))
                raise
            finally:
                duration_s = time.perf_counter() - start_ts
                runtime.metrics.record_command(
                    command_type=command_type,
                    status=status_label,
                    backend=backend or "unknown",
                    dry_run=dry_run,
                    duration_s=duration_s,
                )


@contextmanager
def traced_step(
    step_id: str,
    step_code: str,
    step_title: str,
    command_type: str | None = None,
    mandatory: bool = False,
    optional: bool = False,
    execute: bool = True,
    skipped: bool = False,
) -> Iterator[Span]:
    """Start an orchestration step child span, bind step_id context, and record step duration metrics."""
    span_name = f"goe.step.{step_id}"
    tracer = get_tracer()
    runtime = get_observability_runtime()
    start_ts = time.perf_counter()
    status_label = "skipped" if skipped else "success"

    with bind_step_context(step_id):
        with tracer.start_as_current_span(span_name, kind=SpanKind.INTERNAL) as span:
            if span.is_recording():
                span.set_attribute("goe.step.id", step_id)
                span.set_attribute("goe.step.code", step_code)
                span.set_attribute("goe.step.title", step_title)
                span.set_attribute("goe.step.mandatory", mandatory)
                span.set_attribute("goe.step.optional", optional)
                span.set_attribute("goe.step.execute", execute)
                if command_type:
                    span.set_attribute("goe.step.command_type", command_type)
            if skipped:
                if span.is_recording():
                    span.set_attribute("goe.step.status", "skipped")
                    span.set_status(Status(StatusCode.OK))
                runtime.metrics.record_step(
                    step_id=step_id,
                    command_type=command_type or "unknown",
                    status="skipped",
                    mandatory=mandatory,
                    duration_s=0.0,
                )
                yield span
                return

            try:
                yield span
                if span.is_recording():
                    span.set_attribute("goe.step.status", "success")
                    span.set_status(Status(StatusCode.OK))
            except Exception as exc:
                status_label = "error"
                if span.is_recording():
                    span.set_attribute("goe.step.status", "error")
                    span.record_exception(exc)
                    span.set_status(Status(StatusCode.ERROR, str(exc)))
                raise
            finally:
                duration_s = time.perf_counter() - start_ts
                runtime.metrics.record_step(
                    step_id=step_id,
                    command_type=command_type or "unknown",
                    status=status_label,
                    mandatory=mandatory,
                    duration_s=duration_s,
                )


def record_step_details_on_span(span: Span, step_details: Mapping[str, Any] | None) -> None:
    """Attach scalar entries from step_details onto an active step span."""
    if not step_details or not span.is_recording():
        return
    for key, val in step_details.items():
        if isinstance(val, (bool, int, float, str)):
            span.set_attribute(f"goe.step.detail.{key}", val)
```
