# SPDX-FileCopyrightText: 2026 The GOE Authors
# SPDX-License-Identifier: Apache-2.0

import sqlglot
from sqlglot import exp

from goe.offload.bigquery.bigquery_column import (
    BIGQUERY_TYPE_BIGNUMERIC,
    BIGQUERY_TYPE_DATE,
    BIGQUERY_TYPE_INT64,
    BIGQUERY_TYPE_NUMERIC,
    BIGQUERY_TYPE_STRING,
    BigQueryColumn,
)
from goe.offload.bigquery.bigquery_sql import (
    BigQuerySqlEngine,
    get_bigquery_dialect_engine,
)
from goe.offload.column_metadata import ColumnPartitionInfo
from goe.offload.sql_engine import BigQueryDialectEngineProtocol


def test_bigquery_dialect_engine_satisfies_protocol() -> None:
    """Verify BigQuerySqlEngine satisfies BigQueryDialectEngineProtocol."""
    engine = get_bigquery_dialect_engine()
    assert isinstance(engine, BigQueryDialectEngineProtocol)
    assert isinstance(engine, BigQuerySqlEngine)


def test_create_table_with_partition_cluster_and_options_parity() -> None:
    """Verify create_table returns BigQuery DDL string matching legacy semantics and escaping options."""
    engine = get_bigquery_dialect_engine()
    cols = [
        BigQueryColumn("ID", BIGQUERY_TYPE_INT64, nullable=False),
        BigQueryColumn("AMT", BIGQUERY_TYPE_BIGNUMERIC, data_precision=38, data_scale=0, nullable=True),
        BigQueryColumn("PRICE", BIGQUERY_TYPE_NUMERIC, data_precision=38, data_scale=9, nullable=True),
        BigQueryColumn(
            "SALE_DATE",
            BIGQUERY_TYPE_DATE,
            nullable=True,
            partition_info=ColumnPartitionInfo(position=1, granularity="M"),
        ),
    ]
    part_clause = engine.partition_by_clause(cols[3])
    rendered = engine.create_table(
        project="my-proj",
        db_name="sh_ds",
        table_name="sales",
        columns=cols,
        partition_expr_sql=part_clause,
        cluster_cols=["ID", "AMT"],
        options={"kms_key_name": "projects/p/locations/l/keyRings/r/cryptoKeys/k", "description": "o'brien"},
        replace=True,
    )
    assert isinstance(rendered, str)
    assert rendered.startswith("CREATE OR REPLACE TABLE `my-proj`.`sh_ds`.`sales`")
    assert "BIGNUMERIC(38, 0)" in rendered
    assert "PARTITION BY DATE_TRUNC(`SALE_DATE`, MONTH)" in rendered
    assert "CLUSTER BY `ID`, `AMT`" in rendered
    assert "kms_key_name='projects/p/locations/l/keyRings/r/cryptoKeys/k'" in rendered
    parsed = sqlglot.parse_one(rendered, read="bigquery")
    assert isinstance(parsed, exp.Create)


def test_create_external_table_escapes_uris_and_format() -> None:
    """Verify create_external_table builds safe OPTIONS literals and returns str."""
    engine = get_bigquery_dialect_engine()
    cols = [BigQueryColumn("ID", BIGQUERY_TYPE_INT64), BigQueryColumn("NAME", BIGQUERY_TYPE_STRING)]
    rendered = engine.create_external_table(
        project="my-proj",
        db_name="sh_stage",
        table_name="sales_ext",
        columns=cols,
        storage_format="PARQUET",
        uris=["gs://bucket/path/*.parquet"],
        replace=True,
    )
    assert "CREATE OR REPLACE EXTERNAL TABLE `my-proj`.`sh_stage`.`sales_ext`" in rendered
    assert "uris=['gs://bucket/path/*.parquet']" in rendered


def test_insert_select_and_alter_add_columns() -> None:
    """Verify insert_select and alter_add_columns generate valid BigQuery DML and DDL strings."""
    engine = get_bigquery_dialect_engine()
    ins_sql = engine.insert_select(
        project="my-proj",
        db_name="sh_ds",
        table_name="sales",
        insert_col_names=["ID", "NAME"],
        select_expr_tuples=[("CAST(`ID` AS INT64)", "ID"), ("`NAME`", "NAME")],
        from_project="my-proj",
        from_db_name="sh_stage",
        from_table_name="sales_ext",
        where_clauses=["`ID` > 100"],
    )
    assert ins_sql.startswith("INSERT INTO `my-proj`.`sh_ds`.`sales` (`ID`, `NAME`) SELECT")
    assert "WHERE `ID` > 100" in ins_sql

    alter_sql = engine.alter_add_columns(
        project="my-proj",
        db_name="sh_ds",
        table_name="sales",
        columns=[BigQueryColumn("NEW_COL", BIGQUERY_TYPE_STRING, nullable=True)],
    )
    assert alter_sql == "ALTER TABLE `my-proj`.`sh_ds`.`sales` ADD COLUMN `NEW_COL` STRING"

### Step 2: Create `src/goe/offload/sql_engine.py`

Create `src/goe/offload/sql_engine.py` defining the library-agnostic protocols and domain value objects (`RenderedSQL`, `ValidationQuerySpec`, `BigQueryDialectEngineProtocol`, `OracleExtractionQueryProtocol`, `PredicateDialectEngineProtocol`, `ValidationQueryEngineProtocol`):

```python
# SPDX-FileCopyrightText: 2026 The GOE Authors
# SPDX-License-Identifier: Apache-2.0

"""Library-agnostic SQL dialect and AST engine protocols and value objects."""

from collections.abc import Mapping, Sequence
from typing import TYPE_CHECKING, Any, Protocol, runtime_checkable

import msgspec

if TYPE_CHECKING:
    from goe.offload.column_metadata import ColumnMetadataInterface

__all__ = [
    "BigQueryDialectEngineProtocol",
    "OracleExtractionQueryProtocol",
    "PredicateDialectEngineProtocol",
    "RenderedSQL",
    "ValidationQueryEngineProtocol",
    "ValidationQuerySpec",
]


class RenderedSQL(msgspec.Struct, frozen=True, kw_only=True):
    """Rendered SQL text paired with named bind parameters."""

    sql: str
    params: dict[str, Any] = msgspec.field(default_factory=dict)


class ValidationQuerySpec(msgspec.Struct, frozen=True, kw_only=True):
    """Domain specification for constructing a cross-DB aggregate validation query."""

    table_identifier: str
    select_items: Sequence[str]
    where_clauses: Sequence[str] = ()
    group_by_cols: Sequence[str] = ()
    order_by_cols: Sequence[str] = ()
    query_hints: str = ""
    prior_to_where: str = ""
    dialect: str = "oracle"


@runtime_checkable
class BigQueryDialectEngineProtocol(Protocol):
    """Protocol for generating BigQuery DDL, DML, and projection SQL strings."""

    def partition_by_clause(self, partition_column: "ColumnMetadataInterface") -> str:
        """Render the BigQuery PARTITION BY expression for a partition column."""

    def properties_clause(
        self,
        *,
        partition_expr_sql: str | None = None,
        cluster_cols: Sequence[str] | None = None,
        options: Mapping[str, Any] | None = None,
    ) -> str:
        """Render the BigQuery PARTITION BY / CLUSTER BY / OPTIONS clause."""

    def create_table(
        self,
        *,
        project: str | None,
        db_name: str,
        table_name: str,
        columns: Sequence["ColumnMetadataInterface"],
        partition_expr_sql: str | None = None,
        cluster_cols: Sequence[str] | None = None,
        options: Mapping[str, Any] | None = None,
        replace: bool = False,
    ) -> str:
        """Render a BigQuery CREATE [OR REPLACE] TABLE statement."""

    def create_external_table(
        self,
        *,
        project: str | None,
        db_name: str,
        table_name: str,
        columns: Sequence["ColumnMetadataInterface"],
        storage_format: str,
        uris: Sequence[str],
        extra_options: Mapping[str, Any] | None = None,
        replace: bool = True,
    ) -> str:
        """Render a BigQuery CREATE [OR REPLACE] EXTERNAL TABLE statement."""

    def ctas(
        self,
        *,
        project: str | None,
        db_name: str,
        table_name: str,
        from_project: str | None,
        from_db_name: str,
        from_table_name: str,
        select_expr_tuples: Sequence[tuple[str, str]] | None = None,
        options: Mapping[str, Any] | None = None,
        row_limit: int | None = None,
        replace: bool = False,
    ) -> str:
        """Render a BigQuery CREATE [OR REPLACE] TABLE ... AS SELECT statement."""

    def insert_select(
        self,
        *,
        project: str | None,
        db_name: str,
        table_name: str,
        insert_col_names: Sequence[str],
        select_expr_tuples: Sequence[tuple[str, str]],
        from_project: str | None = None,
        from_db_name: str | None = None,
        from_table_name: str | None = None,
        from_object_override: str | None = None,
        where_clauses: Sequence[str] | None = None,
    ) -> str:
        """Render a BigQuery INSERT INTO ... SELECT statement."""

    def alter_add_columns(
        self,
        *,
        project: str | None,
        db_name: str,
        table_name: str,
        columns: Sequence["ColumnMetadataInterface"],
    ) -> str:
        """Render a BigQuery ALTER TABLE ... ADD COLUMN statement."""

    def create_view(
        self,
        *,
        project: str | None,
        db_name: str,
        view_name: str,
        select_expr_tuples: Sequence[tuple[str, str]],
        from_tables: str,
        where_clauses: Sequence[str] | None = None,
        replace: bool = True,
    ) -> str:
        """Render a BigQuery CREATE [OR REPLACE] VIEW statement."""

    def select_count(
        self,
        *,
        project: str | None,
        db_name: str,
        table_name: str,
        filter_clause: str | None = None,
    ) -> str:
        """Render a BigQuery SELECT COUNT(*) statement."""

    def select_max(
        self,
        *,
        project: str | None,
        db_name: str,
        table_name: str,
        column_names: Sequence[str],
        column_cast_overrides: Mapping[str, str] | None = None,
        filter_clause: str | None = None,
    ) -> str:
        """Render a BigQuery SELECT MAX(...) statement."""

    def aggregate_select(
        self,
        *,
        project: str | None,
        db_name: str,
        table_name: str,
        select_expr_tuples: Sequence[tuple[str, str]],
        where_clauses: Sequence[str] | None = None,
        group_by_exprs: Sequence[str] | None = None,
        order_by_exprs: Sequence[str] | None = None,
    ) -> str:
        """Render a BigQuery aggregate SELECT statement."""


@runtime_checkable
class OracleExtractionQueryProtocol(Protocol):
    """Protocol for building and parsing Oracle extraction queries with AS OF SCN and PARTITION clauses."""

    def snapshot_scn_clause(self, transport_scn: int | None) -> str:
        """Render an Oracle AS OF SCN clause from an integer SCN."""

    def iot_partition_clause(self, partition_name: str | None) -> str:
        """Render an Oracle PARTITION("name") clause for an index-organized table."""

    def table_from_clause(
        self,
        *,
        owner: str,
        table_name: str,
        alias: str | None = "g",
        partition_names: Sequence[str] | None = None,
        is_subpartition: bool = False,
        transport_scn: int | None = None,
    ) -> str:
        """Render an Oracle table reference with optional PARTITION/SUBPARTITION and AS OF SCN clauses."""

    def id_range_query(
        self,
        *,
        owner: str,
        table_name: str,
        id_col_name: str,
        hint_block: str = "",
        iot_partition_name: str | None = None,
        transport_scn: int | None = None,
        where_clause: str = "",
    ) -> str:
        """Render an Oracle MIN/MAX ID range query."""

    def parse_and_render(self, sql: str) -> str:
        """Parse an Oracle query (including PARTITION/SUBPARTITION and AS OF SCN) and render back to SQL."""


@runtime_checkable
class PredicateDialectEngineProtocol(Protocol):
    """Protocol for rendering typed Lark predicate trees to dialect SQL and validating WHERE clauses."""

    def render_literal_sql(
        self,
        typed_tree: Any,
        columns: Sequence["ColumnMetadataInterface"],
    ) -> str:
        """Render a typed predicate tree into a SQL WHERE clause with inline literals."""

    def render_bind_sql(
        self,
        typed_tree: Any,
        columns: Sequence["ColumnMetadataInterface"],
    ) -> RenderedSQL:
        """Render a typed predicate tree into a SQL WHERE clause with named bind parameters."""

    def validate_where_clause(self, raw_sql: str, dialect: str) -> str:
        """Validate a raw SQL WHERE condition against an AST allowlist and return normalized SQL."""


@runtime_checkable
class ValidationQueryEngineProtocol(Protocol):
    """Protocol for validating cross-DB filter clauses and building aggregate validation SQL."""

    def validate_filter_clause(self, raw_filter: str, dialect: str) -> str:
        """Validate a raw filter condition string for the given dialect."""

    def build_aggregate_query(self, spec: ValidationQuerySpec) -> str:
        """Construct an aggregate validation SELECT query from a ValidationQuerySpec."""
```
