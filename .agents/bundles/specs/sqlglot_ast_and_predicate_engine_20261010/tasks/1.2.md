## Context

- **Custom Upstreamable Oracle Dialect Extension (`src/goe/offload/oracle/oracle_dialect.py`)**:
  - Stock `sqlglot.dialects.oracle.Oracle` does not enable `SUPPORTS_PARTITION_SELECTION` or `AS OF SCN` / `VERSIONS BETWEEN` out of the box.
  - Registering the extensions below on `Oracle`, `OracleParser`, and `OracleGenerator` gives 100% lossless AST round-trip parsing and generation for `SELECT /*+ FULL(s) */ a FROM "SH"."SALES" PARTITION ("P1") AS OF SCN 123456 s WHERE a > :b1`, `SUBPARTITION ("SP1") AS OF SCN :scn_bind s`, and `VERSIONS BETWEEN SCN 100 AND 200`:

```python
# SPDX-FileCopyrightText: 2026 The GOE Authors
# SPDX-License-Identifier: Apache-2.0

from typing import Final
from sqlglot import exp
from sqlglot.dialects.oracle import Oracle
from sqlglot.generators.oracle import OracleGenerator
from sqlglot.parsers.oracle import OracleParser
from sqlglot.tokens import TokenType
from sqlspec.builder._generation import invalidate_generator_dispatch

_ORACLE_EXTENSIONS_REGISTERED_ATTR: Final[str] = "_goe_oracle_extensions_registered"


def _oracle_version_sql(generator: OracleGenerator, expression: exp.Version) -> str:
    """Render Oracle Flashback Query AS OF {SCN | TIMESTAMP} and VERSIONS BETWEEN clauses."""
    this = expression.name.upper()
    kind = expression.text("kind").upper()
    expr_sql = generator.sql(expression, "expression")
    if this in {"SCN", "TIMESTAMP"} and kind == "AS OF":
        return f"AS OF {this} {expr_sql}"
    if this in {"SCN", "TIMESTAMP"} and kind == "BETWEEN":
        expr_node = expression.expression
        if isinstance(expr_node, exp.Tuple) and len(expr_node.expressions) == 2:
            start_sql = generator.sql(expr_node.expressions[0])
            end_sql = generator.sql(expr_node.expressions[1])
            return f"VERSIONS BETWEEN {this} {start_sql} AND {end_sql}"
    return f"FOR {this} {kind} {expr_sql}"


def register_oracle_dialect_extensions() -> None:
    """Register Oracle PARTITION/SUBPARTITION selection and AS OF SCN/TIMESTAMP flashback extensions."""
    if getattr(OracleParser, _ORACLE_EXTENSIONS_REGISTERED_ATTR, False):
        return
    Oracle.Tokenizer.KEYWORDS["SUBPARTITION"] = TokenType.PARTITION
    Oracle.ALIAS_POST_VERSION = True
    OracleParser.SUPPORTS_PARTITION_SELECTION = True
    OracleParser.VERSION_PHRASES = {
        **OracleParser.VERSION_PHRASES,
        ("AS", "OF", "SCN"): "SCN",
        ("AS", "OF", "TIMESTAMP"): "TIMESTAMP",
        ("VERSIONS", "SCN"): "SCN",
        ("VERSIONS", "TIMESTAMP"): "TIMESTAMP",
    }
    OracleGenerator.TRANSFORMS[exp.Version] = _oracle_version_sql
    invalidate_generator_dispatch(OracleGenerator)
    setattr(OracleParser, _ORACLE_EXTENSIONS_REGISTERED_ATTR, True)
```

- **Predicate Rendering (`src/goe/offload/predicate_offload.py`, `oracle_predicate.py`, `bigquery_predicate.py`)**:
  - Inside `PredicateSqlEngine` for `"oracle"`:
    - Bind mode emits `TO_DATE(:bind, 'YYYY-MM-DD HH24:MI:SS')` for `ORACLE_TYPE_DATE` and `TO_TIMESTAMP(:bind, 'YYYY-MM-DD HH24:MI:SS.FF9')` for `ORACLE_TYPE_TIMESTAMP*`.
    - Literal mode emits `TO_DATE('YYYY-MM-DD HH:MM:SS', 'YYYY-MM-DD HH24:MI:SS')` for `ORACLE_TYPE_DATE` and `TIMESTAMP 'YYYY-MM-DD HH:MM:SS.fffffffff'` (preserving up to 9 fractional digits from `np.datetime64`) for `ORACLE_TYPE_TIMESTAMP*`.
    - Never parse `TIMESTAMP '...'` through `sqlglot.parse_one`, which truncates nanoseconds to 6 digits (`FF6`). Construct `exp.Var(this=f"TIMESTAMP '{formatted_ts}'")` and `exp.Anonymous(this="TO_TIMESTAMP", expressions=[...])` directly.

```python
# SPDX-FileCopyrightText: 2026 The GOE Authors
# SPDX-License-Identifier: Apache-2.0

import sqlglot
from sqlglot import exp

from goe.offload.oracle.oracle_dialect import (
    OracleSqlEngine,
    get_oracle_dialect_engine,
    register_oracle_dialect_extensions,
)
from goe.offload.sql_engine import OracleExtractionQueryProtocol


def test_oracle_dialect_extensions_round_trip_partition_and_as_of_scn() -> None:
    """Verify PARTITION, SUBPARTITION, and AS OF SCN parse to exp.Partition/exp.Version and round-trip."""
    register_oracle_dialect_extensions()
    sql = 'SELECT /*+ FULL(s) */ a FROM "SH"."SALES" PARTITION ("P1") AS OF SCN 123456 s WHERE a > :b1'
    parsed = sqlglot.parse_one(sql, read="oracle")
    table_node = parsed.find(exp.Table)
    assert table_node is not None
    assert isinstance(table_node.args.get("partition"), exp.Partition)
    assert isinstance(table_node.args.get("version"), exp.Version)
    rendered = parsed.sql("oracle")
    assert rendered == 'SELECT /*+ FULL(s) */ a FROM "SH"."SALES" PARTITION("P1") AS OF SCN 123456 s WHERE a > :b1'

    subpart_sql = 'SELECT a FROM "SH"."SALES" SUBPARTITION ("SP1") AS OF SCN :scn_bind s'
    subpart_parsed = sqlglot.parse_one(subpart_sql, read="oracle")
    subpart_table = subpart_parsed.find(exp.Table)
    assert subpart_table is not None
    assert subpart_table.args["partition"].args.get("subpartition") is True
    assert subpart_parsed.sql("oracle") == 'SELECT a FROM "SH"."SALES" SUBPARTITION("SP1") AS OF SCN :scn_bind s'


def test_oracle_dialect_engine_satisfies_protocol_and_builds_table_clauses() -> None:
    """Verify OracleSqlEngine satisfies OracleExtractionQueryProtocol."""
    engine = get_oracle_dialect_engine()
    assert isinstance(engine, OracleExtractionQueryProtocol)
    assert isinstance(engine, OracleSqlEngine)

    from_clause = engine.table_from_clause(
        owner="SH",
        table_name="SALES",
        alias="g",
        partition_names=["P_2025"],
        is_subpartition=False,
        transport_scn=987654,
    )
    assert from_clause == '"SH"."SALES" PARTITION("P_2025") AS OF SCN 987654 g'
```
