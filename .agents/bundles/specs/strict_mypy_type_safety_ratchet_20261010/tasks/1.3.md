### Step 1: Write Configuration Enforcement Test in `tests/unit/test_mypy_strict_config.py`

Create `tests/unit/test_mypy_strict_config.py` asserting that `pyproject.toml` has zero `disable_error_code` suppressions and zero `goe.*` modules in `ignore_errors = true` overrides:

```python
# SPDX-FileCopyrightText: 2026 The GOE Authors
# SPDX-License-Identifier: Apache-2.0

"""Unit tests enforcing strict mypy configuration in pyproject.toml."""

from pathlib import Path
import tomllib


def test_pyproject_mypy_has_no_disabled_error_codes_or_ignored_goe_packages() -> None:
    """Verify pyproject.toml [tool.mypy] does not suppress error codes or ignore goe.* modules."""
    pyproject_path = Path(__file__).resolve().parents[2] / "pyproject.toml"
    data = tomllib.loads(pyproject_path.read_text(encoding="utf-8"))
    mypy_cfg = data["tool"]["mypy"]

    assert mypy_cfg.get("disable_error_code", []) == []
    assert mypy_cfg["check_untyped_defs"] is True
    assert mypy_cfg["disallow_incomplete_defs"] is True
    assert mypy_cfg["warn_redundant_casts"] is True
    assert mypy_cfg["warn_unused_configs"] is True

    for override in mypy_cfg.get("overrides", []):
        if override.get("ignore_errors") is True:
            modules = override.get("module", [])
            if isinstance(modules, str):
                modules = [modules]
            for mod in modules:
                assert not mod.startswith("goe."), f"Production module {mod} must not have ignore_errors=true"
```
