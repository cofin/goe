# Task 1.4: Cut every column call site over to ColumnSpec and the registry and delete the column subclasses

## Objective

1. Every module that imports `goe.offload.column_metadata` or a `*_column` module imports `goe.offload.schema.column_spec` (and `goe.offload.schema.type_registry` where it maps types) instead.
2. Every column constructor becomes `ColumnSpec(ColumnDialect.<X>, ...)`; every `ColumnPartitionInfo` becomes `ColumnPartitionSpec`; every column mutation becomes `replace(...)`.
3. Every mapping owner delegates `to_canonical_column`, `from_canonical_column`, `valid_canonical_override`, `gen_default_numeric_column`, and synthetic partition column creation to `TYPE_REGISTRY`; the old mapping bodies and their private helpers are deleted.
4. `src/goe/offload/column_metadata.py` and the eight `*_column.py` modules are deleted. The golden characterization (`tests/unit/offload/schema/*`) and `tests/unit/characterization/test_offload_bigquery_golden.py` stay green with unchanged expected values.

  ```python
  """Rewrite column class usage to ColumnSpec (scratch script, not committed)."""

  import re
  import sys
  from pathlib import Path

  CLASS_TO_DIALECT = {
      "CanonicalColumn": "CANONICAL",
      "OracleColumn": "ORACLE",
      "MSSQLColumn": "MSSQL",
      "BigQueryColumn": "BIGQUERY",
      "SnowflakeColumn": "SNOWFLAKE",
      "SynapseColumn": "SYNAPSE",
      "HadoopColumn": "HADOOP",
      "StagingParquetColumn": "PARQUET",
      "StagingAvroColumn": "AVRO",
  }
  RENAMES = {
      "ColumnMetadataInterface": "ColumnSpec",
      "ColumnPartitionInfo": "ColumnPartitionSpec",
      "ColumnMetadataException": "ColumnSpecError",
  }
  OLD_MODULES = (
      "goe.offload.column_metadata",
      "goe.offload.oracle.oracle_column",
      "goe.offload.microsoft.mssql_column",
      "goe.offload.bigquery.bigquery_column",
      "goe.offload.snowflake.snowflake_column",
      "goe.offload.microsoft.synapse_column",
      "goe.offload.hadoop.hadoop_column",
      "goe.offload.staging.parquet.parquet_column",
      "goe.offload.staging.avro.avro_column",
  )
  IMPORT_RE = re.compile(
      r"^(?P<indent>[ \t]*)from (?:"
      + "|".join(re.escape(m) for m in OLD_MODULES)
      + r") import (?P<names>\([^)]*\)|[^\n]+)",
      re.M,
  )
  CLASSES = "|".join(CLASS_TO_DIALECT)
  ISINSTANCE_RE = re.compile(r"isinstance\(\s*([\w.\[\]]+)\s*,\s*(" + CLASSES + r")\s*\)")
  CTOR_RE = re.compile(r"\b(" + CLASSES + r")\(")
  BARE_RE = re.compile(r"\b(" + CLASSES + r")\b")


  def rewrite_import(match: re.Match[str]) -> str:
      names = [n.strip() for n in match.group("names").strip("()").replace("\n", " ").split(",") if n.strip()]
      out: list[str] = []
      for name in names:
          if name in CLASS_TO_DIALECT:
              out.extend(["ColumnDialect", "ColumnSpec"])
          elif name == "ORACLE_SPLIT_HIGH_VALUE_RE":
              continue
          else:
              out.append(RENAMES.get(name, name))
      unique = sorted(set(out))
      return f"{match.group('indent')}from goe.offload.schema.column_spec import ({', '.join(unique)})"


  def rewrite(text: str) -> str:
      text = IMPORT_RE.sub(rewrite_import, text)
      text = ISINSTANCE_RE.sub(
          lambda m: (
              f"(isinstance({m.group(1)}, ColumnSpec) and {m.group(1)}.dialect is ColumnDialect.{CLASS_TO_DIALECT[m.group(2)]})"
          ),
          text,
      )
      text = CTOR_RE.sub(lambda m: f"ColumnSpec(ColumnDialect.{CLASS_TO_DIALECT[m.group(1)]}, ", text)
      for old, new in RENAMES.items():
          text = re.sub(rf"\b{old}\b", new, text)
      return BARE_RE.sub("ColumnSpec", text)


  for path in map(Path, sys.argv[1:]):
      path.write_text(rewrite(path.read_text()))
  ```
