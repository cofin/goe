# Task 1.1: Create `src/goe/util/gcp_credentials.py` and register GCP ADC/impersonation config options

## Objective
Create `src/goe/util/gcp_credentials.py` with `GcpCredentialBundle`, `resolve_gcp_credentials`, and `build_gcloud_impersonation_env` supporting standard Application Default Credentials (ADC), Workload Identity Federation, and `GOOGLE_IMPERSONATE_SERVICE_ACCOUNT`, and register the new configuration options (`google_cloud_project`, `google_cloud_quota_project`, `google_impersonate_service_account`, `google_dataproc_impersonate_service_account`) in `OrchestrationConfig`.

### Step 1: Write unit tests in `tests/unit/util/test_gcp_credentials.py`
Create `tests/unit/util/test_gcp_credentials.py` testing:
1. Direct ADC resolution (`google.auth.default`) with explicit `google_cloud_project` and `google_cloud_quota_project` precedence (`GOOGLE_CLOUD_PROJECT` -> `BIGQUERY_DATASET_PROJECT` -> ADC default project).
2. Wrapping source credentials in `google.auth.impersonated_credentials.Credentials` when `google_impersonate_service_account` or `GOOGLE_IMPERSONATE_SERVICE_ACCOUNT` is set.

"""Unit tests for unified GCP Application Default Credentials and impersonation resolution."""

from __future__ import annotations

from types import SimpleNamespace
from unittest.mock import MagicMock, patch

import pytest

from goe.exceptions import OffloadOptionError
from goe.util.gcp_credentials import (
    DEFAULT_GCP_SCOPES,
    GcpCredentialBundle,
    build_gcloud_impersonation_env,
    resolve_gcp_credentials,
)


def test_resolve_gcp_credentials_direct_adc(monkeypatch: pytest.MonkeyPatch) -> None:
    """Verify standard ADC resolution without impersonation."""
    monkeypatch.delenv("GOOGLE_IMPERSONATE_SERVICE_ACCOUNT", raising=False)
    monkeypatch.delenv("GOOGLE_CLOUD_PROJECT", raising=False)
    monkeypatch.delenv("GOOGLE_CLOUD_QUOTA_PROJECT", raising=False)
    fake_creds = MagicMock()
    fake_creds.service_account_email = "runner@test-proj.iam.gserviceaccount.com"
    config = SimpleNamespace(
        google_cloud_project="cfg-project",
        bigquery_dataset_project="bq-project",
        google_cloud_quota_project="quota-project",
        google_impersonate_service_account=None,
    )
    with patch("goe.util.gcp_credentials.google.auth.default", return_value=(fake_creds, "adc-project")) as mock_default:
        bundle = resolve_gcp_credentials(config)

    mock_default.assert_called_once_with(
        scopes=list(DEFAULT_GCP_SCOPES),
        quota_project_id="quota-project",
    )
    assert isinstance(bundle, GcpCredentialBundle)
    assert bundle.credentials is fake_creds
    assert bundle.project == "cfg-project"
    assert bundle.quota_project == "quota-project"
    assert bundle.impersonated_principal is None
    assert bundle.principal == "runner@test-proj.iam.gserviceaccount.com"


def test_resolve_gcp_credentials_with_impersonation(monkeypatch: pytest.MonkeyPatch) -> None:
    """Verify source ADC credentials are wrapped in impersonated_credentials.Credentials."""
    monkeypatch.delenv("GOOGLE_IMPERSONATE_SERVICE_ACCOUNT", raising=False)
    source_creds = MagicMock()
    impersonated_creds = MagicMock()
    config = SimpleNamespace(
        google_cloud_project="target-proj",
        bigquery_dataset_project=None,
        google_cloud_quota_project="quota-proj",
        google_impersonate_service_account="goe-sa@target-proj.iam.gserviceaccount.com",
    )
    with (
        patch("goe.util.gcp_credentials.google.auth.default", return_value=(source_creds, "adc-proj")),
        patch(
            "goe.util.gcp_credentials.impersonated_credentials.Credentials",
            return_value=impersonated_creds,
        ) as mock_impersonate,
    ):
        bundle = resolve_gcp_credentials(config)

    mock_impersonate.assert_called_once_with(
        source_credentials=source_creds,
        target_principal="goe-sa@target-proj.iam.gserviceaccount.com",
        target_scopes=list(DEFAULT_GCP_SCOPES),
        lifetime=3600,
        quota_project_id="quota-proj",
    )
    assert bundle.credentials is impersonated_creds
    assert bundle.impersonated_principal == "goe-sa@target-proj.iam.gserviceaccount.com"
    assert bundle.principal == "goe-sa@target-proj.iam.gserviceaccount.com"


def test_resolve_gcp_credentials_conflict_on_mismatched_impersonated_adc() -> None:
    """Verify an error is raised if ADC is already impersonated to a different service account."""
    source_creds = MagicMock()
    source_creds._target_principal = "other-sa@proj.iam.gserviceaccount.com"
    config = SimpleNamespace(
        google_cloud_project="proj",
        bigquery_dataset_project=None,
        google_cloud_quota_project=None,
        google_impersonate_service_account="goe-sa@proj.iam.gserviceaccount.com",
    )
    with (
        patch("goe.util.gcp_credentials.google.auth.default", return_value=(source_creds, "proj")),
        patch("goe.util.gcp_credentials._is_already_impersonated", return_value=True),
        pytest.raises(OffloadOptionError, match="already impersonating"),
    ):
        resolve_gcp_credentials(config)


def test_build_gcloud_impersonation_env() -> None:
    """Verify gcloud subprocess environment includes impersonation and project overrides."""
    config = SimpleNamespace(
        google_cloud_project="my-gcp-proj",
        google_dataproc_project="my-dp-proj",
        google_impersonate_service_account="base-sa@my-gcp-proj.iam.gserviceaccount.com",
        google_dataproc_impersonate_service_account="dp-caller@my-gcp-proj.iam.gserviceaccount.com",

```python
# SPDX-FileCopyrightText: 2026 The GOE Authors
# SPDX-License-Identifier: Apache-2.0

"""Unified Google Cloud Application Default Credentials (ADC) and service account impersonation resolver."""

from __future__ import annotations

from dataclasses import dataclass
import os
from typing import TYPE_CHECKING, Any

import google.auth
from google.auth import impersonated_credentials
from google.auth.credentials import Credentials

from goe.exceptions import OffloadOptionError

if TYPE_CHECKING:
    from collections.abc import Sequence

__all__ = [
    "DEFAULT_GCP_SCOPES",
    "GcpCredentialBundle",
    "build_gcloud_impersonation_env",
    "resolve_gcp_credentials",
]

DEFAULT_GCP_SCOPES: tuple[str, ...] = ("https://www.googleapis.com/auth/cloud-platform",)
_IMPERSONATION_LIFETIME_SECONDS = 3600


@dataclass(frozen=True)
class GcpCredentialBundle:
    """Resolved GCP credentials, project bindings, and principal metadata."""

    credentials: Credentials
    project: str | None
    quota_project: str | None
    impersonated_principal: str | None
    principal: str | None
    credential_type: str
    using_json_key_file: bool = False


def resolve_gcp_credentials(
    config: Any = None,
    scopes: Sequence[str] | None = None,
) -> GcpCredentialBundle:
    """Resolve Google Cloud credentials using ADC and optional service account impersonation.

    Resolution precedence for target project:
    1. config.google_cloud_project or GOOGLE_CLOUD_PROJECT environment variable
    2. config.bigquery_dataset_project or BIGQUERY_DATASET_PROJECT environment variable
    3. Default project returned by google.auth.default()
    """
    target_scopes = list(scopes) if scopes is not None else list(DEFAULT_GCP_SCOPES)
    quota_project = _resolve_quota_project(config)
    target_principal = _resolve_impersonate_principal(config)
    using_json_key = _detect_json_key_file()

    source_creds, adc_project = google.auth.default(
        scopes=target_scopes,
        quota_project_id=quota_project,
    )
    resolved_project = _resolve_project(config, adc_project)

    if target_principal:
        if _is_already_impersonated(source_creds):
            existing_target = getattr(source_creds, "_target_principal", None) or getattr(
                source_creds, "service_account_email", None
            )
            if existing_target and existing_target != target_principal:
                raise OffloadOptionError(
                    f"ADC credentials are already impersonating '{existing_target}', "
                    f"which conflicts with GOOGLE_IMPERSONATE_SERVICE_ACCOUNT='{target_principal}'."
                )
            active_creds = source_creds
        else:
            active_creds = impersonated_credentials.Credentials(
                source_credentials=source_creds,
                target_principal=target_principal,
                target_scopes=target_scopes,
                lifetime=_IMPERSONATION_LIFETIME_SECONDS,
                quota_project_id=quota_project,
            )
        principal = target_principal
        cred_type = "impersonated_service_account"
    else:
        active_creds = source_creds
        principal = _extract_principal(source_creds)
        cred_type = _classify_credentials(source_creds, using_json_key)

    return GcpCredentialBundle(
        credentials=active_creds,
        project=resolved_project,
        quota_project=quota_project,
        impersonated_principal=target_principal,
        principal=principal,
        credential_type=cred_type,
        using_json_key_file=using_json_key,
    )


def build_gcloud_impersonation_env(
    config: Any = None,
    base_env: dict[str, str] | None = None,
) -> dict[str, str]:
    """Build an environment dictionary for gcloud CLI subprocesses with impersonation and project overrides."""
    env = dict(os.environ if base_env is None else base_env)
    dataproc_impersonate = getattr(config, "google_dataproc_impersonate_service_account", None) or os.environ.get(
        "GOOGLE_DATAPROC_IMPERSONATE_SERVICE_ACCOUNT"
    )
    impersonate_sa = dataproc_impersonate or _resolve_impersonate_principal(config)
    if impersonate_sa:
        env["CLOUDSDK_AUTH_IMPERSONATE_SERVICE_ACCOUNT"] = impersonate_sa

    dataproc_project = getattr(config, "google_dataproc_project", None) or os.environ.get("GOOGLE_DATAPROC_PROJECT")
    cloud_project = dataproc_project or _resolve_project(config, None)
    if cloud_project:
        env["CLOUDSDK_CORE_PROJECT"] = cloud_project
    return env


def _resolve_project(config: Any, adc_project: str | None) -> str | None:
    """Resolve the active GCP project from config, environment, or ADC fallback."""
    cfg_project = getattr(config, "google_cloud_project", None) if config is not None else None
    bq_project = getattr(config, "bigquery_dataset_project", None) if config is not None else None
    return (
        cfg_project
        or os.environ.get("GOOGLE_CLOUD_PROJECT")
        or bq_project
        or os.environ.get("BIGQUERY_DATASET_PROJECT")
        or adc_project
    )


def _resolve_quota_project(config: Any) -> str | None:
    """Resolve the optional quota/billing project from config or environment."""
    cfg_quota = getattr(config, "google_cloud_quota_project", None) if config is not None else None
    return cfg_quota or os.environ.get("GOOGLE_CLOUD_QUOTA_PROJECT") or None


def _resolve_impersonate_principal(config: Any) -> str | None:
    """Resolve the target service account email for control-plane impersonation."""
    cfg_sa = getattr(config, "google_impersonate_service_account", None) if config is not None else None
    return cfg_sa or os.environ.get("GOOGLE_IMPERSONATE_SERVICE_ACCOUNT") or None


def _is_already_impersonated(creds: Credentials) -> bool:
    """Return True if the ADC credentials object is already an impersonated service account credential."""
    return isinstance(creds, impersonated_credentials.Credentials) or hasattr(creds, "_target_principal")


def _extract_principal(creds: Credentials) -> str | None:
    """Extract a human-readable principal identifier from a google-auth Credentials instance."""
    for attr in ("service_account_email", "_target_principal", "signer_email"):
        val = getattr(creds, attr, None)
        if isinstance(val, str) and val:
            return val
    return None


def _classify_credentials(creds: Credentials, using_json_key: bool) -> str:
    """Classify the credential type for diagnostics and preflight checks."""
    if using_json_key:
        return "service_account_key"
    cls_name = type(creds).__name__.lower()
    if "impersonated" in cls_name:
        return "impersonated_service_account"
    if "external" in cls_name or "identitypool" in cls_name or "aws" in cls_name:
        return "workload_identity_federation"
    if "user" in cls_name:
        return "authorized_user"
    return cls_name


def _detect_json_key_file() -> bool:
    """Detect whether GOOGLE_APPLICATION_CREDENTIALS points to a legacy service_account private key JSON file."""
    creds_path = os.environ.get("GOOGLE_APPLICATION_CREDENTIALS")
    if not creds_path or not os.path.isfile(creds_path):
        return False
    try:
        with open(creds_path, encoding="utf-8") as fh:
            head = fh.read(4096)
        return '"type"' in head and '"service_account"' in head and '"private_key"' in head
    except OSError:
        return False
```
