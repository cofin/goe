# Task 1.3: Replace OracleGateway with registry leases and named statements

## Objective

1. Proxy connections made by `OracleFrontendApi(conn_user_override=<schema>)` authenticate with the admin password (fixes `ORA-01017` when the admin and application passwords differ).
2. `OracleFrontendApi.get_distinct_column_values` encloses the partition name: `PARTITION ("P_2011 Q1")`.
3. `OracleFrontendApi` leases one pooled connection from `OracleDatabaseRegistry` per API object (admin pool by default, application pool for the application user override), borrows `existing_connection` unchanged, opens a dedicated proxy connection for any other override, and releases or closes it in `close()`.
4. No per-cursor `ALTER SESSION SET TRACEFILE_IDENTIFIER`; a lease idle for `LEASE_IDLE_PING_SECONDS` (60) is pinged before use and renewed after a lost-session error; named statements retry once after a lost session.
5. `src/goe/persistence/statements.py` (`NamedStatements`, `load_sql_file`, `validate_oracle_identifier`) replaces `OracleGateway`; frontend and repository named statements run through `registry.driver_for(...)` drivers that share the pool `StatementCacheHolder`; `oracle_repo.sql` uses `/* slot: */` markers.
6. `src/goe/persistence/oracle_gateway.py`, its tests, `attach_db_observability`, and `apply_oracle_session_trace_info` are deleted.

```python
# SPDX-FileCopyrightText: 2026 The GOE Authors
# SPDX-License-Identifier: Apache-2.0

"""Named SQL statements loaded with SQLSpec SQLFileLoader and filled with validated identifier slots."""

import re
from collections.abc import Mapping
from functools import cache
from pathlib import Path
from typing import Final

from sqlspec.loader import SQLFileLoader

__all__ = ("NamedStatements", "load_sql_file", "validate_oracle_identifier")

_IDENTIFIER_RE: Final = re.compile(r"[A-Za-z][A-Za-z0-9_$#]*")


def validate_oracle_identifier(identifier: str) -> str:
    """Return ``identifier`` upper-cased after enforcing Oracle unquoted-identifier rules.

    Surrounding double quotes are stripped first; any other character outside ``[A-Za-z0-9_$#]`` or a leading
    non-letter raises ``ValueError``, which blocks SQL injection through identifier slots.
    """
    candidate = identifier.strip()
    if len(candidate) >= 2 and candidate.startswith('"') and candidate.endswith('"'):
        candidate = candidate[1:-1]
    if not _IDENTIFIER_RE.fullmatch(candidate):
        msg = f"Invalid Oracle identifier: {identifier!r}"
        raise ValueError(msg)
    return candidate.upper()


@cache
def load_sql_file(path: Path) -> SQLFileLoader:
    """Return a process-wide cached ``SQLFileLoader`` holding the named statements in ``path``."""
    loader = SQLFileLoader()
    loader.load_sql(path)
    return loader


class NamedStatements:
    """Executable texts of the named statements in one SQL file, cached per slot values."""

    def __init__(self, path: Path, slots: Mapping[str, str] | None = None) -> None:
        """Load ``path`` and validate the identifier ``slots`` shared by every statement."""
        self._loader = load_sql_file(path)
        self._slots = {key: validate_oracle_identifier(value) for key, value in (slots or {}).items()}
        self._texts: dict[tuple[str, tuple[tuple[str, str], ...]], str] = {}

    def names(self) -> list[str]:
        """Return the statement names defined in the file."""
        return self._loader.list_queries()

    def text(self, name: str, slots: Mapping[str, str] | None = None) -> str:
        """Return statement ``name`` with slots filled and the trailing semicolon removed.

        Raises:
            ValueError: A slot value is not a valid Oracle identifier.
            SQLSlotError: A slot the statement declares has no value.
        """
        extra = tuple(sorted((key, validate_oracle_identifier(value)) for key, value in (slots or {}).items()))
        key = (name, extra)
        cached = self._texts.get(key)
        if cached is None:
            values = {**self._slots, **dict(extra)}
            declared = {slot.name for slot in self._loader.get_query_slots(name)}
            statement = self._loader.get_sql(name, **{k: v for k, v in values.items() if k in declared})
            cached = statement.raw_sql.strip().removesuffix(";").rstrip()
            self._texts[key] = cached
        return cached
```

```python
def _active_connection(self) -> Any:
    """Return the attached connection, pinging a lease idle for LEASE_IDLE_PING_SECONDS and renewing a lost one."""
    if self._db_conn is None:
        return None
    if self._lease_role is not None and time.monotonic() - self._last_used >= LEASE_IDLE_PING_SECONDS:
        run_with_session_retry(lambda: self._db_conn.ping(), self._renew_connection)
    self._last_used = time.monotonic()
    return self._db_conn


def _connect(self) -> None:
    """Attach the borrowed, proxied, or leased connection selected by the constructor arguments."""
    assert self._registry is not None
    override = (self._conn_user_override or "").upper()
    adm_user = self._upper_or_empty(self._connection_options.ora_adm_user)
    app_user = self._upper_or_empty(self._connection_options.ora_app_user)
    if self._existing_connection:
        self._db_conn = self._existing_connection
        self._db_conn.module = FRONTEND_TRACE_MODULE
        self._db_conn.action = self._trace_action
    elif override and override not in (adm_user, app_user):
        self._db_conn = self._registry.connect_proxy(self._conn_user_override, action=self._trace_action)
    else:
        self._lease_role = "app" if override and override == app_user else "adm"
        self._db_conn = self._registry.lease(self._lease_role, action=self._trace_action)
    self._db_conn.outputtypehandler = self._connection_output_type_handler
    self._last_used = time.monotonic()


def _disconnect(self, force: bool = False) -> None:
    """Release a lease, close a proxy connection, or close a borrowed connection only when forced."""
    self._debug("Disconnecting from DB")
    if self._existing_connection and not force:
        return
    if self._db_curs:
        self._close_cursor()
    connection, self._db_conn = self._db_conn, None
    lease_role, self._lease_role = self._lease_role, None
    if connection is not None:
        try:
            if lease_role is not None and self._registry is not None:
                connection.outputtypehandler = None
                self._registry.release(lease_role, connection)
            else:
                connection.close()
        except Exception as exc:
            self._log("Exception closing connection:\n%s" % str(exc), detail=VVERBOSE)
    self._existing_connection = None


def _open_cursor(self) -> None:
    """Open a fresh cursor on the active connection, or clear the cursor when unconnected."""
    connection = self._active_connection()
    self._db_curs = connection.cursor() if connection is not None else None


def _renew_connection(self, failure: BaseException | None = None) -> None:
    """Discard a lost lease or proxy connection and attach a fresh one."""
    self._log(f"Reconnecting to Oracle due to: {failure}", detail=VVERBOSE)
    logger.info(f"Reconnecting to Oracle due to: {failure}")
    connection, self._db_conn = self._db_conn, None
    lease_role, self._lease_role = self._lease_role, None
    if connection is not None and lease_role is not None and self._registry is not None:
        self._registry.release(lease_role, connection, discard=True)
    elif connection is not None and not self._existing_connection:
        try:
            connection.close()
        except Exception as exc:
            self._log("Exception closing connection:\n%s" % str(exc), detail=VVERBOSE)
    self._connect()


def _run_statement(self, statements, name, params, slots, schema_type):
    """Execute a named statement on the active connection through a pool-shared SQLSpec driver."""
    assert self._registry is not None
    driver = self._registry.driver_for(self._active_connection(), self._lease_role or "adm")
    text = statements.text(name, slots)
    binds = dict(params or {})
    with SQLCommenterContext.scope({"module": FRONTEND_TRACE_MODULE, "action": self._trace_action}):
        if schema_type is None:
            return driver.select(text, binds) if binds else driver.select(text)
        if binds:
            return driver.select(text, binds, schema_type=schema_type)
        return driver.select(text, schema_type=schema_type)


def _select_tuples(self, name: str, params: Mapping[str, Any] | None = None) -> list[tuple[Any, ...]] | None:
    """Return rows of a ``frontend_metadata.sql`` statement as tuples in projection order, or None unconnected."""
    if self._db_conn is None or self._registry is None:
        return None
    rows = run_with_session_retry(
        lambda: self._run_statement(self._metadata_statements, name, params, None, None), self._renew_connection
    )
    return [tuple(row.values()) for row in rows]
```
