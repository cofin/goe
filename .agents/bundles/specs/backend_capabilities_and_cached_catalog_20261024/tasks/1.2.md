  # SPDX-FileCopyrightText: 2026 The GOE Authors
  # SPDX-License-Identifier: Apache-2.0

  """Backend session parameter management, credential decryption, and log redaction."""

  from collections.abc import Callable, Mapping, Sequence
  import logging
  from typing import Any

  from goe.offload.offload_messages import VERBOSE, VVERBOSE
  from goe.typing import FrozenBaseStruct
  from goe.util.password_tools import PasswordTools

  __all__ = [
      "BackendSessionManager",
      "LogRedactionRule",
  ]

  logger = logging.getLogger(__name__)
  logger.addHandler(logging.NullHandler())


  class LogRedactionRule(FrozenBaseStruct):
      """Single sensitive-token replacement rule for backend SQL and command logs."""

      item: str
      sub: str = "?"


  class BackendSessionManager:
      """Manages session parameter merging, option validation, password decryption, and log redaction."""

      def __init__(
          self,
          connection_options: Any,
          backend_display_name: str,
          messages: Any,
          *,
          fixed_parameters: Mapping[str, Any] | None = None,
          do_not_connect: bool = False,
      ) -> None:
          self._connection_options = connection_options
          self._backend_display_name = backend_display_name
          self._messages = messages
          self._fixed_parameters = dict(fixed_parameters or {})
          user_params = getattr(connection_options, "backend_session_parameters", None)
          self._global_session_parameters = self.merge_session_parameters(
              user_params,
              self._fixed_parameters,
              log_clashes=not do_not_connect,
          )

      @property
      def global_session_parameters(self) -> dict[str, Any]:
          """Return the merged global session parameters dictionary."""
          return dict(self._global_session_parameters)

      def merge_session_parameters(
          self,
          user_parameters: Mapping[str, Any] | None,
          fixed_parameters: Mapping[str, Any] | None = None,
          *,
          log_clashes: bool = True,
      ) -> dict[str, Any]:
          """Merge user session parameters with fixed backend parameters, giving fixed keys precedence."""
          merged: dict[str, Any] = {}
          if user_parameters:
              merged.update(user_parameters)
          fixed = dict(fixed_parameters if fixed_parameters is not None else self._fixed_parameters)
          if fixed:
              fixed_lower = {k.lower() for k in fixed}
              for existing_key in [k for k in list(merged) if k.lower() in fixed_lower]:
                  if log_clashes:
                      self.log(
                          f"Removing OFFLOAD_BACKEND_SESSION_PARAMETERS[{existing_key}] "
                          "because of fixed parameter clash",
                          detail=VVERBOSE,
                      )
                  del merged[existing_key]
              merged.update(fixed)
          return merged

      def validate_connection_options(
          self,
          required_options: Sequence[str],
          exc_factory: Callable[[str], Exception],
      ) -> None:
          """Raise via exc_factory when any required attribute is missing on connection_options."""
          missing = [opt for opt in required_options if not hasattr(self._connection_options, opt)]
          if missing:
              raise exc_factory("Missing connection options: %s" % ", ".join(missing))

      def decrypt_password(self, password: str | None) -> str | None:
          """Decrypt an encrypted backend password when password_key_file is configured."""
          key_file = getattr(self._connection_options, "password_key_file", None)
          if key_file and password is not None:
              pass_tool = PasswordTools()
              goe_key = pass_tool.get_password_key_from_key_file(key_file)
              self.log(f"Decrypting {self._backend_display_name} password", detail=VVERBOSE)
              return pass_tool.b64decrypt(password, goe_key)
          return password

      @staticmethod
      def validate_no_log_items(
          no_log_items: Mapping[str, str] | Sequence[Mapping[str, str]] | None,
      ) -> list[ dict[str, str]] | None:
          """Validate and normalize no_log_items into a list of replacement dicts."""
          if no_log_items is None:
              return None
          if isinstance(no_log_items, Mapping):
              items = [dict(no_log_items)]
          elif isinstance(no_log_items, Sequence):
              items = [dict(entry) for entry in no_log_items]
          else:
              raise AssertionError(f"Invalid no_log_items type: {type(no_log_items)}")
          assert all(k in ("item", "sub") for k in items[0])
          return items

      @classmethod
      def redact_message(
          cls,
          msg: str,
          no_log_items: Mapping[str, str] | Sequence[Mapping[str, str]] | None = None,
      ) -> str:
          """Return msg with all sensitive items replaced by their substitution token."""
          validated = cls.validate_no_log_items(no_log_items)
          if not validated:
              return msg
          redacted = msg
          for entry in validated:
              redacted = redacted.replace(entry["item"], entry.get("sub", "?"))
          return redacted

      def log(
          self,
          msg: str,
          detail: int | None = None,
          ansi_code: str | None = None,
          no_log_items: Mapping[str, str] | Sequence[Mapping[str, str]] | None = None,
      ) -> None:
          """Write a redacted log line to OffloadMessages and standard Python logger."""
          log_msg = self.redact_message(msg, no_log_items=no_log_items)
          self._messages.log(log_msg, detail=detail, ansi_code=ansi_code)
          if detail == VVERBOSE:
              logger.debug(log_msg)
          else:
              logger.info(log_msg)

      def log_or_not(
          self,
          msg: str,
          log_level: int | None = VERBOSE,
          no_log_items: Mapping[str, str] | Sequence[Mapping[str, str]] | None = None,
      ) -> None:
          """Log msg only when log_level is not None."""
          if log_level is not None:
              self.log(msg, detail=log_level, no_log_items=no_log_items)
  ```

  # SPDX-FileCopyrightText: 2026 The GOE Authors
  # SPDX-License-Identifier: Apache-2.0

  """Composable backend DDL, projection, high-water-mark, and synthetic partition SQL compiler."""

  from collections.abc import Callable, Sequence
  from typing import Any

  from goe.offload.offload_constants import (
      PART_COL_DATE_GRANULARITIES,
  )
  from goe.offload.offload_messages import VVERBOSE
  from goe.offload.schema.column_spec import ColumnSpec

  __all__ = [
      "BackendDDLCompiler",
  ]


  class BackendDDLCompiler:
      """Compiles common backend DDL clauses, projection lists, HWM queries, and synthetic partition SQL."""

      def __init__(
          self,
          enclose_identifier: Callable[[str], str],
          enclose_object_reference: Callable[[str, str], str],
          *,
          not_null_column_supported: Callable[[], bool],
          max_decimal_precision: Callable[[], int],
          generic_string_data_type: Callable[[], str],
          regexp_extract_sql: Callable[[str, str], str] | None = None,
          regexp_extract_decimal_scale_pattern: Callable[[], str] | None = None,
          length_sql: Callable[[str], str] | None = None,
          log_fn: Callable[..., None] | None = None,
          debug_fn: Callable[[str], None] | None = None,
      ) -> None:
          self._enclose_identifier = enclose_identifier
          self._enclose_object_reference = enclose_object_reference
          self._not_null_supported = not_null_column_supported
          self._max_decimal_precision = max_decimal_precision
          self._generic_string_data_type = generic_string_data_type
          self._regexp_extract_sql = regexp_extract_sql
          self._regexp_extract_decimal_scale_pattern = regexp_extract_decimal_scale_pattern
          self._length_sql = length_sql
          self._log_fn = log_fn
          self._debug_fn = debug_fn

      def format_column_nn_clause(self, column: Any, *, external: bool = False) -> str:
          """Return NOT NULL when supported, non-external, and column.nullable is explicitly False."""
          if not self._not_null_supported() or column.nullable or column.nullable is None or external:
              return ""
          return "NOT NULL"

      def format_create_table_columns_clause(self, column_list: Sequence[Any], *, external: bool = False) -> str:
          """Format aligned column definitions for a CREATE TABLE statement."""
          sql_cols = [
              (
                  self._enclose_identifier(col.name),
                  col.format_data_type(),
                  self.format_column_nn_clause(col, external=external),
              )
              for col in column_list
          ]
          max_name = max(len(col[0]) for col in sql_cols)
          col_template = f"%-{max_name}s %s%s"
          return "    " + "\n,   ".join(
              [col_template % (col[0], col[1], f" {col[2]}" if col[2] else "") for col in sql_cols]
          )

      def format_select_projection(self, select_expr_tuples: Sequence[tuple[str, str]]) -> str:
          """Format a list of (expr, alias) tuples into an aligned SELECT projection."""
          return "\n,      ".join(f"{expr} AS {self._enclose_identifier(alias)}" for expr, alias in select_expr_tuples)

      def extract_date_part_sql_expression(self, date_part: str, column: str | Any) -> str:
          """Return an EXTRACT(DAY|MONTH|YEAR FROM col) expression."""
          assert isinstance(date_part, str)
          assert date_part.upper() in ("DAY", "MONTH", "YEAR")
          column_name = column if isinstance(column, str) else column.name
          return f"EXTRACT({date_part.upper()} FROM {self._enclose_identifier(column_name)})"

      def format_column_comparison(
          self,
          left_col: Any,
          operator: str,
          right_col: Any,
          left_alias: str | None = None,
          right_alias: str | None = None,
      ) -> str:
          """Format a qualified column-operator-column comparison expression."""
          left_ident = self._enclose_identifier(left_col.name)
          if left_alias:
              left_ident = f"{self._enclose_identifier(left_alias)}.{left_ident}"
          right_ident = self._enclose_identifier(right_col.name)
          if right_alias:
              right_ident = f"{self._enclose_identifier(right_alias)}.{right_ident}"
          return f"{left_ident} {operator} {right_ident}"

      def decimal_scale_validation_expression(self, column_name: str, column_scale: int) -> str:
          """Return a SQL predicate that evaluates True when a staged string exceeds column_scale."""
          assert self._regexp_extract_sql is not None
          assert self._regexp_extract_decimal_scale_pattern is not None
          assert self._length_sql is not None
          extract_scale_expr = self._regexp_extract_sql(column_name, self._regexp_extract_decimal_scale_pattern())
          length_expr = self._length_sql(extract_scale_expr)
          return f"{length_expr} > {column_scale}"

      def apply_decimal_padding_digits(
          self,
          column_name: str,
          data_precision: int | None,
          data_scale: int | None,
          decimal_padding_digits: int,
      ) -> tuple[int | None, int | None]:
          """Pad decimal scale and precision up to max_decimal_precision."""
          if not decimal_padding_digits:
              return data_precision, data_scale
          if data_precision is None or data_scale is None:
              if self._log_fn:
                  self._log_fn(
                      f"No padding applied to {column_name} for precision/scale: {data_precision}/{data_scale}",
                      detail=VVERBOSE,
                  )
              return data_precision, data_scale
          if self._debug_fn:
              self._debug_fn(f"Aligning precision/scale for {column_name}")
          max_prec = self._max_decimal_precision()
          new_precision = data_precision
          new_scale = data_scale
          if new_precision <= max_prec - decimal_padding_digits:
              new_scale += min(decimal_padding_digits, max_prec - new_precision)
              new_precision += min(decimal_padding_digits, max_prec - new_precision)
          if new_precision < max_prec:
              new_precision += min(decimal_padding_digits, max_prec - new_precision)
          if (new_precision != data_precision or new_scale != data_scale) and self._log_fn:
              self._log_fn(
                  f"Padded precision/scale for {column_name} in line with "
                  f"DECIMAL_PADDING_DIGITS={decimal_padding_digits}: "
                  f"{data_precision},{data_scale} -> {new_precision},{new_scale}",
                  detail=VVERBOSE,
              )
          return new_precision, new_scale

      def build_max_column_values_sql(
          self,
          db_name: str,
          table_name: str,
          column_name_list: Sequence[str],
          filter_clause: str | None = None,
          columns_to_cast_to_string: Sequence[str] | None = None,
      ) -> str:
          """Build a single-column MAX(col) or multi-column GROUP BY / ORDER BY DESC LIMIT 1 query."""
          assert column_name_list and isinstance(column_name_list, (list, tuple))
          if self._debug_fn:
              self._debug_fn(f"Generate max SQL for columns: {list(column_name_list)}")
          cast_targets = set(columns_to_cast_to_string or ())

          def _add_sql_cast(col_name: str, col_expr: str | None = None, add_alias: bool = False) -> str:
              expr = col_expr or col_name
              if col_name in cast_targets:
                  cast_str = f"CAST({expr} AS {self._generic_string_data_type()})"
              else:
                  cast_str = expr
              if add_alias:
                  cast_str += f" AS {col_name}"
              return cast_str

          if len(column_name_list) == 1:
              proj_col = _add_sql_cast(
                  column_name_list[0],
                  col_expr=f"MAX({self._enclose_identifier(column_name_list[0])})",
              )
              sql = f"SELECT {proj_col} FROM {self._enclose_object_reference(db_name, table_name)}"
              if filter_clause:
                  sql += " WHERE " + filter_clause
              return sql

          cols = ",".join(column_name_list)
          proj_cols = ",".join(_add_sql_cast(c, add_alias=True) for c in column_name_list)
          order_by = ",".join(f"{c} DESC" for c in column_name_list)
          where = f" WHERE {filter_clause}" if filter_clause else ""
          return (
              f"SELECT {proj_cols} FROM {self._enclose_object_reference(db_name, table_name)}"
              f"{where} GROUP BY {cols} ORDER BY {order_by} LIMIT 1"
          )

      def build_select_sql(
          self,
          db_name: str,
          table_name: str,
          column_names: Sequence[str] | None = None,
          filter_clauses: Sequence[str] | None = None,
          measures: Sequence[str] | None = None,
          agg_fns: Sequence[str] | None = None,
          *,
          exc_factory: Callable[[str], Exception] = ValueError,
      ) -> str:
          """Build a backend SELECT query with optional aggregate measures, WHERE, GROUP BY, and ORDER BY."""
          assert db_name and table_name
          projection: list[str] = []
          if column_names:
              projection.extend(self._enclose_identifier(c) for c in column_names)
          if measures and agg_fns:
              projection.extend(
                  f"{fn.upper()}({self._enclose_identifier(m)})" for m in measures for fn in agg_fns
              )
          if not projection:
              raise exc_factory(f"No columns have been specified for projection: {db_name}.{table_name}")
          projection_clause = "\n,      ".join(projection)
          where_clause = "\nWHERE  " + "\nAND    ".join(filter_clauses) if filter_clauses else ""
          group_by_clause = (
              "\nGROUP BY " + "\n,        ".join(self._enclose_identifier(c) for c in column_names)
              if measures and column_names
              else ""
          )
          order_by_clause = (
              "\nORDER BY " + "\n,        ".join(self._enclose_identifier(c) for c in column_names)
              if agg_fns and column_names
              else ""
          )
          return (
              f"SELECT {projection_clause}\n"
              f"FROM   {self._enclose_identifier(db_name)}.{self._enclose_identifier(table_name)}"
              f"{where_clause}{group_by_clause}{order_by_clause}"
          )

      def build_select_count_sql(
          self,
          db_name: str,
          table_name: str,
          filter_clause: str | None = None,
      ) -> str:
          """Build a SELECT COUNT(*) query for a backend table."""
          assert db_name and table_name
          where_clause = f" WHERE {filter_clause}" if filter_clause else ""
          return f"SELECT COUNT(*) FROM {self._enclose_object_reference(db_name, table_name)}{where_clause}"

      def build_insert_values_sql(
          self,
          db_name: str,
          table_name: str,
          column_names: Sequence[str],
          literal_csv_list: Sequence[str],
          *,
          split_by_cr: bool = True,
      ) -> str:
          """Build an INSERT INTO ... VALUES (...) statement."""
          join_str = "\n," if split_by_cr else ","
          cols_csv = ",".join(self._enclose_identifier(c) for c in column_names)
          insert_prefix = f"INSERT INTO {self._enclose_object_reference(db_name, table_name)} ({cols_csv}) VALUES"
          return insert_prefix + "\n " + join_str.join(f"({row})" for row in literal_csv_list)

      def build_synthetic_date_as_string_sql(
          self,
          date_value: str | None,
          granularity: str,
          source_column_cast: str,
          date_part_fn: Callable[[str, int, str], str],
      ) -> str | None:
          """Build a CONCAT(YEAR, '-', MONTH, '-', DAY) synthetic partition expression up to granularity."""
          assert granularity in PART_COL_DATE_GRANULARITIES, f"Unexpected granularity: {granularity}"
          assert source_column_cast
          if date_value is None:
              return None
          assert isinstance(date_value, str)
          components: list[tuple[str, int]] = [("YEAR", 4), ("MONTH", 2), ("DAY", 2)]
          parts: list[str] = []
          for comp_name, width in components:
              parts.append(date_part_fn(comp_name, width, source_column_cast))
              if granularity == comp_name[0]:
                  break
          return "CONCAT(%s)" % ",'-',".join(parts)

      def build_synthetic_sql_cast(
          self,
          partition_column: ColumnSpec | Any,
          source_column: ColumnSpec | Any,
          source_column_cast: str,
          *,
          partition_function_sql: Callable[[Any, str], str],
          date_truncated_sql: Callable[[str, str, str, str], str],
          date_as_string_sql: Callable[[str, str, str], str | None],
          string_granularity_sql: Callable[[str, int], str],
          number_granularity_sql: Callable[[str, Any, int, int | None], str],
      ) -> str:
          """Compile the backend SQL expression that populates a synthetic partition column."""
          part_info = partition_column.partition_info
          assert part_info is not None
          source_column_name = part_info.source_column_name
          digits = part_info.digits
          granularity = part_info.granularity
          if part_info.function:
              return partition_function_sql(part_info, source_column_cast)
          if source_column.is_date_based() and partition_column.is_date_based():
              return date_truncated_sql(
                  partition_column.name,
                  source_column_name,
                  granularity,
                  source_column_cast,
              )
          if source_column.is_date_based() and partition_column.is_string_based():
              expr = date_as_string_sql(source_column_name, granularity, source_column_cast)
              assert expr is not None
              return expr
          if source_column.is_string_based():
              return string_granularity_sql(source_column_name, int(granularity))
          if source_column.is_number_based():
              return number_granularity_sql(source_column_cast, source_column, int(granularity), digits)
          raise NotImplementedError(f"Unsupported synthetic partition column data type: {source_column.data_type}")
  ```

  class StagingLoadService:

      def build_final_table_casts(
          self,
          backend_columns: Sequence[Any],
          rdbms_columns: Sequence[Any],
          staging_columns: Sequence[Any],
          *,
          is_synthetic_partition_column: Callable[[Any], bool],
          get_column: Callable[[str], Any],
          staging_to_backend_cast: Callable[[Any, Any, Any], tuple[str, str | None, str | None]],
          gen_synthetic_sql_cast: Callable[[Any, str], str],
          log_fn: Callable[..., None] | None = None,
      ) -> dict[str, dict[str, str | None]]:
          """Build the per-column cast dictionary mapping staged columns to target table expressions."""
          if not rdbms_columns or not staging_columns:
              return {}
          final_table_casts: dict[str, dict[str, str | None]] = {}
          for backend_col in backend_columns:
              source_backend_col = backend_col
              if backend_col.partition_info and is_synthetic_partition_column(backend_col):
                  if log_fn:
                      log_fn(
                          f"Picking up source column for partition column {backend_col.name}: "
                          f"{backend_col.partition_info.source_column_name}",
                          detail=VVERBOSE,
                      )
                  source_backend_col = get_column(backend_col.partition_info.source_column_name)
              rdbms_col = match_table_column(source_backend_col.name, rdbms_columns)
              staging_col = match_table_column(source_backend_col.name, staging_columns)
              cast_expr, cast_type, vcast_expr = staging_to_backend_cast(rdbms_col, source_backend_col, staging_col)
              if backend_col.partition_info and is_synthetic_partition_column(backend_col):
                  partition_expr = gen_synthetic_sql_cast(backend_col, cast_expr)
                  partition_vexpr = gen_synthetic_sql_cast(backend_col, vcast_expr or cast_expr)
                  entry = CastPlanEntry(
                      column_name=backend_col.name.upper(),
                      cast_expr=partition_expr,
                      cast_type=backend_col.data_type,
                      verify_cast_expr=partition_vexpr,
                  )
              else:
                  entry = CastPlanEntry(
                      column_name=backend_col.name.upper(),
                      cast_expr=cast_expr,
                      cast_type=cast_type,
                      verify_cast_expr=vcast_expr,
                  )
              final_table_casts[entry.column_name] = entry.to_legacy_dict()
          return final_table_casts

      @staticmethod
      def build_cast_verification_sql(
          staging_object_name: str,
          projection_list: Sequence[str],
          predicate_or_list: Sequence[str],
          limit: int | None = None,
      ) -> str:
          """Format the SELECT query used to count or sample rows failing staging-to-final casts."""
          limit_clause = f"\nLIMIT {limit}" if limit else ""
          return (
              "SELECT " + "\n,      ".join(projection_list) + "\n"
              f"FROM   {staging_object_name}\n"
              "WHERE  " + "\nOR     ".join(predicate_or_list) + limit_clause
          )

      def validate_final_table_casts(
          self,
          cast_validation_cols: Sequence[tuple[str, str]],
          *,
          staging_object_name: str,
          staging_col_is_not_null_expr: Callable[[str], str],
          format_staging_col_name: Callable[[str], str],
          execute_query_fetch_one: Callable[..., Any],
          query_options: dict[str, Any] | None,
          log_fn: Callable[..., None],
          exc_factory: Callable[[str], Exception],
          log_profile: Any = None,
      ) -> None:
          """Verify that no non-null staged value turns NULL when cast to its target column type."""
          if not cast_validation_cols:
              return
          cast_check_preds = [
              f"({staging_col_is_not_null_expr(cname)} AND {ccast} IS NULL)"
              for cname, ccast in cast_validation_cols
          ]
          verify_sql = self.build_cast_verification_sql(staging_object_name, ["COUNT(*)"], cast_check_preds)
          verify_set = execute_query_fetch_one(
              verify_sql,
              query_options=query_options,
              log_level=VERBOSE,
              not_when_dry_running=True,
              profile=log_profile,
          )
          if verify_set and verify_set[0] > 0:
              cast_check_cols = [format_staging_col_name(cname) for cname, _ in cast_validation_cols]
              identify_proj = [f"MAX(CASE WHEN {pred} THEN {i} END)" for i, pred in enumerate(cast_check_preds)]
              identify_sql = self.build_cast_verification_sql(
                  staging_object_name,
                  identify_proj,
                  cast_check_preds,
              )
              identify_set = execute_query_fetch_one(identify_sql, query_options=query_options)
              failing_casts = [cast_check_preds[idx] for idx in identify_set if idx is not None]
              failing_cols = [cast_check_cols[idx] for idx in identify_set if idx is not None]
              log_fn(
                  f"CAST() of load data will cause data loss due to lack of precision "
                  f"in target data type in {verify_set[0]} rows",
                  ansi_code="red",
              )
              log_fn("Failing casts are:\n%s" % "\n".join(failing_casts), ansi_code="red")
              if failing_casts:
                  suggest_sql = self.build_cast_verification_sql(
                      staging_object_name,
                      failing_cols,
                      failing_casts,
                      limit=min(verify_set[0], 50),
                  )
                  log_fn(f"The SQL below will assist identification of problem data:\n{suggest_sql}")
              raise exc_factory(CAST_VALIDATION_EXCEPTION_TEXT)

      def build_staged_validation_rules(
          self,
          *,
          backend_columns: Sequence[Any],
          non_synthetic_columns: Sequence[Any],
          partition_columns: Sequence[Any],
          rdbms_part_cols: Sequence[Any],
          rdbms_columns: Sequence[Any],
          staging_columns: Sequence[Any],
          not_null_supported: bool,
          nan_supported: bool,
          allow_decimal_scale_rounding: bool,
          backend_display_name: str,
          format_staging_col_name: Callable[[Any], str],
          max_decimal_scale: Callable[[str], int],
          decimal_scale_validation_expr: Callable[[str, int], str],
          is_nan_sql_expr: Callable[[str], str],
          is_synthetic_partition_column: Callable[[str], bool],
          get_column: Callable[[str], Any],
          get_verification_cast: Callable[[str], str],
          gen_synthetic_part_number_granularity_sql_expr: Callable[..., str],
          length_sql_expr: Callable[[str], str],
          partition_key_out_of_range_message: Callable[[Any], str],
          debug_fn: Callable[[str], None] | None = None,
      ) -> list[StagedValidationRule]:
          """Assemble the list of StagedValidationRule predicates for staged table verification."""
          rules: list[StagedValidationRule] = []
          checked_cols: list[str] = []
          upper_fn = lambda s: s.upper()

          if not_null_supported:
              nn_cols = get_column_names([c for c in backend_columns if c.nullable is False], upper_fn)
              for col_name in nn_cols:
                  staging_col = match_table_column(col_name, staging_columns)
                  rules.append(
                      StagedValidationRule(
                          column_name=col_name,
                          expression=f"{format_staging_col_name(staging_col)} IS NULL",
                          fatal=True,
                          message=f"{DATA_VALIDATION_NOT_NULL_EXCEPTION_TEXT}: {col_name}",
                      )
                  )
              checked_cols.extend(nn_cols)

          if rdbms_part_cols and len(rdbms_part_cols) > 1:
              extra_rdbms = set(get_column_names(rdbms_part_cols, upper_fn)).difference(checked_cols)
              for col_name in extra_rdbms:
                  staging_col = match_table_column(col_name, staging_columns)
                  rules.append(
                      StagedValidationRule(
                          column_name=col_name,
                          expression=f"{format_staging_col_name(staging_col)} IS NULL",
                          fatal=False,
                          message=f"Source RDBMS partition key {col_name} has NULL values",
                      )
                  )
              checked_cols.extend(extra_rdbms)

          if partition_columns:
              source_names = [
                  c.partition_info.source_column_name.upper()
                  for c in partition_columns
                  if c.partition_info
              ]
              for col_name in set(source_names).difference(checked_cols):
                  staging_col = match_table_column(col_name, staging_columns)
                  rules.append(
                      StagedValidationRule(
                          column_name=col_name.upper(),
                          expression=f"{format_staging_col_name(staging_col)} IS NULL",
                          fatal=False,
                          message=f"{NULL_BACKEND_PARTITION_VALUE_WARNING_TEXT}: {col_name}",
                      )
                  )

          nan_cols = [c.name for c in rdbms_columns if c.is_nan_capable()]
          if nan_cols and nan_supported:
              for col_name in nan_cols:
                  staging_col = match_table_column(col_name, staging_columns)
                  rules.append(
                      StagedValidationRule(
                          column_name=col_name.upper(),
                          expression=is_nan_sql_expr(format_staging_col_name(staging_col)),
                          fatal=False,
                          message=(
                              f'Column "{col_name.upper()}" has NaN values, RDBMS and '
                              f"{backend_display_name} predicate comparisons may not be consistent"
                          ),
                      )
                  )

          if not allow_decimal_scale_rounding:
              for scale_col in [c for c in non_synthetic_columns if c.is_number_based()]:
                  staging_col = match_table_column(scale_col.name, staging_columns)
                  assert staging_col is not None
                  if not staging_col.is_string_based():
                      continue
                  rdbms_col = match_table_column(scale_col.name, rdbms_columns)
                  assert rdbms_col is not None
                  if rdbms_col.data_scale == 0:
                      if debug_fn:
                          debug_fn("RDBMS column has integral data so no need to check data")
                      continue
                  backend_scale = scale_col.data_scale
                  if backend_scale is None:
                      backend_scale = max_decimal_scale(scale_col.data_type)
                  if rdbms_col.data_scale and 0 < rdbms_col.data_scale < backend_scale:
                      if debug_fn:
                          debug_fn(
                              f"RDBMS column scale is safely below the backend so no need to check it: "
                              f"{rdbms_col.data_scale} < {backend_scale}"
                          )
                      continue
                  invalid_scale_expr = decimal_scale_validation_expr(
                      format_staging_col_name(staging_col),
                      backend_scale,
                  )
                  rules.append(
                      StagedValidationRule(
                          column_name=scale_col.name.upper(),
                          expression=invalid_scale_expr,
                          fatal=True,
                          message=(
                              f"Scale for column {scale_col.name.upper()} exceeds column maximum: "
                              f"{backend_scale}\n{DATA_VALIDATION_SCALE_EXCEPTION_TEXT}"
                          ),
                      )
                  )

          for part_col in partition_columns:
              if (
                  is_synthetic_partition_column(part_col.name)
                  and part_col.is_string_based()
                  and part_col.partition_info
                  and not part_col.partition_info.function
              ):
                  source_col = get_column(part_col.partition_info.source_column_name)
                  if source_col.is_number_based():
                      gran_expr = gen_synthetic_part_number_granularity_sql_expr(
                          get_verification_cast(source_col.name.upper()),
                          source_col,
                          part_col.partition_info.granularity,
                          part_col.partition_info.digits,
                          with_padding=False,
                      )
                      len_expr = length_sql_expr(gran_expr)
                      src_upper = part_col.partition_info.source_column_name.upper()
                      rules.append(
                          StagedValidationRule(
                              column_name=src_upper,
                              expression=f"{len_expr} > {part_col.partition_info.digits}",
                              fatal=True,
                              message=(
                                  f"Synthetic value for {src_upper} exceeds digits "
                                  f"{part_col.partition_info.digits}, re-offload with larger value "
                                  "for --partition-digits"
                              ),
                          )
                      )
              elif not is_synthetic_partition_column(part_col.name):
                  if (
                      part_col.partition_info
                      and part_col.partition_info.range_start is not None
                      and part_col.partition_info.range_end is not None
                  ):
                      vcast = get_verification_cast(part_col.name)
                      rules.append(
                          StagedValidationRule(
                              column_name=part_col.name.upper(),
                              expression=(
                                  f"{vcast} < {part_col.partition_info.range_start} "
                                  f"OR {vcast} > {part_col.partition_info.range_end}"
                              ),
                              fatal=False,
                              message=partition_key_out_of_range_message(part_col),
                          )
                      )

          return rules

      def validate_staged_data(
          self,
          rules: Sequence[StagedValidationRule],
          *,
          staging_object_name: str,
          expected_rows: int | None,
          execute_query_fetch_one: Callable[..., Any],
          query_options: dict[str, Any] | None,
          log_fn: Callable[..., None],
          warning_fn: Callable[..., None],
          exc_factory: Callable[[str], Exception],
          log_profile: Any = None,
      ) -> None:
          """Execute the combined staged data validation query and enforce row counts and fatal rules."""
          projection = [f"MAX(CASE WHEN {r.expression} THEN {i} END)" for i, r in enumerate(rules)] + ["COUNT(*)"]
          pred_messages = "\n,      ".join(projection)
          sql = f"SELECT {pred_messages}\nFROM   {staging_object_name}"
          validation_set = execute_query_fetch_one(
              sql,
              query_options=query_options,
              log_level=VERBOSE,
              not_when_dry_running=True,
              profile=log_profile,
          )
          if validation_set:
              row_values = list(validation_set)
              count_star = row_values.pop()
              if expected_rows is None:
                  log_fn(f"Load table row count: {count_star}", detail=VERBOSE)
              elif expected_rows == count_star:
                  log_fn(
                      f"Offload row count matches load table row count ({expected_rows} == {count_star})",
                      detail=VERBOSE,
                  )
              else:
                  log_fn(
                      f"Offload row count != load table row count ({expected_rows} != {count_star})",
                      detail=VVERBOSE,
                  )
                  raise exc_factory(
                      f"Offload row count does not match load table row count ({expected_rows} != {count_star})"
                  )
              for msg_code in row_values:
                  if msg_code is not None:
                      warning_fn(rules[msg_code].message, ansi_code="red")
              errors = [
                  rules[msg_code].message
                  for msg_code in row_values
                  if msg_code is not None and rules[msg_code].fatal
              ]
              if errors:
                  raise exc_factory("\n".join(errors))
          else:
              log_fn("Staged data validation returned an empty set", detail=VERBOSE)


  class FinalTableMerger:
      """Compiles final INSERT INTO ... SELECT and materialized join statements from staged data."""

      def build_subchunk_filter_clauses(
          self,
          *,
          requested_chunk_column: str | None,
          partition_columns: Sequence[Any],
          get_final_table_cast: Callable[[Any], str],
          enclose_identifier: Callable[[str], str],
          load_db_name: str,
          load_table_name: str,
          backend_type: str,
          dry_run: bool,
          execute_query_fetch_all: Callable[..., Any],
          log_fn: Callable[..., None],
          warning_fn: Callable[[str], None],
          debug_fn: Callable[[str], None] | None = None,
      ) -> list[str] | None:
          """Discover distinct partition subchunks when --offload-chunk-column is active."""
          if not requested_chunk_column:
              return None
          chunk_cols = [
              c
              for c in partition_columns
              if c.partition_info and c.partition_info.source_column_name.upper() == requested_chunk_column.upper()
          ]
          chunk_col = chunk_cols[0] if chunk_cols else None
          if not chunk_col:
              warning_fn(f"Unknown partition column, ignoring --offload-chunk-column: {requested_chunk_column}")
              return None
          subchunk_expr = get_final_table_cast(chunk_col)
          if debug_fn:
              debug_fn(f"Subchunk expression: {subchunk_expr}")
          order_by = "\nORDER BY COUNT(*) DESC" if backend_type == DBTYPE_IMPALA else ""
          subchunk_sql = (
              f"SELECT {subchunk_expr}, COUNT(*)\n"
              f"FROM {enclose_identifier(load_db_name)}.{enclose_identifier(load_table_name)}\n"
              f"GROUP BY {subchunk_expr}{order_by}"
          )
          log_fn(subchunk_sql, detail=VERBOSE)
          if not dry_run:
              subchunks = execute_query_fetch_all(subchunk_sql, log_level=VERBOSE, not_when_dry_running=True)
              log_fn(OFFLOAD_CHUNK_COLUMN_MESSAGE_PATTERN % len(subchunks), detail=VERBOSE)
          else:
              subchunks = [("?", -1)]
          clauses: list[str] = []
          for subchunk_id, _ in subchunks:
              formatted_id = f"'{subchunk_id}'" if chunk_col.data_type == HADOOP_TYPE_STRING else str(subchunk_id)
              clauses.append(f"{subchunk_expr} = {formatted_id}")
          return clauses

      def build_final_insert_plan(
          self,
          select_expr_tuples: Sequence[tuple[str, str]],
          *,
          threshold_clauses: Sequence[str] | None = None,
          sort_expr_list: Sequence[str] | None = None,
          for_materialized_join: bool = False,
          subchunk_clauses: Sequence[str] | None = None,
          query_options: dict[str, Any] | None = None,
          format_insert_sql: Callable[..., str],
          debug_fn: Callable[[str], None] | None = None,
      ) -> FinalInsertPlan:
          """Build the sequence of INSERT INTO ... SELECT SQL statements and session options."""
          base_thresholds = list(threshold_clauses or ())
          sqls: list[str] = []
          if subchunk_clauses:
              if debug_fn:
                  debug_fn(f"Subchunk clauses: {list(subchunk_clauses)}")
              for chunk_clause in subchunk_clauses:
                  sqls.append(
                      format_insert_sql(
                          list(select_expr_tuples),
                          base_thresholds + [chunk_clause],
                          sort_expr_list,
                          for_materialized_join=for_materialized_join,
                      )
                  )
          else:
              sqls.append(
                  format_insert_sql(
                      list(select_expr_tuples),
                      base_thresholds,
                      sort_expr_list,
                      for_materialized_join=for_materialized_join,
                  )
              )
          return FinalInsertPlan(sql_statements=tuple(sqls), query_options=query_options)

      def build_materialized_join_plan(
          self,
          select_expr_tuples: Sequence[tuple[str, str]],
          *,
          threshold_cols: Sequence[Any] | None = None,
          gte_threshold_vals: Sequence[Any] | None = None,
          lt_threshold_vals: Sequence[Any] | None = None,
          insert_predicates: Sequence[str] | None = None,
          sort_expr_list: Sequence[str] | None = None,
          ipa_predicate_type: str | None = None,
          to_backend_literal: Callable[..., str],
          enclose_identifier: Callable[[str], str],
          subchunk_clauses: Sequence[str] | None = None,
          query_options: dict[str, Any] | None = None,
          format_insert_sql: Callable[..., str],
          debug_fn: Callable[[str], None] | None = None,
      ) -> FinalInsertPlan:
          """Build the INSERT INTO ... SELECT plan for a materialized join pushdown."""
          if threshold_cols:
              assert valid_column_list(threshold_cols)
          if insert_predicates:
              threshold_clauses = list(insert_predicates)
          else:
              if debug_fn:
                  debug_fn(f"Incremental insert columns: {str_list_of_columns(threshold_cols)}")
                  debug_fn(f"Incremental insert lt: {lt_threshold_vals}")
                  debug_fn(f"Incremental insert gte: {gte_threshold_vals}")
              threshold_clauses = []
              if threshold_cols and lt_threshold_vals:
                  lt_literals = hvs_to_backend_sql_literals(
                      threshold_cols,
                      lt_threshold_vals,
                      ipa_predicate_type,
                      to_backend_literal,
                  )
                  lt_clause, _ = get_hybrid_threshold_clauses(
                      threshold_cols,
                      threshold_cols,
                      ipa_predicate_type,
                      lt_literals,
                      col_enclosure_fn=enclose_identifier,
                  )
                  threshold_clauses.append(lt_clause)
                  if gte_threshold_vals:
                      gte_literals = hvs_to_backend_sql_literals(
                          threshold_cols,
                          gte_threshold_vals,
                          ipa_predicate_type,
                          to_backend_literal,
                      )
                      _, gte_clause = get_hybrid_threshold_clauses(
                          threshold_cols,
                          threshold_cols,
                          ipa_predicate_type,
                          gte_literals,
                          col_enclosure_fn=enclose_identifier,
                      )
                      threshold_clauses.append(gte_clause)

          return self.build_final_insert_plan(
              select_expr_tuples,
              threshold_clauses=threshold_clauses,
              sort_expr_list=sort_expr_list,
              for_materialized_join=True,
              subchunk_clauses=subchunk_clauses,
              query_options=query_options,
              format_insert_sql=format_insert_sql,
              debug_fn=debug_fn,
          )
