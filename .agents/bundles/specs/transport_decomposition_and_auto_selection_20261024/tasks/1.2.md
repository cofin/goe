# SPDX-FileCopyrightText: 2026 The GOE Authors

"""Spark transport polling, split planning, command execution, and safe PySpark codegen."""

from collections.abc import Callable
from queue import Empty, Queue
import threading
import time
from typing import Any

from goe.typing import FrozenBaseStruct
from goe.util.serializers import encode_json


class PySparkJobSpec(FrozenBaseStruct):
    """Immutable configuration payload serialized into generated PySpark scripts."""

    rdbms_source_query: str
    jdbc_url: str
    jdbc_username: str
    target_path_uri: str
    staging_format: str
    compression_codec: str | None = None
    num_partitions: int = 1
    fetch_size: int = 10_000
    extra_options: dict[str, str] = {}


class PySparkCodegen:
    """Generates valid Python source for PySpark JDBC extraction jobs using JSON literals."""

    @staticmethod
    def render_pyspark_body(spec: PySparkJobSpec) -> str:
        """Render a self-contained PySpark script body safe against embedded quotes in SQL."""
        payload_json = encode_json(spec)
        payload_literal = repr(payload_json)
        return (
            "import json\n"
            "from pyspark.sql import SparkSession\n\n"
            f"SPEC = json.loads({payload_literal})\n"
            "spark = SparkSession.builder.getOrCreate()\n"
            "reader = (\n"
            "    spark.read.format('jdbc')\n"
            "    .option('url', SPEC['jdbc_url'])\n"
            "    .option('dbtable', SPEC['rdbms_source_query'])\n"
            "    .option('user', SPEC['jdbc_username'])\n"
            "    .option('fetchsize', str(SPEC['fetch_size']))\n"
            "    .option('numPartitions', str(SPEC['num_partitions']))\n"
            ")\n"
            "for key, val in SPEC.get('extra_options', {}).items():\n"
            "    reader = reader.option(key, val)\n"
            "df = reader.load()\n"
            "writer = df.write.mode('overwrite').format(SPEC['staging_format'].lower())\n"
            "if SPEC.get('compression_codec'):\n"
            "    writer = writer.option('compression', SPEC['compression_codec'].lower())\n"
            "writer.save(SPEC['target_path_uri'])\n"
        )


class PollingThread(threading.Thread):
    """Daemon background thread that invokes a target callable at a fixed interval."""

    def __init__(
        self,
        target: Callable[..., Any] | None = None,
        name: str | None = None,
        args: tuple[Any, ...] = (),
        kwargs: dict[str, Any] | None = None,
        interval: float = 1.0,
    ) -> None:
        super().__init__(name=name, daemon=True)
        self._target_fn = target
        self._args = args
        self._kwargs = kwargs or {}
        self._interval = interval
        self._stopped = threading.Event()
        self._exc_queue: Queue[BaseException] = Queue()

    def run(self) -> None:
        """Execute target callable repeatedly until stop() is signaled."""
        while not self._stopped.is_set():
            try:
                if self._target_fn is not None:
                    self._target_fn(*self._args, **self._kwargs)
            except BaseException as exc:
                self._exc_queue.put(exc)
            self._stopped.wait(self._interval)

    def stop(self) -> None:
        """Signal the polling loop to exit."""
        self._stopped.set()

    def get_exceptions(self) -> list[BaseException]:
        """Drain and return any exceptions captured during polling iterations."""
        exceptions: list[BaseException] = []
        while not self._exc_queue.empty():
            try:
                exceptions.append(self._exc_queue.get(block=False))
            except Empty:
                break
        return exceptions

    def DrainExceptions(self) -> list[BaseException]:
        """Compatibility alias for get_exceptions."""
        return self.get_exceptions()
