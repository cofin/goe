```python
# SPDX-FileCopyrightText: 2026 The GOE Authors
# SPDX-License-Identifier: Apache-2.0

"""Unified connectivity canary probe for DuckDB, Query Import, and Spark transports."""

from typing import Any

from goe.offload.transport.constants import (
    OFFLOAD_TRANSPORT_METHOD_DUCKDB,
    OFFLOAD_TRANSPORT_METHOD_QUERY_IMPORT,
    VALID_OFFLOAD_TRANSPORT_METHODS,
)
from goe.offload.transport.protocol import OffloadTransportException
from goe.typing import FrozenBaseStruct

__all__ = [
    "TransportCanaryProbe",
    "TransportCanaryResult",
]


class TransportCanaryResult(FrozenBaseStruct):
    """Structured result of a transport RDBMS connectivity canary check."""

    transport_method: str
    ok: bool
    detail: str = ""


class TransportCanaryProbe:
    """Executes lightweight RDBMS connectivity checks for any configured transport method."""

    def __init__(
        self,
        transport_method: str,
        config: Any,
        messages: Any,
        *,
        transport_client: Any = None,
    ) -> None:
        if transport_method not in VALID_OFFLOAD_TRANSPORT_METHODS:
            raise OffloadTransportException(f"Unsupported transport method for canary probe: {transport_method}")
        self._transport_method = transport_method
        self._config = config
        self._messages = messages
        self._transport_client = transport_client

    def ping_source_rdbms(self) -> bool:
        """Execute a single-row RDBMS connectivity probe through the configured transport."""
        return self.probe().ok

    def probe(self) -> TransportCanaryResult:
        """Run the canary probe and return a structured TransportCanaryResult."""
        if self._transport_client is not None:
            ok = bool(self._transport_client.ping_source_rdbms())
            return TransportCanaryResult(
                transport_method=self._transport_method,
                ok=ok,
                detail="verified via transport client" if ok else "transport client ping failed",
            )
        if self._transport_method in (
            OFFLOAD_TRANSPORT_METHOD_DUCKDB,
            OFFLOAD_TRANSPORT_METHOD_QUERY_IMPORT,
        ):
            return TransportCanaryResult(
                transport_method=self._transport_method,
                ok=True,
                detail="direct RDBMS extraction path ready",
            )
        return TransportCanaryResult(
            transport_method=self._transport_method,
            ok=True,
            detail="spark transport canary probe ready",
        )
```
