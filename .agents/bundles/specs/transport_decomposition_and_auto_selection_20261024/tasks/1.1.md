- [ ] 3. Create `src/goe/offload/transport/protocol.py` defining `OffloadTransportException`, `TransportBatchSplit`, `TransportPlan`, `TransportExecutionSummary`, and `OffloadTransportProtocol`:

```python
# SPDX-FileCopyrightText: 2026 The GOE Authors
# SPDX-License-Identifier: Apache-2.0

"""Transport execution protocol and immutable plan/summary value objects."""

from typing import Any, Protocol, runtime_checkable

from goe.exceptions import GOEError
from goe.typing import FrozenBaseStruct

__all__ = [
    "OffloadTransportException",
    "OffloadTransportProtocol",
    "TransportBatchSplit",
    "TransportExecutionSummary",
    "TransportPlan",
]


class OffloadTransportException(GOEError):
    """Raised when transport planning, selection, or execution fails."""


class TransportBatchSplit(FrozenBaseStruct, array_like=True):
    """Immutable descriptor for one parallel extraction split."""

    split_index: int
    split_type: str
    sql_text: str
    bind_params: dict[str, Any]
    target_uri: str


class TransportPlan(FrozenBaseStruct):
    """Immutable transport execution plan produced before data movement."""

    transport_method: str
    source_owner: str
    source_table: str
    staging_format: str
    staging_prefix_uri: str
    parallelism: int
    consistent_read_scn: int | None = None
    estimated_bytes: int | None = None
    compression_codec: str | None = None
    splits: tuple[TransportBatchSplit, ...] = ()


class TransportExecutionSummary(FrozenBaseStruct):
    """Immutable transport telemetry returned after staging files are written."""

    transport_method: str
    rows_imported: int
    bytes_staged: int
    staged_file_uris: tuple[str, ...]
    elapsed_seconds: float
    consistent_read_scn: int | None = None


@runtime_checkable
class OffloadTransportProtocol(Protocol):
    """Runtime protocol implemented by all GOE offload transport engines."""

    def transport(self, partition_chunk: Any = None) -> int | None:
        """Execute data extraction and staging, returning imported row count."""

    def ping_source_rdbms(self) -> bool:
        """Verify connectivity from the transport runtime to the source database."""
```
