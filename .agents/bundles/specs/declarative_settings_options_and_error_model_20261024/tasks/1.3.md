```python
# SPDX-FileCopyrightText: 2026 The GOE Authors
# SPDX-License-Identifier: Apache-2.0

"""Typed operational option structs for GOE CLI and Listener requests."""

from __future__ import annotations

from collections.abc import Mapping
from typing import Literal

import msgspec

from goe.exceptions import ValidationError
from goe.typing import FrozenBaseStruct

__all__ = (
    "AggValidateOptions",
    "ConnectOptions",
    "OffloadOptions",
    "SchemaSyncOptions",
    "StatusReportOptions",
    "TestModeFaults",
)


class TestModeFaults(FrozenBaseStruct):
    """Typed fault-injection configuration for test-mode execution."""

    fault_point: Literal[
        "none",
        "before_transport",
        "during_transport",
        "before_backend_load",
        "before_metadata_commit",
        "before_hybrid_view",
    ] = "none"
    message: str = "Injected test-mode fault"

    @classmethod
    def from_token(cls, token: str | None) -> TestModeFaults:
        """Parse a CLI or test token into a validated TestModeFaults struct."""
        if not token:
            return cls()
        parts = token.split(":", maxsplit=1)
        point = parts[0].strip()
        msg = parts[1].strip() if len(parts) > 1 else "Injected test-mode fault"
        try:
            return msgspec.convert({"fault_point": point, "message": msg}, type=cls)
        except msgspec.ValidationError as exc:
            raise ValidationError(f"Invalid test fault token '{token}': {exc}") from exc


class OffloadOptions(FrozenBaseStruct):
    """Validated, immutable options for an offload or present operation."""

    owner_table: str
    execute: bool = False
    verbose: bool = False
    quiet: bool = False
    ansi: bool = False
    offload_type: Literal["FULL", "INCREMENTAL"] | None = None
    offload_transport_method: Literal[
        "AUTO",
        "DUCKDB",
        "QUERY_IMPORT",
        "SPARK_SUBMIT",
        "SPARK_THRIFTSERVER",
        "SPARK_LIVY",
        "DATAPROC",
    ] = "AUTO"
    offload_transport_parallelism: int = 4
    older_than_date: str | None = None
    older_than_days: int | None = None
    less_than_value: str | None = None
    equal_to_values: tuple[str, ...] = ()
    offload_predicate: str | None = None
    offload_predicate_type: str | None = None
    offload_chunk_column: str | None = None
    offload_chunk_size: int | None = None
    reset_backend_table: bool = False
    reset_hybrid_view: bool = False
    preserve_load_table: bool = False
    create_backend_db: bool = False
    verify_type: Literal["NONE", "ROWCOUNT", "AGGREGATE"] = "ROWCOUNT"
    decimal_columns_csv: tuple[str, ...] = ()
    decimal_padding_digits: tuple[int, int] | None = None
    date_columns_csv: tuple[str, ...] = ()
    timestamp_tz_columns_csv: tuple[str, ...] = ()
    synthetic_partition_column: str | None = None
    synthetic_partition_function: str | None = None
    synthetic_partition_digits: int | None = None
    bucket_hash_column: str | None = None
    sort_columns_csv: tuple[str, ...] = ()
    test_faults: TestModeFaults = TestModeFaults()

    @property
    def owner(self) -> str:
        """Return the normalized uppercase schema/owner name."""
        return self.owner_table.split(".", maxsplit=1)[0].upper()

    @property
    def table_name(self) -> str:
        """Return the normalized uppercase table name."""
        parts = self.owner_table.split(".", maxsplit=1)
        if len(parts) != 2 or not parts[0] or not parts[1]:
            raise ValidationError(f"Expected OWNER.TABLE format, got '{self.owner_table}'")
        return parts[1].upper()

    @classmethod
    def from_mapping(cls, data: Mapping[str, object]) -> OffloadOptions:
        """Validate a mapping of CLI or HTTP request options into OffloadOptions."""
        normalized = _normalize_offload_mapping(data)
        try:
            instance = msgspec.convert(normalized, type=cls, strict=False)
        except msgspec.ValidationError as exc:
            raise ValidationError(str(exc)) from exc
        _validate_offload_invariants(instance)
        return instance
```
