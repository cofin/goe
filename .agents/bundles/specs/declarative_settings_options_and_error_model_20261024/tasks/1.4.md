```python
# SPDX-FileCopyrightText: 2026 The GOE Authors
# SPDX-License-Identifier: Apache-2.0

"""Composable Click option-group decorators generated from OptionSpec."""

from __future__ import annotations

from collections.abc import Callable
from typing import TypeVar, cast

import rich_click as click

from goe.config.option_registry import OptionGroupName, OptionSpec, get_option_specs_for_group

__all__ = (
    "column_transform_options",
    "common_options",
    "partition_options",
    "table_target_options",
    "transport_options",
)

F = TypeVar("F", bound=Callable[..., object])


def common_options(func: F) -> F:
    """Attach common execution, logging, and OpenTelemetry CLI options."""
    return _apply_group_options(func, "common")


def table_target_options(func: F) -> F:
    """Attach owner.table target and schema reset CLI options."""
    return _apply_group_options(func, "table_target")


def partition_options(func: F) -> F:
    """Attach partition boundary, IPA, and predicate CLI options."""
    return _apply_group_options(func, "partition")


def transport_options(func: F) -> F:
    """Attach transport method, parallelism, and chunking CLI options."""
    return _apply_group_options(func, "transport")


def column_transform_options(func: F) -> F:
    """Attach column casting, synthetic partition, bucketing, and sorting CLI options."""
    return _apply_group_options(func, "column_transform")


def _apply_group_options(func: F, group: OptionGroupName) -> F:
    """Apply all OptionSpec entries for a named option group in reverse order."""
    decorated: Callable[..., object] = func
    for spec in reversed(get_option_specs_for_group(group)):
        decorated = _click_option_from_spec(spec)(decorated)
    return cast(F, decorated)


def _click_option_from_spec(spec: OptionSpec) -> Callable[[F], F]:
    """Build a rich_click.option decorator from a declarative OptionSpec."""
    option_type: object = str
    if spec.choices is not None:
        option_type = click.Choice(list(spec.choices), case_sensitive=False)
    elif spec.value_type == "int":
        option_type = int
    elif spec.value_type == "float":
        option_type = float
    elif spec.value_type == "bool":
        option_type = bool
    return click.option(
        *spec.flags,
        default=spec.default,
        required=spec.required,
        is_flag=spec.is_flag,
        type=option_type,
        help=spec.help,
        hidden=spec.hidden,
    )
```
