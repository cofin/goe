# Task 1.1: Build AggregateQuestion and CrossDbValidationBuilder with 64-aggregate batch chunking

## Objective

1. Create `src/goe/offload/validation/__init__.py` and `src/goe/offload/validation/ast_builder.py` defining `AggregateFunc`, `AggregateQuestion(FrozenBaseStruct)`, `ValidationBatchQuery(FrozenBaseStruct)`, and `CrossDbValidationBuilder`.
2. Build dialect-targeted aggregate `SELECT` queries (`COUNT(*)`, `MIN`, `MAX`, `SUM`, `AVG`, `STDDEV`, and sample-hash queries) via `DialectSqlEngine` and `SqlGlotPredicateCompiler`, chunking column aggregates into batches of at most `64` aggregate expressions per query.
3. Emit Oracle `/*+ NO_INDEX(t) PARALLEL(t, n) */` hints, `AS OF SCN :scn` snapshot clauses with bound `:scn` parameters, and explicit `ORDER BY <col> ASC NULLS LAST` on every `GROUP BY` column.
4. Provide `CrossDbValidationBuilder.parse_filter_clause` to validate and transpile raw filter expressions without regex splitting or `.replace('"', '`')` string corruption.

## Context

- Current `src/goe/offload/offload_validation.py:103-145` builds aggregate queries via `ValidationSqlEngine` using `exp.Var(this=item)` string concatenation and calls `register_oracle_dialect_extensions()` (`:39, :109, :118`).
- Current `src/goe/scripts/agg_validate.py:46, 109-121` parses `--filters` using `REGEX_FILTER` and rewrites identifiers by replacing double quotes with backticks, corrupting string literals that contain double quotes.
- Chapter 7.1 delivers `DialectSqlEngine` and `TableRef` in `src/goe/offload/sql_ast/dialect_engine.py`, and Chapter 7.2 delivers `PredicateExpr`, `parse_predicate_dsl`, and `SqlGlotPredicateCompiler` in `src/goe/offload/sql_ast/predicate_compiler.py`.
- Architecture boundary (`tools/verify_architecture.py` from Chapter 5.2): if `src/goe/offload/validation/ast_builder.py` needs `sqlglot` AST construction, it either delegates through `goe.offload.sql_ast.dialect_engine` or `tools/verify_architecture.py` permits `goe.offload.sql_ast` to provide the AST helper. Specifically, `CrossDbValidationBuilder` calls `DialectSqlEngine` and `SqlGlotPredicateCompiler` from `goe.offload.sql_ast` so all `sqlglot` imports stay inside `goe.offload.sql_ast`, or uses `sqlspec.sql` / `DialectSqlEngine` helpers without violating the `sqlglot` import rule.

  ```python
  # SPDX-FileCopyrightText: 2026 The GOE Authors
  # SPDX-License-Identifier: Apache-2.0

  """Validation aggregate question models and dialect-aware query batch builder."""

  from collections.abc import Sequence
  from typing import Any, Literal

  from goe.exceptions import GOEError
  from goe.offload.sql_ast.dialect_engine import SqlDialect, TableRef, get_dialect_engine
  from goe.offload.sql_ast.predicate_compiler import PredicateExpr, SqlGlotPredicateCompiler
  from goe.typing import FrozenBaseStruct

  __all__ = (
      "MAX_AGGREGATES_PER_QUERY",
      "AggregateFunc",
      "AggregateQuestion",
      "CrossDbValidationBuilder",
      "ValidationBatchQuery",
  )

  type AggregateFunc = Literal["COUNT", "MIN", "MAX", "SUM", "AVG", "STDDEV", "HASH"]

  MAX_AGGREGATES_PER_QUERY: int = 64


  class AggregateQuestion(FrozenBaseStruct):
      """Single aggregate projection question in a cross-database validation batch."""

      column_name: str | None
      func: AggregateFunc
      alias: str
      is_approximate: bool = False


  class ValidationBatchQuery(FrozenBaseStruct):
      """Compiled frontend or backend validation query batch with bind parameters."""

      dialect: SqlDialect
      sql: str
      bind_params: dict[str, Any]
      group_by_columns: tuple[str, ...]
      questions: tuple[AggregateQuestion, ...]


  class CrossDbValidationBuilder:
      """Builds chunked aggregate and sample-hash validation queries via DialectSqlEngine."""

      def __init__(
          self,
          *,
          max_aggregates_per_query: int = MAX_AGGREGATES_PER_QUERY,
          predicate_compiler: SqlGlotPredicateCompiler | None = None,
      ) -> None:
          if max_aggregates_per_query < 1:
              raise GOEError("max_aggregates_per_query must be at least 1")
          self._max_aggregates = max_aggregates_per_query
          self._predicate_compiler = predicate_compiler or SqlGlotPredicateCompiler()

      @property
      def max_aggregates_per_query(self) -> int:
          """Return the maximum number of aggregate projections per SQL query."""
          return self._max_aggregates

      def build_questions(
          self,
          *,
          columns: Sequence[tuple[str, bool]],
          aggs: Sequence[AggregateFunc] = ("MIN", "MAX", "COUNT"),
          include_row_count: bool = True,
      ) -> tuple[AggregateQuestion, ...]:
          """Construct ordered AggregateQuestion items for the requested columns and functions."""
          questions: list[AggregateQuestion] = []
          if include_row_count:
              questions.append(
                  AggregateQuestion(
                      column_name=None,
                      func="COUNT",
                      alias="GOE_ROW_COUNT",
                      is_approximate=False,
                  )
              )
          for col_name, is_floating in columns:
              for func in aggs:
                  if func == "COUNT" and col_name == "*":
                      continue
                  approx = is_floating or func in ("AVG", "STDDEV")
                  alias = f"{func}_{col_name}".upper()
                  questions.append(
                      AggregateQuestion(
                          column_name=col_name,
                          func=func,
                          alias=alias,
                          is_approximate=approx,
                      )
                  )
          return tuple(questions)

      def chunk_questions(
          self,
          questions: Sequence[AggregateQuestion],
      ) -> tuple[tuple[AggregateQuestion, ...], ...]:
          """Split aggregate questions into batches of at most max_aggregates_per_query."""
          if not questions:
              return ()
          size = self._max_aggregates
          return tuple(tuple(questions[idx : idx + size]) for idx in range(0, len(questions), size))

      def build_batch_queries(
          self,
          *,
          dialect: SqlDialect,
          table: TableRef,
          questions: Sequence[AggregateQuestion],
          group_bys: Sequence[str] = (),
          predicate: PredicateExpr | None = None,
          raw_filters: Sequence[str] = (),
          as_of_scn: int | None = None,
          parallelism: int | None = None,
      ) -> tuple[ValidationBatchQuery, ...]:
          """Build chunked SELECT aggregate queries for the given dialect and table."""
          engine = get_dialect_engine(dialect)
          batches = self.chunk_questions(questions)
          compiled_pred_sql: str | None = None
          bind_params: dict[str, Any] = {}
          if predicate is not None:
              rendered = self._predicate_compiler.compile(predicate, dialect=dialect, bind=(dialect == "oracle"))
              compiled_pred_sql = rendered.sql
              bind_params.update(rendered.parameters)
          if dialect == "oracle" and as_of_scn is not None:
              bind_params["scn"] = as_of_scn

          filter_clauses = [
              self.parse_filter_clause(raw, source_dialect="oracle", target_dialect=dialect)
              for raw in raw_filters
              if raw.strip()
          ]
          if compiled_pred_sql:
              filter_clauses.insert(0, compiled_pred_sql)

          result: list[ValidationBatchQuery] = []
          for chunk in batches:
              sql = engine.validation_aggregate_query(
                  table=table,
                  questions=[(q.column_name, q.func, q.alias) for q in chunk],
                  group_bys=tuple(group_bys),
                  where_clauses=tuple(filter_clauses),
                  as_of_scn_bind=(dialect == "oracle" and as_of_scn is not None),
                  parallelism=parallelism if dialect == "oracle" else None,
              )
              result.append(
                  ValidationBatchQuery(
                      dialect=dialect,
                      sql=sql,
                      bind_params=dict(bind_params),
                      group_by_columns=tuple(group_bys),
                      questions=chunk,
                  )
              )
          return tuple(result)

      def build_sample_hash_query(
          self,
          *,
          dialect: SqlDialect,
          table: TableRef,
          key_columns: Sequence[str],
          value_columns: Sequence[str],
          predicate: PredicateExpr | None = None,
          as_of_scn: int | None = None,
      ) -> ValidationBatchQuery:
          """Build a dialect-aware sample hash query over key and value columns."""
          engine = get_dialect_engine(dialect)
          bind_params: dict[str, Any] = {}
          where_clauses: list[str] = []
          if predicate is not None:
              rendered = self._predicate_compiler.compile(predicate, dialect=dialect, bind=(dialect == "oracle"))
              where_clauses.append(rendered.sql)
              bind_params.update(rendered.parameters)
          if dialect == "oracle" and as_of_scn is not None:
              bind_params["scn"] = as_of_scn
          sql = engine.validation_sample_hash_query(
              table=table,
              key_columns=tuple(key_columns),
              value_columns=tuple(value_columns),
              where_clauses=tuple(where_clauses),
              as_of_scn_bind=(dialect == "oracle" and as_of_scn is not None),
          )
          question = AggregateQuestion(
              column_name=None,
              func="HASH",
              alias="GOE_SAMPLE_HASH",
              is_approximate=False,
          )
          return ValidationBatchQuery(
              dialect=dialect,
              sql=sql,
              bind_params=bind_params,
              group_by_columns=(),
              questions=(question,),
          )

      @staticmethod
      def parse_filter_clause(
          raw_filter: str,
          *,
          source_dialect: SqlDialect = "oracle",
          target_dialect: SqlDialect = "oracle",
      ) -> str:
          """Validate a raw filter expression and render it in target_dialect."""
          return get_dialect_engine(target_dialect).transpile_safe_filter(raw_filter, source_dialect=source_dialect)
  ```
