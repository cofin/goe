# Task 1.2: Implement CrossDbValidator, ValidationResult, and ColumnMismatch with run_bounded and exact Decimal comparison

## Objective

1. Create `src/goe/offload/validation/runner.py` with `ColumnMismatch(FrozenBaseStruct)`, `ValidationResult(FrozenBaseStruct)`, and `CrossDbValidator`.
2. Dispatch frontend and backend validation batch queries concurrently via `run_bounded(..., max_workers=2)` (propagating OpenTelemetry context across worker threads) with Oracle `AS OF SCN :scn` bind parameters.
3. Align grouped validation rows by normalized group-key tuple (`None`, `Decimal`, ISO `date`/`datetime`, `str`) rather than positional `zip`, eliminating false mismatches when `NULL` sort order differs across databases.
4. Compare `COUNT`, `MIN`, `MAX`, and `SUM` on exact numeric columns using exact `Decimal` equality (`Decimal(str(front)) == Decimal(str(back))`), applying quantized `Decimal` tolerance (`Decimal("1e-6")`) only when `question.is_approximate` is `True` (`AVG`, `STDDEV`, or floating-point columns).

## Context

- Current `src/goe/offload/offload_validation.py:686-705` converts numbers via `float(front)` and `float(back)` inside `isclose(float(front), float(back))`, losing precision on 16..38-digit Oracle `NUMBER` and BigQuery `NUMERIC`/`BIGNUMERIC` values.
- Current `src/goe/offload/offload_validation.py:706-735` zips `front_results[row_no]` with `back_results[row_no]` positionally.
- Task 1.1 provides `AggregateQuestion`, `ValidationBatchQuery`, and `CrossDbValidationBuilder` in `src/goe/offload/validation/ast_builder.py`.
- Chapter 5.2 provides `run_bounded` in `goe.util.concurrency` (or `goe.util.platform`), which runs callables concurrently across a bounded `ThreadPoolExecutor` while preserving `contextvars` / OpenTelemetry context.

  ```python
  # SPDX-FileCopyrightText: 2026 The GOE Authors
  # SPDX-License-Identifier: Apache-2.0

  """Concurrent cross-database validation runner with exact Decimal comparison."""

  from collections.abc import Callable, Sequence
  from datetime import date, datetime
  from decimal import Decimal, InvalidOperation
  from typing import Any

  from goe.offload.validation.ast_builder import (
      AggregateQuestion,
      CrossDbValidationBuilder,
      ValidationBatchQuery,
  )
  from goe.typing import FrozenBaseStruct
  from goe.util.concurrency import run_bounded

  __all__ = (
      "DEFAULT_APPROXIMATE_TOLERANCE",
      "ColumnMismatch",
      "CrossDbValidator",
      "QueryExecutorFn",
      "ValidationResult",
  )

  DEFAULT_APPROXIMATE_TOLERANCE: Decimal = Decimal("0.000001")

  type QueryExecutorFn = Callable[[str, dict[str, Any]], Sequence[tuple[Any, ...]]]


  class ColumnMismatch(FrozenBaseStruct):
      """Details of a single aggregate or group mismatch between frontend and backend."""

      group_key: tuple[str, ...]
      column_name: str | None
      func: str
      alias: str
      frontend_value: str | None
      backend_value: str | None


  class ValidationResult(FrozenBaseStruct):
      """Immutable outcome of a cross-database validation execution."""

      passed: bool
      batches_executed: int
      rows_compared: int
      mismatches: tuple[ColumnMismatch, ...] = ()
      frontend_sql: tuple[str, ...] = ()
      backend_sql: tuple[str, ...] = ()


  class CrossDbValidator:
      """Executes frontend and backend validation batches concurrently and compares results."""

      def __init__(
          self,
          *,
          frontend_executor: QueryExecutorFn,
          backend_executor: QueryExecutorFn,
          builder: CrossDbValidationBuilder | None = None,
          approximate_tolerance: Decimal = DEFAULT_APPROXIMATE_TOLERANCE,
      ) -> None:
          self._frontend_executor = frontend_executor
          self._backend_executor = backend_executor
          self._builder = builder or CrossDbValidationBuilder()
          self._tolerance = approximate_tolerance

      @property
      def builder(self) -> CrossDbValidationBuilder:
          """Return the underlying CrossDbValidationBuilder."""
          return self._builder

      def execute_batches(
          self,
          *,
          frontend_batches: Sequence[ValidationBatchQuery],
          backend_batches: Sequence[ValidationBatchQuery],
          execute: bool = True,
      ) -> ValidationResult:
          """Execute paired frontend and backend query batches and compare all aggregate values."""
          front_sqls = tuple(b.sql for b in frontend_batches)
          back_sqls = tuple(b.sql for b in backend_batches)
          if not execute:
              return ValidationResult(
                  passed=True,
                  batches_executed=0,
                  rows_compared=0,
                  mismatches=(),
                  frontend_sql=front_sqls,
                  backend_sql=back_sqls,
              )

          mismatches: list[ColumnMismatch] = []
          rows_compared = 0
          for front_batch, back_batch in zip(frontend_batches, backend_batches, strict=True):
              tasks: list[Callable[[], Sequence[tuple[Any, ...]]]] = [
                  lambda fb=front_batch: self._frontend_executor(fb.sql, fb.bind_params),
                  lambda bb=back_batch: self._backend_executor(bb.sql, bb.bind_params),
              ]
              front_rows, back_rows = run_bounded(tasks, max_workers=2)
              batch_mismatches, batch_rows = self._compare_batch(
                  front_batch=front_batch,
                  front_rows=front_rows,
                  back_rows=back_rows,
              )
              mismatches.extend(batch_mismatches)
              rows_compared += batch_rows

          return ValidationResult(
              passed=len(mismatches) == 0,
              batches_executed=len(frontend_batches),
              rows_compared=rows_compared,
              mismatches=tuple(mismatches),
              frontend_sql=front_sqls,
              backend_sql=back_sqls,
          )

      def _compare_batch(
          self,
          *,
          front_batch: ValidationBatchQuery,
          front_rows: Sequence[tuple[Any, ...]],
          back_rows: Sequence[tuple[Any, ...]],
      ) -> tuple[list[ColumnMismatch], int]:
          group_cols = front_batch.group_by_columns
          num_keys = len(group_cols)
          front_map = self._index_by_group_key(front_rows, group_cols)
          back_map = self._index_by_group_key(back_rows, group_cols)
          all_keys = list(dict.fromkeys([*front_map.keys(), *back_map.keys()]))
          mismatches: list[ColumnMismatch] = []

          for key in all_keys:
              f_vals = front_map.get(key)
              b_vals = back_map.get(key)
              if f_vals is None or b_vals is None:
                  for q in front_batch.questions:
                      mismatches.append(
                          ColumnMismatch(
                              group_key=key,
                              column_name=q.column_name,
                              func=q.func,
                              alias=q.alias,
                              frontend_value="MISSING" if f_vals is None else "PRESENT",
                              backend_value="MISSING" if b_vals is None else "PRESENT",
                          )
                      )
                  continue

              for idx, question in enumerate(front_batch.questions):
                  f_cell = f_vals[num_keys + idx]
                  b_cell = b_vals[num_keys + idx]
                  if not self._cells_match(f_cell, b_cell, question=question):
                      mismatches.append(
                          ColumnMismatch(
                              group_key=key,
                              column_name=question.column_name,
                              func=question.func,
                              alias=question.alias,
                              frontend_value=self._format_scalar(f_cell),
                              backend_value=self._format_scalar(b_cell),
                          )
                      )
          return mismatches, len(all_keys)

      def _index_by_group_key(
          self,
          rows: Sequence[tuple[Any, ...]],
          group_cols: tuple[str, ...],
      ) -> dict[tuple[str, ...], tuple[Any, ...]]:
          if not group_cols:
              if not rows:
                  return {}
              return {("TOTAL",): rows[0]}
          indexed: dict[tuple[str, ...], tuple[Any, ...]] = {}
          num_keys = len(group_cols)
          for row in rows:
              key = tuple(
                  f"{col}={self._normalize_key_scalar(val)}"
                  for col, val in zip(group_cols, row[:num_keys], strict=True)
              )
              indexed[key] = row
          return indexed

      def _cells_match(
          self,
          front: Any,
          back: Any,
          *,
          question: AggregateQuestion,
      ) -> bool:
          if front is None and back is None:
              return True
          if front is None or back is None:
              return False
          f_dec = self._to_decimal(front)
          b_dec = self._to_decimal(back)
          if f_dec is not None and b_dec is not None:
              if question.is_approximate:
                  return abs(f_dec - b_dec) <= self._tolerance
              return f_dec == b_dec
          return self._normalize_key_scalar(front) == self._normalize_key_scalar(back)

      @staticmethod
      def _to_decimal(val: Any) -> Decimal | None:
          if isinstance(val, bool):
              return None
          if isinstance(val, Decimal):
              return val
          if isinstance(val, (int, float)):
              return Decimal(str(val))
          if isinstance(val, str):
              try:
                  return Decimal(val)
              except InvalidOperation:
                  return None
          return None

      @staticmethod
      def _normalize_key_scalar(val: Any) -> str:
          if val is None:
              return "NULL"
          if isinstance(val, datetime):
              return val.strftime("%Y-%m-%d %H:%M:%S.%f")
          if isinstance(val, date):
              return val.strftime("%Y-%m-%d")
          if isinstance(val, Decimal):
              return format(val.normalize(), "f")
          return str(val)

      @classmethod
      def _format_scalar(cls, val: Any) -> str | None:
          if val is None:
              return None
          return cls._normalize_key_scalar(val)
  ```
