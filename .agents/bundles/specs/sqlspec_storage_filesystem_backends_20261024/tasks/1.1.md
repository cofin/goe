## Objective

## Context

- `src/goe/filesystem/goe_gcs.py:258` enumerates all objects under a prefix when checking whether a URI is a virtual directory instead of bounding the prefix probe to 2 entries.
- `src/goe/filesystem/goe_azure.py:14, 159, 326` catches `azure.common.AzureMissingResourceHttpError`, which does not exist in Azure Blob v12 (`ResourceNotFoundError`).
- `src/goe/filesystem/goe_s3.py:137` sends unchunked key lists to S3 batch delete, failing when a staging prefix has more than 1,000 objects.
- `.venv/lib/python3.14/site-packages/sqlspec/storage/` provides `StorageBridge` and `ObStoreBackend` / `FsspecBackend` supporting local and cloud object stores.

```python
# SPDX-FileCopyrightText: 2026 The GOE Authors
# SPDX-License-Identifier: Apache-2.0

"""Unified cloud and local staging storage service backed by sqlspec.storage."""

from collections.abc import Iterable, Iterator
from pathlib import Path
from typing import Any
from urllib.parse import urlparse

from goe.typing import FrozenBaseStruct

__all__ = [
    "DELETE_BATCH_LIMIT",
    "StagingStorageService",
    "StorageObjectStat",
]

DELETE_BATCH_LIMIT = 1000
CLOUD_STORAGE_SCHEMES = frozenset({"file", "gs", "s3", "s3a", "abfs", "abfss", "wasb", "wasbs", "adl"})


class StorageObjectStat(FrozenBaseStruct):
    """Immutable metadata returned by StagingStorageService.stat_uri."""

    uri: str
    kind: str
    size_bytes: int = 0


class StagingStorageService:
    """Provides scheme-preserving object storage operations on top of sqlspec.storage."""

    def __init__(
        self,
        scheme: str,
        *,
        container: str | None = None,
        dry_run: bool = False,
        backend: Any = None,
        credentials_provider: Any = None,
    ) -> None:
        normalized = scheme.lower().rstrip(":/") or "file"
        if normalized not in CLOUD_STORAGE_SCHEMES:
            raise ValueError(f"Unsupported storage scheme: {scheme}")
        self._scheme = normalized
        self._container = container
        self._dry_run = dry_run
        self._backend = backend
        self._credentials_provider = credentials_provider

    @property
    def scheme(self) -> str:
        """Return the canonical URI scheme configured for this service."""
        return self._scheme

    @property
    def dry_run(self) -> bool:
        """Return whether mutating calls should log only without modifying storage."""
        return self._dry_run

    def format_uri(self, path: str, *, container_override: str | None = None) -> str:
        """Construct a scheme-exact URI for the given container and object key."""
        if "://" in path:
            return path
        clean_path = path.lstrip("/")
        if self._scheme == "file":
            return f"file:///{clean_path}"
        bucket = container_override or self._container or ""
        return f"{self._scheme}://{bucket}/{clean_path}" if bucket else f"{self._scheme}:///{clean_path}"

    def upload_stream(
        self,
        target_uri: str,
        chunks: Iterable[bytes] | bytes,
        *,
        overwrite: bool = False,
    ) -> int:
        """Stream bytes to target_uri, enforcing create-only semantics when overwrite is False."""
        payload = chunks if isinstance(chunks, bytes) else b"".join(chunks)
        if self._dry_run:
            return len(payload)
        if self._backend is not None:
            self._backend.put(target_uri, payload, overwrite=overwrite)
            return len(payload)
        local_path = self._resolve_local_path(target_uri)
        if not overwrite and local_path.exists():
            raise FileExistsError(f"Target URI already exists and overwrite=False: {target_uri}")
        local_path.parent.mkdir(parents=True, exist_ok=True)
        local_path.write_bytes(payload)
        return len(payload)

    def download_stream(self, source_uri: str, *, chunk_size: int = 1024 * 1024) -> Iterator[bytes]:
        """Yield byte chunks from source_uri."""
        if self._backend is not None:
            data = self._backend.get(source_uri)
            for offset in range(0, len(data), chunk_size):
                yield data[offset : offset + chunk_size]
            return
        local_path = self._resolve_local_path(source_uri)
        with local_path.open("rb") as handle:
            while chunk := handle.read(chunk_size):
                yield chunk

    def list_prefix(self, prefix_uri: str, *, max_results: int | None = None) -> list[str]:
        """List URIs under prefix_uri, optionally bounding enumeration to max_results."""
        if self._backend is not None:
            items = list(self._backend.list_prefix(prefix_uri, max_results=max_results))
            return items[:max_results] if max_results is not None else items
        local_path = self._resolve_local_path(prefix_uri)
        if not local_path.exists():
            return []
        if local_path.is_file():
            return [self.format_uri(str(local_path))]
        uris: list[str] = []
        for child in sorted(local_path.rglob("*")):
            if child.is_file():
                uris.append(self.format_uri(str(child)))
                if max_results is not None and len(uris) >= max_results:
                    break
        return uris

    def stat_uri(self, uri: str) -> StorageObjectStat | None:
        """Inspect uri as a file or bounded virtual directory prefix (max_results=2)."""
        if self._backend is not None:
            head = self._backend.head(uri)
            if head is not None:
                return StorageObjectStat(uri=uri, kind="FILE", size_bytes=int(head.get("size", 0)))
            probe = self.list_prefix(uri, max_results=2)
            if probe:
                return StorageObjectStat(uri=uri, kind="DIRECTORY", size_bytes=0)
            return None
        local_path = self._resolve_local_path(uri)
        if local_path.is_file():
            return StorageObjectStat(uri=uri, kind="FILE", size_bytes=local_path.stat().st_size)
        if local_path.is_dir():
            return StorageObjectStat(uri=uri, kind="DIRECTORY", size_bytes=0)
        return None

    def delete_prefix(self, prefix_uri: str) -> int:
        """Delete all objects under prefix_uri in batches of at most 1,000 keys."""
        matched = self.list_prefix(prefix_uri)
        if not matched or self._dry_run:
            return len(matched)
        if self._backend is not None:
            for start in range(0, len(matched), DELETE_BATCH_LIMIT):
                batch = matched[start : start + DELETE_BATCH_LIMIT]
                self._backend.delete_many(batch)
            return len(matched)
        for item_uri in matched:
            path = self._resolve_local_path(item_uri)
            if path.exists():
                path.unlink()
        return len(matched)

    @staticmethod
    def _resolve_local_path(uri_or_path: str) -> Path:
        """Convert a file:// URI or local filesystem path into a Path object."""
        if "://" in uri_or_path:
            parsed = urlparse(uri_or_path)
            return Path(parsed.path)
        return Path(uri_or_path)
```
