## Context

- `msgspec==0.22.0` (`.venv/lib/python3.14/site-packages/msgspec/__init__.py`) provides `msgspec.Struct`, `msgspec.UnsetType`, `msgspec.UNSET`, `msgspec.structs.asdict`, `msgspec.json.Encoder`, `msgspec.json.Decoder`, and `msgspec.json.format`.
- Every `msgspec.Struct` subclass in GOE must use `gc=False` so CPython omits `PyGC_Head` tracking on value objects.
- `src/goe/util/json_tools.py:1-56` provides `is_json(json_str)` and `check_and_format_json(json_str, compact=True)`, used in:
  - `src/goe/offload/bigquery/bigquery_backend_table.py:25`
  - `src/goe/offload/snowflake/snowflake_backend_table.py:28`
  - `src/goe/offload/microsoft/synapse_backend_table.py:25`
  Replacing `check_and_format_json(val, compact=True)` with `format_json(val, indent=0)` (or `encode_json(decode_json(val))` when compact) preserves exact validation and formatting semantics.

```python
# SPDX-FileCopyrightText: 2026 The GOE Authors
# SPDX-License-Identifier: Apache-2.0

"""Core typing primitives and GC-exempt msgspec Struct base classes."""

from collections.abc import Mapping
from typing import Any

import msgspec
from msgspec import UNSET, UnsetType

__all__ = [
    "UNSET",
    "BaseStruct",
    "CamelStruct",
    "FrozenBaseStruct",
    "FrozenCamelStruct",
    "UnsetType",
    "WireFrameStruct",
    "filter_unset",
]


class BaseStruct(msgspec.Struct, gc=False):
    """Mutable msgspec Struct with cyclic garbage collection disabled."""

    def to_dict(self, *, exclude_unset: bool = False) -> dict[str, Any]:
        """Convert the struct into a plain dictionary."""
        raw = msgspec.structs.asdict(self)
        return filter_unset(raw) if exclude_unset else raw


class FrozenBaseStruct(msgspec.Struct, frozen=True, gc=False):
    """Immutable msgspec Struct with cyclic garbage collection disabled."""

    def to_dict(self, *, exclude_unset: bool = False) -> dict[str, Any]:
        """Convert the struct into a plain dictionary."""
        raw = msgspec.structs.asdict(self)
        return filter_unset(raw) if exclude_unset else raw


class WireFrameStruct(msgspec.Struct, frozen=True, gc=False, array_like=True):
    """Compact tuple-backed immutable struct for hot-path wire and plan records."""


class CamelStruct(msgspec.Struct, rename="camel", gc=False):
    """Mutable msgspec Struct with camelCase field renaming and cyclic GC disabled."""


class FrozenCamelStruct(msgspec.Struct, rename="camel", frozen=True, gc=False):
    """Immutable msgspec Struct with camelCase field renaming and cyclic GC disabled."""


def filter_unset(data: Mapping[str, Any] | msgspec.Struct) -> dict[str, Any]:
    """Return a dictionary with all keys whose value is UNSET removed."""
    source = msgspec.structs.asdict(data) if isinstance(data, msgspec.Struct) else data
    return {key: value for key, value in source.items() if value is not UNSET}
```

```python
# SPDX-FileCopyrightText: 2026 The GOE Authors
# SPDX-License-Identifier: Apache-2.0

"""Centralized msgspec JSON serialization and deserialization utilities."""

from datetime import date, datetime
from decimal import Decimal
from pathlib import Path
from typing import Any, overload
from uuid import UUID

import msgspec

__all__ = [
    "decode_json",
    "encode_json",
    "encode_json_bytes",
    "encode_json_into",
    "format_json",
]


def encode_json_bytes(value: Any) -> bytes:
    """Serialize value to UTF-8 JSON bytes using the shared msgspec encoder."""
    return _ENCODER.encode(value)


def encode_json(value: Any) -> str:
    """Serialize value to a UTF-8 JSON string."""
    return _ENCODER.encode(value).decode("utf-8")


def encode_json_into(value: Any, buffer: bytearray, offset: int = 0) -> None:
    """Serialize value directly into a pre-allocated bytearray at offset."""
    _ENCODER.encode_into(value, buffer, offset)


def format_json(value: str | bytes, *, indent: int = 2) -> str:
    """Validate and format a JSON string or bytes payload."""
    raw = value.encode("utf-8") if isinstance(value, str) else value
    _DECODER.decode(raw)
    formatted = msgspec.json.format(raw, indent=indent)
    return formatted.decode("utf-8")


@overload
def decode_json[T](payload: str | bytes | bytearray | memoryview, *, target_type: type[T]) -> T: ...


@overload
def decode_json(payload: str | bytes | bytearray | memoryview, *, target_type: None = None) -> Any: ...


def decode_json[T](
    payload: str | bytes | bytearray | memoryview,
    *,
    target_type: type[T] | None = None,
) -> T | Any:
    """Deserialize a JSON string or byte buffer into target_type or generic Python primitives."""
    if target_type is None:
        return _DECODER.decode(payload)
    return msgspec.json.decode(payload, type=target_type, dec_hook=_default_dec_hook)


def _default_enc_hook(obj: Any) -> Any:
    """Encode non-native domain types into JSON-compatible primitives."""
    if isinstance(obj, Decimal):
        return str(obj)
    if isinstance(obj, (datetime, date)):
        return obj.isoformat()
    if isinstance(obj, (Path, UUID)):
        return str(obj)
    if isinstance(obj, (set, frozenset)):
        return sorted(obj, key=str)
    if hasattr(obj, "as_str") and callable(obj.as_str):
        return obj.as_str()
    raise TypeError(f"Object of type {type(obj).__name__} is not JSON serializable")


def _default_dec_hook(target_type: type[Any], obj: Any) -> Any:
    """Decode custom domain types during typed msgspec deserialization."""
    if target_type is Decimal and isinstance(obj, (str, int, float)):
        return Decimal(str(obj))
    if target_type is Path and isinstance(obj, str):
        return Path(obj)
    raise TypeError(f"Cannot decode {type(obj).__name__} into {target_type}")


_ENCODER = msgspec.json.Encoder(enc_hook=_default_enc_hook)
_DECODER = msgspec.json.Decoder()
```
