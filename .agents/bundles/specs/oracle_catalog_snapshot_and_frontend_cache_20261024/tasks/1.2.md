# Task 1.2: Implement FrontendCatalogCache with pool session context and LAST_DDL_TIME validation

## Objective

1. Create `src/goe/offload/frontend_cache.py` implementing a thread-safe, bounded `FrontendCatalogCache` (`maxsize=512`, `ttl_seconds=30.0`) that caches `OracleSessionContext` per pool DSN and `OracleTableSnapshot` per `(dsn, owner, table)`.
2. Support two consistency modes on `get_table_snapshot`:
   - Execution-pinned mode (`snapshot_scn` provided): returns the cached snapshot pinned to that `(dsn, owner, table, snapshot_scn)` without re-querying for the duration of the command.
   - Live/Listener mode (`snapshot_scn is None`): serves the cached snapshot within the 30-second TTL window and, once the TTL expires, validates `MAX(LAST_DDL_TIME)` from `ALL_OBJECTS` (`OracleCatalogService.load_object_last_ddl_time`); if `last_ddl_time` matches the cached snapshot's `last_ddl_time`, extends the TTL without re-running the full catalog queries, otherwise reloads the snapshot.
3. Provide `invalidate_table(dsn, owner, table)`, `invalidate_schema(dsn, owner)`, and `clear()` so DDL executed by GOE deterministically evicts affected entries.

## Context

- During a single offload, schema sync, or status report execution, `OracleFrontendApi` and `OracleSourceTable` are queried repeatedly for the same `(owner, table)` (`get_columns`, `get_partition_columns`, `get_table_size`, `get_table_default_parallelism`, `is_view`). Caching `OracleTableSnapshot` in `FrontendCatalogCache` eliminates redundant round trips across all callers sharing the same `OracleDatabaseRegistry`.
- In the long-running Listener process, tables may be altered externally in Oracle. Validating expired cache entries with the lightweight `get_object_last_ddl_time` query (`MAX(last_ddl_time)` from `all_objects`) refreshes the 30-second TTL in 1 lightweight round trip when no DDL has occurred, and reloads the full snapshot when `last_ddl_time` has advanced.
- `OracleSessionContext` (`session_user`, `db_unique_name`, `version`, `compatible`, `db_block_size`, `max_string_size`) is invariant for a given `(dsn, role)` pool and is cached for the lifetime of the registry.

```python
# SPDX-FileCopyrightText: 2026 The GOE Authors
# SPDX-License-Identifier: Apache-2.0

"""Bounded thread-safe frontend catalog cache for OracleSessionContext and OracleTableSnapshot."""

from collections import OrderedDict
from collections.abc import Callable
import threading
import time
from typing import Any, Final

import msgspec

from goe.offload.oracle.catalog import (
    OracleCatalogService,
    OracleSessionContext,
    OracleTableSnapshot,
)

__all__ = (
    "DEFAULT_CATALOG_CACHE_MAXSIZE",
    "DEFAULT_CATALOG_CACHE_TTL_SECONDS",
    "FrontendCatalogCache",
    "get_frontend_catalog_cache",
)

DEFAULT_CATALOG_CACHE_MAXSIZE: Final = 512
DEFAULT_CATALOG_CACHE_TTL_SECONDS: Final = 30.0


class _CachedSnapshotEntry(msgspec.Struct, frozen=True, gc=False, array_like=True):
    """Internal cache envelope pairing an ``OracleTableSnapshot`` with its monotonic expiration time."""

    snapshot: OracleTableSnapshot
    expires_at: float


class FrontendCatalogCache:
    """Thread-safe LRU cache for ``OracleSessionContext`` and ``OracleTableSnapshot``."""

    def __init__(
        self,
        catalog_service: OracleCatalogService | None = None,
        *,
        maxsize: int = DEFAULT_CATALOG_CACHE_MAXSIZE,
        ttl_seconds: float = DEFAULT_CATALOG_CACHE_TTL_SECONDS,
        clock: Callable[[], float] = time.monotonic,
    ) -> None:
        self._catalog = catalog_service or OracleCatalogService()
        self._maxsize = maxsize
        self._ttl_seconds = ttl_seconds
        self._clock = clock
        self._lock = threading.RLock()
        self._session_contexts: dict[str, OracleSessionContext] = {}
        self._snapshots: OrderedDict[tuple[str, str, str], _CachedSnapshotEntry] = OrderedDict()

    @property
    def catalog_service(self) -> OracleCatalogService:
        """Return the underlying ``OracleCatalogService``."""
        return self._catalog

    def get_session_context(self, dsn: str, driver: Any) -> OracleSessionContext:
        """Return the cached ``OracleSessionContext`` for ``dsn``, loading it on first access."""
        with self._lock:
            cached = self._session_contexts.get(dsn)
            if cached is not None:
                return cached
        loaded = self._catalog.load_session_context(driver)
        with self._lock:
            self._session_contexts[dsn] = loaded
        return loaded

    def get_table_snapshot(
        self,
        dsn: str,
        owner: str,
        table_name: str,
        driver: Any,
        *,
        snapshot_scn: int | None = None,
        force_refresh: bool = False,
    ) -> OracleTableSnapshot:
        """Return a fresh or validated ``OracleTableSnapshot`` for ``(dsn, owner, table_name)``."""
        key = (dsn, owner.upper(), table_name.upper())
        now = self._clock()
        if not force_refresh:
            with self._lock:
                entry = self._snapshots.get(key)
                if entry is not None:
                    if snapshot_scn is not None and entry.snapshot.snapshot_scn == snapshot_scn:
                        self._snapshots.move_to_end(key)
                        return entry.snapshot
                    if now <= entry.expires_at:
                        self._snapshots.move_to_end(key)
                        return entry.snapshot

            if entry is not None and entry.snapshot.exists and entry.snapshot.last_ddl_time is not None:
                current_ddl = self._catalog.load_object_last_ddl_time(driver, key[1], key[2])
                if current_ddl is not None and current_ddl == entry.snapshot.last_ddl_time:
                    refreshed = _CachedSnapshotEntry(
                        snapshot=entry.snapshot,
                        expires_at=self._clock() + self._ttl_seconds,
                    )
                    with self._lock:
                        self._snapshots[key] = refreshed
                        self._snapshots.move_to_end(key)
                    return entry.snapshot

        session_ctx = self.get_session_context(dsn, driver)
        snapshot = self._catalog.load_table_snapshot(
            driver,
            key[1],
            key[2],
            db_block_size=session_ctx.db_block_size,
            snapshot_scn=snapshot_scn,
        )
        self.put_table_snapshot(dsn, snapshot)
        return snapshot

    def put_table_snapshot(self, dsn: str, snapshot: OracleTableSnapshot) -> None:
        """Store ``snapshot`` in the cache under ``(dsn, snapshot.owner, snapshot.table_name)``."""
        key = (dsn, snapshot.owner.upper(), snapshot.table_name.upper())
        entry = _CachedSnapshotEntry(
            snapshot=snapshot,
            expires_at=self._clock() + self._ttl_seconds,
        )
        with self._lock:
            self._snapshots[key] = entry
            self._snapshots.move_to_end(key)
            while len(self._snapshots) > self._maxsize:
                self._snapshots.popitem(last=False)

    def invalidate_table(self, dsn: str, owner: str, table_name: str) -> bool:
        """Evict the cached snapshot for ``(dsn, owner, table_name)`` if present."""
        key = (dsn, owner.upper(), table_name.upper())
        with self._lock:
            return self._snapshots.pop(key, None) is not None

    def invalidate_schema(self, dsn: str, owner: str) -> int:
        """Evict all cached snapshots for ``(dsn, owner)``."""
        owner_upper = owner.upper()
        with self._lock:
            victims = [k for k in self._snapshots if k[0] == dsn and k[1] == owner_upper]
            for key in victims:
                self._snapshots.pop(key, None)
            return len(victims)

    def clear(self) -> None:
        """Clear all cached session contexts and table snapshots."""
        with self._lock:
            self._session_contexts.clear()
            self._snapshots.clear()


_DEFAULT_CACHE = FrontendCatalogCache()


def get_frontend_catalog_cache() -> FrontendCatalogCache:
    """Return the process-wide ``FrontendCatalogCache`` instance."""
    return _DEFAULT_CACHE
```
