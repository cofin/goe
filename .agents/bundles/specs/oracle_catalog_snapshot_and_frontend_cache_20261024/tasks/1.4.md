```python
# SPDX-FileCopyrightText: 2026 The GOE Authors
# SPDX-License-Identifier: Apache-2.0

"""Adapters converting frozen OraclePartitionMeta and OracleSubpartitionMeta into RdbmsPartition objects."""

from collections.abc import Callable
from typing import Any

from goe.offload.offload_source_table import RdbmsPartition
from goe.offload.oracle.catalog import OracleTableSnapshot

__all__ = (
    "rdbms_partitions_from_snapshot",
    "rdbms_subpartitions_from_snapshot",
)


def rdbms_partitions_from_snapshot(
    snapshot: OracleTableSnapshot,
    *,
    populate_hvs: bool = True,
    decode_fn: Callable[[str | None], tuple[tuple[Any, ...], tuple[str, ...]]] | None = None,
) -> list[Any]:
    """Convert ``snapshot.partitions`` into ``RdbmsPartition`` objects in descending position order."""
    result: list[Any] = []
    for meta in snapshot.partitions:
        hv_ind: tuple[str, ...] | None = None
        hv_py: tuple[Any, ...] | None = None
        if populate_hvs:
            if decode_fn is not None and meta.high_value_csv is not None:
                hv_py, hv_ind = decode_fn(meta.high_value_csv)
            else:
                hv_ind = meta.high_value_tokens
                hv_py = meta.high_value_python
        result.append(
            RdbmsPartition.by_name(
                partition_count=1,
                partition_name=meta.partition_name,
                partition_position=meta.partition_position,
                subpartition_count=meta.subpartition_count,
                subpartition_names=meta.subpartition_names,
                high_values_csv=meta.high_value_csv,
                partition_size=meta.size_in_bytes,
                num_rows=meta.num_rows,
                high_values_individual=hv_ind,
                high_values_python=hv_py,
            )
        )
    return result


def rdbms_subpartitions_from_snapshot(
    snapshot: OracleTableSnapshot,
    *,
    populate_hvs: bool = True,
    decode_fn: Callable[[str | None], tuple[tuple[Any, ...], tuple[str, ...]]] | None = None,
) -> list[Any]:
    """Convert ``snapshot.subpartitions`` into ``RdbmsPartition`` objects sorted by high value descending."""
    part_count = len(snapshot.partitions)
    result: list[Any] = []
    for meta in snapshot.subpartitions:
        hv_ind: tuple[str, ...] | None = None
        hv_py: tuple[Any, ...] | None = None
        if populate_hvs:
            if decode_fn is not None and meta.high_value_csv is not None:
                hv_py, hv_ind = decode_fn(meta.high_value_csv)
            else:
                hv_ind = meta.high_value_tokens
                hv_py = meta.high_value_python
        result.append(
            RdbmsPartition.by_name(
                partition_count=part_count,
                partition_name=meta.partition_name,
                partition_position=meta.partition_position,
                subpartition_count=meta.subpartition_count,
                subpartition_name=meta.subpartition_name,
                subpartition_position=meta.subpartition_position,
                high_values_csv=meta.high_value_csv,
                partition_size=meta.size_in_bytes,
                num_rows=meta.num_rows,
                high_values_individual=hv_ind,
                high_values_python=hv_py,
            )
        )
    if populate_hvs:
        result.sort(key=lambda x: x.high_values_python or (), reverse=True)
    return result
```
