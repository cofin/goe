# Task 1.1: Add OracleCatalogService and frozen snapshot models in src/goe/offload/oracle/catalog.py

## Objective

1. Create `src/goe/offload/oracle/sql/oracle_catalog.sql` consolidating Oracle session context, object `LAST_DDL_TIME`, table/view/segment/compression/parallelism header, column + partition key + optimizer statistics, constraint + index metadata, top-level partition metadata, and subpartition metadata into named `SQLFileLoader` queries without joining `dba_tab_partitions` (`LONG` `high_value`) to `dba_tab_subpartitions`.
2. Create `src/goe/offload/oracle/catalog.py` defining `OracleSessionContext`, `OracleStorageStats`, `OracleColumnMeta`, `OracleConstraintMeta`, `OracleIndexMeta`, `OracleSubpartitionMeta`, `OraclePartitionMeta`, `OracleTableSnapshot`, pure-Python Oracle `RAW` statistics decoders (`decode_oracle_raw_stat`), partition `HIGH_VALUE` literal parsers (`parse_high_value_literals`), and `OracleCatalogService`.
3. Create `tests/unit/offload/oracle/test_catalog.py` verifying snapshot hydration, `HIGH_VALUE` parsing across `DATE`, `TIMESTAMP`, `NUMBER`, `VARCHAR2`, `MAXVALUE`, and `DEFAULT`, pure-Python `RAW` statistics decoding, and zero cartesian `LONG` duplication on composite-partitioned tables.

## Context

- `_get_partitions` (`src/goe/offload/oracle/oracle_offload_source_table.py:1158-1208`) currently joins `dba_tab_partitions` to `dba_tab_subpartitions`, duplicating `tp.high_value` (`LONG`) for every subpartition row. Fetching `get_table_partitions` and `get_table_subpartitions` as separate queries transfers each partition `HIGH_VALUE` once and assembles `subpartitions: tuple[OracleSubpartitionMeta, ...]` in Python in $O(P + S)$ time.
- `_get_column_low_high_values` (`oracle_offload_source_table.py:1414-1448`) executes a per-column query (`UTL_RAW.CAST_TO_*` or `offload.get_column_low_high_dates`) against `all_tab_cols`. Fetching `low_value` and `high_value` (`RAW`) directly in `get_table_columns` and decoding Oracle's internal `NUMBER` (base-100 exponent/mantissa), `DATE`/`TIMESTAMP` (7/11-byte century+100, year+100, month, day, hour+1, minute+1, second+1), `BINARY_FLOAT`/`BINARY_DOUBLE`, and `VARCHAR2` formats in pure Python eliminates all per-column stats round trips.
- `OracleFrontendApi` (`src/goe/offload/oracle/oracle_frontend_api.py:297-308, 370-381, 871-875`) and `OracleSourceTable` (`oracle_offload_source_table.py:969-973, 1462-1466`) query `session_user`, `db_unique_name`, `compatible`, `db_block_size`, `max_string_size`, and server version separately; `get_session_context` loads all of them in a single round trip.

,      tp.partition_position
,      tp.subpartition_count
,      CASE WHEN pt.partitioning_type = 'HASH' THEN NULL ELSE tp.high_value END AS high_value
,      tp.num_rows
,      tp.blocks
,      tp.avg_row_len
,      COALESCE(seg.bytes, 0) AS bytes
FROM   dba_tab_partitions tp
JOIN   dba_part_tables pt
  ON   pt.owner = tp.table_owner AND pt.table_name = tp.table_name
JOIN   dba_tables t

LEFT JOIN dba_constraints c
  ON   c.owner = t.owner AND c.table_name = t.table_name AND c.constraint_type = 'P'

       ,      s.partition_name
       ,      SUM(s.bytes) AS bytes
       FROM   dba_segments s
       WHERE  s.owner = :owner
       GROUP BY s.owner, s.segment_name, s.partition_name
) seg
  ON   seg.owner = tp.table_owner
 AND   seg.segment_name = DECODE(t.iot_type, 'IOT', c.index_name, t.table_name)
 AND   seg.partition_name = tp.partition_name
WHERE  tp.table_owner = :owner
AND    tp.table_name = :table_name
ORDER BY tp.partition_position DESC;

-- name: get_table_subpartitions
SELECT tsp.partition_name
,      tp.partition_position
,      tp.subpartition_count
,      tsp.subpartition_name
,      tsp.subpartition_position
,      tsp.high_value
,      NVL(tsp.num_rows, 0) AS num_rows
,      tsp.blocks
,      tsp.avg_row_len
,      COALESCE(seg.bytes, 0) AS bytes
FROM   dba_tab_subpartitions tsp
JOIN   dba_tab_partitions tp
  ON   tp.table_owner = tsp.table_owner
 AND   tp.table_name = tsp.table_name
 AND   tp.partition_name = tsp.partition_name
JOIN   dba_tables t
  ON   t.owner = tsp.table_owner AND t.table_name = tsp.table_name
LEFT JOIN dba_constraints c
  ON   c.owner = t.owner AND c.table_name = t.table_name AND c.constraint_type = 'P'
LEFT JOIN (
       SELECT s.owner
       ,      s.segment_name
       ,      s.partition_name
       ,      SUM(s.bytes) AS bytes
       FROM   dba_segments s
       WHERE  s.owner = :owner
       GROUP BY s.owner, s.segment_name, s.partition_name
) seg
  ON   seg.owner = tsp.table_owner
 AND   seg.segment_name = DECODE(t.iot_type, 'IOT', c.index_name, t.table_name)
 AND   seg.partition_name = tsp.subpartition_name
WHERE  tsp.table_owner = :owner
AND    tsp.table_name = :table_name
ORDER BY tp.partition_position DESC, tsp.subpartition_position ASC;

class OracleTableSnapshot(FrozenBaseStruct, kw_only=True):
    """Consolidated, immutable catalog snapshot for a single Oracle table or view."""

    owner: str
    table_name: str
    object_type: str | None = None
    iot_type: str | None = None
    partitioned: bool = False
    partitioning_type: str | None = None
    subpartitioning_type: str | None = None
    dependencies_enabled: bool = False
    degree: str | None = None
    compression_enabled: bool = False
    storage_stats: OracleStorageStats = OracleStorageStats()
    columns: tuple[OracleColumnMeta, ...] = ()
    constraints: tuple[OracleConstraintMeta, ...] = ()
    indexes: tuple[OracleIndexMeta, ...] = ()
    partitions: tuple[OraclePartitionMeta, ...] = ()
    subpartitions: tuple[OracleSubpartitionMeta, ...] = ()
    last_ddl_time: datetime.datetime | None = None
    snapshot_scn: int | None = None

    @property
    def exists(self) -> bool:
        """Return True when the table or view exists in the Oracle catalog."""
        return self.object_type is not None

    @property
    def is_view(self) -> bool:
        """Return True when the object is a view rather than a base table."""
        return self.object_type == "VIEW"

    @property
    def is_iot(self) -> bool:
        """Return True when the table is an Index-Organized Table."""
        return self.iot_type == "IOT"

    @property
    def is_subpartitioned(self) -> bool:
        """Return True when the table is composite-partitioned."""
        return bool(self.partitioned and self.subpartitioning_type and self.subpartitioning_type != "NONE")

    @property
    def primary_key_columns(self) -> tuple[str, ...]:
        """Return primary key column names in constraint position order."""
        for constraint in self.constraints:
            if constraint.constraint_type == "P":
                return constraint.column_names
        return ()

    @property
    def partition_column_names(self) -> tuple[str, ...]:
        """Return top-level partition key column names in position order."""
        part_cols = [c for c in self.columns if c.partition_position is not None]
        part_cols.sort(key=lambda c: c.partition_position or 0)
        return tuple(c.name for c in part_cols)

    @property
    def subpartition_column_names(self) -> tuple[str, ...]:
        """Return subpartition key column names in position order."""
        subpart_cols = [c for c in self.columns if c.subpartition_position is not None]
        subpart_cols.sort(key=lambda c: c.subpartition_position or 0)
        return tuple(c.name for c in subpart_cols)

    @property
    def hash_bucket_candidate(self) -> str | None:
        """Return the column name with the highest NDV non-null factor from cached column statistics."""
        table_rows = max(int(self.storage_stats.num_rows or 1), 1)
        best_col: str | None = None
        best_score = -1.0
        for col in self.columns:
            if col.data_type in _HASH_UNSUITABLE_TYPES or col.num_distinct is None:
                continue
            nulls = int(col.num_nulls or 0)
            score = float(col.num_distinct) * ((max(table_rows - nulls, 0)) / float(table_rows))
            if score > best_score:
                best_score = score
                best_col = col.name
        return best_col

    def to_table_stats_dict(self) -> dict[str, Any] | None:
        """Return the optimizer statistics dictionary structure consumed by offload planning."""
        if self.storage_stats.num_rows is None and not self.columns:
            return None
        col_stats = {
            col.name: {
                "ndv": col.num_distinct,
                "num_nulls": col.num_nulls,
                "avg_col_len": col.avg_col_len,
                "low_value": col.low_value,
                "high_value": col.high_value,
            }
            for col in self.columns
        }
        part_stats = {
            part.partition_name: {
                "num_rows": part.num_rows,
                "num_bytes": part.size_in_bytes,
                "avg_row_len": part.avg_row_len,
            }
            for part in self.partitions
        }
        return {
            "num_rows": self.storage_stats.num_rows,
            "num_bytes": self.storage_stats.num_bytes,
            "avg_row_len": self.storage_stats.avg_row_len,
            "column_stats": col_stats,
            "partition_stats": part_stats,
        }


def split_high_value_tokens(hv_string: str | None) -> tuple[str, ...]:
    """Split an Oracle ``HIGH_VALUE`` string on top-level commas and map ``MAXVALUE`` / ``DEFAULT`` tokens."""
    if not hv_string:
        return ()
    tokens = [hv_val.strip() for hv_val in ORACLE_SPLIT_HIGH_VALUE_RE.findall(hv_string)]
    return tuple(_map_open_partition_token(token) for token in tokens)


def parse_single_high_value_token(
    token: str,
    data_type: str,
    *,
    strict: bool = True,
) -> Any:
    """Parse one Oracle ``HIGH_VALUE`` literal token into its canonical Python representation."""
    upper_token = token.upper()
    if upper_token in ("DEFAULT", offload_constants.PART_OUT_OF_LIST):
        return offload_constants.PART_OUT_OF_LIST
    dtype = data_type.upper()
    if dtype == ORACLE_TYPE_DATE or dtype.startswith(ORACLE_TYPE_TIMESTAMP):
        if upper_token in ("MAXVALUE", offload_constants.PART_OUT_OF_RANGE):
            return datetime64(datetime.datetime.max)
        if "TO_DATE(" in token:
            parts = token.split("'")
            return datetime64(parts[1].strip())
        if ORACLE_TYPE_TIMESTAMP in token and "'" in token:
            parts = token.split("'")
            return datetime64(parts[1].strip())
        if not strict:
            return None
        raise NotImplementedError(f"Oracle date/time literal not implemented: {token}")
    if dtype == ORACLE_TYPE_NUMBER:
        if upper_token in ("MAXVALUE", offload_constants.PART_OUT_OF_RANGE):
            return float("inf")
        cleaned = token.strip("'").strip()
        if _INT_RE.match(cleaned):
            return int(cleaned)
        if not strict:
            return None
        raise NotImplementedError(f"Oracle NUMBER literal not implemented: {token}")
    if dtype in (ORACLE_TYPE_VARCHAR2, ORACLE_TYPE_NVARCHAR2, ORACLE_TYPE_CHAR, ORACLE_TYPE_NCHAR, ORACLE_TYPE_VARCHAR):
        if upper_token in ("MAXVALUE", offload_constants.PART_OUT_OF_RANGE):
            return offload_constants.PART_OUT_OF_RANGE
        if token.startswith("'") and token.endswith("'"):
            return token[1:-1]
        return token.strip("'")
    if not strict:
        return str(token)
    raise NotImplementedError(f"Unsupported partition key type: {data_type}")


def parse_high_value_literals(
    hv_string: str | None,
    key_columns: tuple[OracleColumnMeta, ...],
    partition_type: str | None,
    *,
    strict: bool = False,
) -> tuple[tuple[str, ...], tuple[Any, ...]]:
    """Parse an Oracle ``HIGH_VALUE`` string into ``(tokens, python_values)`` tuples."""
    tokens = split_high_value_tokens(hv_string)
    if not tokens or not key_columns or not partition_type:
        return tokens, ()
    ptype = partition_type.upper()
    if ptype == "LIST":
        col = key_columns[0]
        py_vals = tuple(parse_single_high_value_token(tok, col.data_type, strict=strict) for tok in tokens)
        return tokens, py_vals
    if ptype == "RANGE":
        py_list: list[Any] = []
        for idx, col in enumerate(key_columns):
            if idx >= len(tokens):
                break
            py_list.append(parse_single_high_value_token(tokens[idx], col.data_type, strict=strict))
        return tokens, tuple(py_list)
    return tokens, ()


def decode_oracle_raw_stat(raw_bytes: bytes | None, data_type: str) -> Any:
    """Decode Oracle ``ALL_TAB_COLS.LOW_VALUE`` / ``HIGH_VALUE`` internal ``RAW`` bytes in pure Python."""
    if not raw_bytes:
        return None
    dtype = data_type.upper()
    try:
        if dtype in (ORACLE_TYPE_NUMBER, ORACLE_TYPE_FLOAT):
            return _decode_oracle_number_raw(raw_bytes)
        if dtype == ORACLE_TYPE_DATE or dtype.startswith(ORACLE_TYPE_TIMESTAMP):
            return _decode_oracle_date_raw(raw_bytes)
        if dtype in (ORACLE_TYPE_VARCHAR2, ORACLE_TYPE_NVARCHAR2, ORACLE_TYPE_CHAR, ORACLE_TYPE_NCHAR):
            return raw_bytes.decode("utf-8", errors="replace")
        if dtype == ORACLE_TYPE_BINARY_FLOAT and len(raw_bytes) == 4:
            return _decode_oracle_binary_float(raw_bytes)
        if dtype == ORACLE_TYPE_BINARY_DOUBLE and len(raw_bytes) == 8:
            return _decode_oracle_binary_double(raw_bytes)
    except Exception:
        return None
    return None


class OracleCatalogService:
    """Service that loads ``OracleSessionContext`` and ``OracleTableSnapshot`` through SQLSpec drivers."""

    def __init__(self, statements: NamedStatements | None = None) -> None:
        self._statements = statements or NamedStatements(ORACLE_CATALOG_SQL_PATH)

    @property
    def statements(self) -> NamedStatements:
        """Return the ``NamedStatements`` catalog for ``oracle_catalog.sql``."""
        return self._statements

    def load_session_context(self, driver: Any) -> OracleSessionContext:
        """Load session user, DB unique name, version, compatibility, and block size in one query."""
        rows = driver.select(self._statements.text("get_session_context"))
        if not rows:
            return OracleSessionContext(
                session_user="",
                db_unique_name="",
                version="19.0.0.0.0",
                compatible="19.0.0.0.0",
            )
        row = rows[0]
        return OracleSessionContext(
            session_user=str(row["session_user"] or ""),
            db_unique_name=str(row["db_unique_name"] or ""),
            version=str(row["version"] or ""),
            compatible=str(row["compatible"] or row["version"] or ""),
            db_block_size=int(row["db_block_size"] or 8192),
            max_string_size=str(row["max_string_size"] or "STANDARD"),
        )

    def load_object_last_ddl_time(self, driver: Any, owner: str, table_name: str) -> datetime.datetime | None:
        """Return ``MAX(LAST_DDL_TIME)`` from ``ALL_OBJECTS`` for ``owner.table_name``."""
        rows = driver.select(
            self._statements.text("get_object_last_ddl_time"),
            {"owner": owner.upper(), "table_name": table_name.upper()},
        )
        if not rows:
            return None
        return rows[0].get("last_ddl_time")

    def load_table_snapshot(
        self,
        driver: Any,
        owner: str,
        table_name: str,
        *,
        db_block_size: int = 8192,
        snapshot_scn: int | None = None,
    ) -> OracleTableSnapshot:
        """Load a complete ``OracleTableSnapshot`` for ``owner.table_name``."""
        binds = {"owner": owner.upper(), "table_name": table_name.upper()}
        header_rows = driver.select(self._statements.text("get_table_header"), binds)
        col_rows = driver.select(self._statements.text("get_table_columns"), binds)
        columns = tuple(_hydrate_column_meta(row) for row in col_rows)
        if not header_rows:
            return OracleTableSnapshot(
                owner=binds["owner"],
                table_name=binds["table_name"],
                columns=columns,
                snapshot_scn=snapshot_scn,
            )

        hdr = header_rows[0]
        obj_type = hdr.get("object_type")
        partitioned = hdr.get("partitioned") == "YES"
        part_type = hdr.get("partitioning_type")
        subpart_type = hdr.get("subpartitioning_type")
        blocks = int(hdr["blocks"]) if hdr.get("blocks") is not None else None
        num_bytes = (blocks * db_block_size) if blocks is not None else None
        size_in_bytes = int(hdr["size_in_bytes"]) if hdr.get("size_in_bytes") is not None else None
        storage_stats = OracleStorageStats(
            num_rows=int(hdr["num_rows"]) if hdr.get("num_rows") is not None else None,
            blocks=blocks,
            num_bytes=num_bytes,
            avg_row_len=int(hdr["avg_row_len"]) if hdr.get("avg_row_len") is not None else None,
            size_in_bytes=size_in_bytes,
        )

        constraints: tuple[OracleConstraintMeta, ...] = ()
        indexes: tuple[OracleIndexMeta, ...] = ()
        partitions: tuple[OraclePartitionMeta, ...] = ()
        subpartitions: tuple[OracleSubpartitionMeta, ...] = ()

        if obj_type == "TABLE":
            cons_rows = driver.select(self._statements.text("get_table_constraints"), binds)
            idx_rows = driver.select(self._statements.text("get_table_indexes"), binds)
            constraints = _group_constraints(cons_rows)
            indexes = _group_indexes(idx_rows)
            if partitioned:
                part_key_cols = tuple(
                    sorted(
                        (c for c in columns if c.partition_position is not None),
                        key=lambda c: c.partition_position or 0,
                    )
                )
                subpart_key_cols = tuple(
                    sorted(
                        (c for c in columns if c.subpartition_position is not None),
                        key=lambda c: c.subpartition_position or 0,
                    )
                )
                part_rows = driver.select(self._statements.text("get_table_partitions"), binds)
                if subpart_type and subpart_type != "NONE":
                    subpart_rows = driver.select(self._statements.text("get_table_subpartitions"), binds)
                else:
                    subpart_rows = []
                partitions, subpartitions = _assemble_partitions(
                    part_rows,
                    subpart_rows,
                    part_key_cols,
                    subpart_key_cols,
                    part_type,
                    subpart_type,
                )

        return OracleTableSnapshot(
            owner=binds["owner"],
            table_name=binds["table_name"],
            object_type=obj_type,
            iot_type=hdr.get("iot_type"),
            partitioned=partitioned,
            partitioning_type=part_type,
            subpartitioning_type=subpart_type,
            dependencies_enabled=hdr.get("dependencies") == "ENABLED",
            degree=hdr.get("degree"),
            compression_enabled=hdr.get("compression_enabled") == "ENABLED",
            storage_stats=storage_stats,
            columns=columns,
            constraints=constraints,
            indexes=indexes,
            partitions=partitions,
            subpartitions=subpartitions,
            last_ddl_time=hdr.get("last_ddl_time"),
            snapshot_scn=snapshot_scn,
        )


def _map_open_partition_token(token: str) -> str:
    upper = token.upper()
    if upper == "MAXVALUE":
        return offload_constants.PART_OUT_OF_RANGE
    if upper == "DEFAULT":
        return offload_constants.PART_OUT_OF_LIST
    return token


def _hydrate_column_meta(row: dict[str, Any]) -> OracleColumnMeta:
    dtype = str(row["data_type"])
    low_raw = bytes(row["low_value"]) if row.get("low_value") is not None else None
    high_raw = bytes(row["high_value"]) if row.get("high_value") is not None else None
    return OracleColumnMeta(
        name=str(row["column_name"]).upper(),
        column_id=int(row["column_id"]),
        nullable=str(row.get("nullable", "Y")).upper() in ("Y", "YES", "TRUE"),
        data_type=dtype,
        data_precision=int(row["data_precision"]) if row.get("data_precision") is not None else None,
        data_scale=int(row["data_scale"]) if row.get("data_scale") is not None else None,
        byte_length=int(row["byte_length"]) if row.get("byte_length") is not None else None,
        data_default=str(row["data_default"]) if row.get("data_default") is not None else None,
        hidden_column=str(row.get("hidden_column") or "NO"),
        char_used=str(row["char_used"]) if row.get("char_used") is not None else None,
        char_length=int(row["char_length"]) if row.get("char_length") is not None else None,
        num_distinct=int(row["num_distinct"]) if row.get("num_distinct") is not None else None,
        num_nulls=int(row["num_nulls"]) if row.get("num_nulls") is not None else None,
        avg_col_len=int(row["avg_col_len"]) if row.get("avg_col_len") is not None else None,
        low_value_raw=low_raw,
        high_value_raw=high_raw,
        low_value=decode_oracle_raw_stat(low_raw, dtype),
        high_value=decode_oracle_raw_stat(high_raw, dtype),
        partition_position=int(row["partition_position"]) if row.get("partition_position") is not None else None,
        subpartition_position=(
            int(row["subpartition_position"]) if row.get("subpartition_position") is not None else None
        ),
    )


def _group_constraints(rows: list[dict[str, Any]]) -> tuple[OracleConstraintMeta, ...]:
    grouped: dict[tuple[str, str], list[str]] = {}
    for row in rows:
        key = (str(row["constraint_name"]), str(row["constraint_type"]))
        grouped.setdefault(key, []).append(str(row["column_name"]).upper())
    return tuple(
        OracleConstraintMeta(constraint_name=name, constraint_type=ctype, column_names=tuple(cols))
        for (name, ctype), cols in grouped.items()
    )


def _group_indexes(rows: list[dict[str, Any]]) -> tuple[OracleIndexMeta, ...]:
    grouped: dict[tuple[str, str, str], list[str]] = {}
    for row in rows:
        key = (str(row["index_name"]), str(row.get("index_type") or ""), str(row.get("uniqueness") or ""))
        grouped.setdefault(key, []).append(str(row["column_name"]).upper())
    return tuple(
        OracleIndexMeta(index_name=name, index_type=itype, uniqueness=uniq, column_names=tuple(cols))
        for (name, itype, uniq), cols in grouped.items()
    )


def _assemble_partitions(
    part_rows: list[dict[str, Any]],
    subpart_rows: list[dict[str, Any]],
    part_key_cols: tuple[OracleColumnMeta, ...],
    subpart_key_cols: tuple[OracleColumnMeta, ...],
    part_type: str | None,
    subpart_type: str | None,
) -> tuple[tuple[OraclePartitionMeta, ...], tuple[OracleSubpartitionMeta, ...]]:
    subparts_by_part: dict[str, list[OracleSubpartitionMeta]] = {}
    all_subparts: list[OracleSubpartitionMeta] = []
    for srow in subpart_rows:
        pname = str(srow["partition_name"])
        hv_csv = str(srow["high_value"]) if srow.get("high_value") is not None else None
        tokens, py_vals = parse_high_value_literals(hv_csv, subpart_key_cols, subpart_type, strict=False)
        meta = OracleSubpartitionMeta(
            partition_name=pname,
            partition_position=int(srow["partition_position"]),
            subpartition_count=int(srow.get("subpartition_count") or 0),
            subpartition_name=str(srow["subpartition_name"]),
            subpartition_position=int(srow["subpartition_position"]),
            high_value_csv=hv_csv,
            high_value_tokens=tokens,
            high_value_python=py_vals,
            size_in_bytes=int(srow.get("bytes") or 0),
            num_rows=int(srow.get("num_rows") or 0),
            blocks=int(srow["blocks"]) if srow.get("blocks") is not None else None,
            avg_row_len=int(srow["avg_row_len"]) if srow.get("avg_row_len") is not None else None,
        )
        subparts_by_part.setdefault(pname, []).append(meta)
        all_subparts.append(meta)

    partitions: list[OraclePartitionMeta] = []
    for prow in part_rows:
        pname = str(prow["partition_name"])
        hv_csv = str(prow["high_value"]) if prow.get("high_value") is not None else None
        tokens, py_vals = parse_high_value_literals(hv_csv, part_key_cols, part_type, strict=False)
        child_subparts = tuple(subparts_by_part.get(pname, ()))
        sub_bytes = sum(sp.size_in_bytes for sp in child_subparts)
        sub_rows = sum(sp.num_rows for sp in child_subparts)
        p_bytes = int(prow.get("bytes") or 0) or sub_bytes
        p_rows = int(prow["num_rows"]) if prow.get("num_rows") is not None else sub_rows
        partitions.append(
            OraclePartitionMeta(
                partition_name=pname,
                partition_position=int(prow["partition_position"]),
                subpartition_count=int(prow.get("subpartition_count") or len(child_subparts)),
                high_value_csv=hv_csv,
                high_value_tokens=tokens,
                high_value_python=py_vals,
                size_in_bytes=p_bytes,
                num_rows=p_rows,
                blocks=int(prow["blocks"]) if prow.get("blocks") is not None else None,
                avg_row_len=int(prow["avg_row_len"]) if prow.get("avg_row_len") is not None else None,
                subpartitions=child_subparts,
            )
        )
    all_subparts.sort(key=lambda sp: sp.high_value_python, reverse=True)
    return tuple(partitions), tuple(all_subparts)


def _decode_oracle_number_raw(raw_bytes: bytes) -> Decimal | int:
    if raw_bytes == b"\x80":
        return 0
    first = raw_bytes[0]
    if first & 0x80:
        exponent = (first & 0x7F) - 65
        digits = [b - 1 for b in raw_bytes[1:]]
        sign = 1
    else:
        exponent = 62 - first
        mantissa_bytes = raw_bytes[1:-1] if raw_bytes[-1] == 102 else raw_bytes[1:]
        digits = [101 - b for b in mantissa_bytes]
        sign = -1
    value = Decimal(0)
    for idx, digit in enumerate(digits):
        value += Decimal(digit) * (Decimal(100) ** (exponent - idx))
    signed = value * sign
    if signed == signed.to_integral_value():
        return int(signed)
    return signed


def _decode_oracle_date_raw(raw_bytes: bytes) -> datetime64 | None:
    if len(raw_bytes) < 7:
        return None
    year = (raw_bytes[0] - 100) * 100 + (raw_bytes[1] - 100)
    month = raw_bytes[2]
    day = raw_bytes[3]
    hour = raw_bytes[4] - 1
    minute = raw_bytes[5] - 1
    second = raw_bytes[6] - 1
    iso = f"{year:04d}-{month:02d}-{day:02d}T{hour:02d}:{minute:02d}:{second:02d}"
    if len(raw_bytes) >= 11:
        nanos = int.from_bytes(raw_bytes[7:11], "big")
        iso = f"{iso}.{nanos:09d}"
    return datetime64(iso)


def _decode_oracle_binary_float(raw_bytes: bytes) -> float:
    b = bytearray(raw_bytes)
    if b[0] & 0x80:
        b[0] &= 0x7F
    else:
        for i in range(4):
            b[i] ^= 0xFF
    return float(struct.unpack(">f", bytes(b))[0])


def _decode_oracle_binary_double(raw_bytes: bytes) -> float:
    b = bytearray(raw_bytes)
    if b[0] & 0x80:
        b[0] &= 0x7F
    else:
        for i in range(8):
            b[i] ^= 0xFF
    return float(struct.unpack(">d", bytes(b))[0])
