    @staticmethod
    def _uuid_as_uuid(u: uuid.UUID | None) -> uuid.UUID | None:
        """Validate that u is a uuid.UUID instance when provided."""
        if u is None:
            return None
        assert isinstance(u, uuid.UUID)
        return u

        """Construct an ExecutionId from an existing uuid.UUID instance."""
