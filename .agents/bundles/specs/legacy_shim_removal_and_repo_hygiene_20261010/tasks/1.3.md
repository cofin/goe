```python
# SPDX-FileCopyrightText: 2026 The GOE Authors
# SPDX-License-Identifier: Apache-2.0

_ANSI_COLORS: dict[str, str] = {
    "grey": "\033[90m",
    "red": "\033[31m",
    "green": "\033[32m",
    "yellow": "\033[33m",
    "blue": "\033[34m",
    "magenta": "\033[35m",
    "cyan": "\033[36m",
    "white": "\033[37m",
}
_ANSI_ATTRS: dict[str, str] = {
    "bold": "\033[1m",
    "underline": "\033[4m",
}
_ANSI_RESET = "\033[0m"


def colored(text: str, color: str | None = None, attrs: list[str] | None = None) -> str:
    """Wrap text in ANSI color and attribute escape sequences."""
    prefix = ""
    if color and color in _ANSI_COLORS:
        prefix += _ANSI_COLORS[color]
    if attrs:
        for attr in attrs:
            if attr in _ANSI_ATTRS:
                prefix += _ANSI_ATTRS[attr]
    if not prefix:
        return str(text)
    return f"{prefix}{text}{_ANSI_RESET}"


def cprint(text: str, color: str | None = None, attrs: list[str] | None = None) -> None:
    """Print ANSI-colored text to stdout."""
    print(colored(text, color=color, attrs=attrs))
```

### Step 4: Ensure `lark>=1.1.9` compatibility in `src/goe/offload/predicate_offload.py`
In `src/goe/offload/predicate_offload.py:147-183` (`handle_parse_errors.__exit__`), ensure that `considered_tokens`, `considered_rules`, and `expected` attributes on `lark` exceptions are safely extracted under `lark>=1.1.9`:
```python
# SPDX-FileCopyrightText: 2026 The GOE Authors
# SPDX-License-Identifier: Apache-2.0

class handle_parse_errors:
    """Context manager converting Lark parse and visit exceptions into OffloadPredicatesException."""

    def __init__(self, logger=None):
        self.logger = logger

    def __enter__(self):
        return self

    def __exit__(self, exc_type, exc, tb):
        if exc_type and issubclass(exc_type, VisitError):
            raise exc.orig_exc from exc
        if exc_type and issubclass(exc_type, Exception):
            if hasattr(exc, "considered_tokens"):
                expected_rules = {
                    t.rule.name for t in exc.considered_tokens if getattr(t, "rule", None)
                }
            elif hasattr(exc, "considered_rules"):
                expected_rules = {str(r.origin.name) for r in exc.considered_rules if getattr(r, "origin", None)}
            elif hasattr(exc, "expected"):
                expected_rules = {str(token_name) for token_name in exc.expected}
            else:
                expected_rules = None

            if expected_rules is not None:
                context = exc.get_context(self.text) if hasattr(self, "text") and hasattr(exc, "get_context") else ""
                message = f"{context}\nExpected one of: {', '.join(sorted(expected_rules))}"
                raise OffloadPredicatesException(message) from exc
            raise OffloadPredicatesException(str(exc)) from exc
```
